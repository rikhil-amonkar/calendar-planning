      import json

      travel_dict = { ... }

      friends_data = [ ... ]

      # Create optimizer
      opt = Optimize()

      # Create variables for each friend
      meets = []
      starts = []
      ends = []
      for i, friend in enumerate(friends_data):
          name = friend['name']
          meets.append(Bool('meet_'+name))
          starts.append(Int('start_'+name))
          ends.append(Int('end_'+name))

      # Constraints for each friend
      for i, friend in enumerate(friends_data):
          # If we meet this friend, then the meeting must be within the available time and have at least the minimum duration
          opt.add(If(meets[i], 
                     And(starts[i] >= friend['available_start'],
                         ends[i] <= friend['available_end'],
                         ends[i] - starts[i] >= friend['min_duration'],
                         starts[i] >= 540 + get_travel_time("Chinatown", friend['location']) # we start at Chinatown at 540
                     ), 
                     True)) # if not meeting, no constraints

      # Constraints for overlapping meetings: for every pair of friends, if both are met, then one must finish and travel to the other before the other starts.
      n = len(friends_data)
      for i in range(n):
          for j in range(i+1, n):
              # Only if both are met
              cond = And(meets[i], meets[j])
              # The disjunctive constraint
              travel_ij = get_travel_time(friends_data[i]['location'], friends_data[j]['location'])
              travel_ji = get_travel_time(friends_data[j]['location'], friends_data[i]['location'])
              constraint = Or(ends[i] + travel_ij <= starts[j], ends[j] + travel_ji <= starts[i])
              opt.add(If(cond, constraint, True))

      # Maximize the number of meetings
      total_meetings = Sum([If(meet, 1, 0) for meet in meets])
      opt.maximize(total_meetings)

      # Check and get model
      if opt.check() == sat:
          m = opt.model()
          # Extract the meetings that are scheduled
          itinerary = []
          for i, friend in enumerate(friends_data):
              if is_true(m.evaluate(meets[i])):
                  start_val = m.evaluate(starts[i])
                  end_val = m.evaluate(ends[i])
                  # Convert to integers
                  start_min = start_val.as_long()
                  end_min = end_val.as_long()
                  itinerary.append({
                      "action": "meet",
                      "location": friend['location'],
                      "person": friend['name'],
                      "start_time": min_to_time(start_min),
                      "end_time": min_to_time(end_min)
                  })
          # Sort itinerary by start_time
          itinerary.sort(key=lambda x: x['start_time'])
          # Output as JSON
          result = {"itinerary": itinerary}
          print(json.dumps(result, indent=2))
      else:
