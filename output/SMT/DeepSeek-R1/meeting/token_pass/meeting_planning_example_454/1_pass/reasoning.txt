   end = [Int(f"end_{i}") for i in range(n)]

   for i in range(n):

        # If we meet, then constraints on time window and duration
        opt.add(Implies(meet[i], start[i] >= info['avail_start']))
        opt.add(Implies(meet[i], end[i] <= info['avail_end']))
        opt.add(Implies(meet[i], end[i] - start[i] >= info['min_dur']))
        # Also, start must be at least the travel time from Presidio to this location

   for i in range(n):

   opt.maximize(Sum([If(meet[i], 1, 0) for i in range(n)]))

   if opt.check() == sat:
