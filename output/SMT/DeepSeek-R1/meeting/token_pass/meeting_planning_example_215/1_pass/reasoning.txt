    # Define the travel matrix
    travel = [
        [0, 19, 25, 25],
        [21, 0, 21, 6],
        [26, 19, 0, 18],
        [26, 8, 18, 0]
    ]

    # Map each person to their location index and available time window (in minutes from 9:00)
    people = [
        {'name': 'Jessica', 'loc': 1, 'start_avail': 465, 'end_avail': 600, 'min_dur': 30},
        {'name': 'Sandra',  'loc': 2, 'start_avail': 570, 'end_avail': 705, 'min_dur': 120},
        {'name': 'Jason',   'loc': 3, 'start_avail': 420, 'end_avail': 465, 'min_dur': 30}
    ]

    # Create variables for each person
    meet = [Bool(f"meet_{p['name']}") for p in people]
    start = [Real(f"start_{p['name']}") for p in people]
    end = [Real(f"end_{p['name']}") for p in people]
    order = [Int(f"order_{p['name']}") for p in people]

    s = Optimize()

    # Constraints for each person
    for i, p in enumerate(people):
        # If we meet this person, then constraints on time and order
        s.add(Implies(meet[i], start[i] >= p['start_avail']))
        s.add(Implies(meet[i], end[i] <= p['end_avail']))
        s.add(Implies(meet[i], end[i] - start[i] >= p['min_dur']))
        s.add(Implies(meet[i], order[i] >= 1))
        s.add(Implies(meet[i], order[i] <= 3))
        s.add(Implies(Not(meet[i]), order[i] == 0))

    # All non-zero orders are distinct
    s.add(Distinct([If(meet[i], order[i], 0) for i in range(3)]))

    # Constraints for travel and order
    for i in range(3):
        for j in range(3):
            if i == j:
                continue
            # If both meetings are scheduled and order[i] < order[j], then travel constraint
            cond = And(meet[i], meet[j], order[i] < order[j])
            # Travel time from person i's location to person j's location
            travel_ij = travel[people[i]['loc']][people[j]['loc']]
            s.add(Implies(cond, end[i] + travel_ij <= start[j]))

    # For the first meeting: if a meeting is scheduled and its order is 1, then we must have left Bayview at time 0 and traveled to that location.
    for i in range(3):
        cond = And(meet[i], order[i] == 1)
        travel_from_bayview = travel[0][people[i]['loc']]  # from Bayview (index0) to person i's location
        s.add(Implies(cond, start[i] >= travel_from_bayview))

    # We want to maximize the number of meetings
    total_meetings = Sum([If(meet[i], 1, 0) for i in range(3)])
    s.maximize(total_meetings)

    # Check and get the model

        # Create a list of meetings that happened with their details.

                end_val = m.evaluate(end[i])
                order_val = m.evaluate(order[i])
                # Convert to float and then to time string

                start_minutes = start_val.as_fraction() if isinstance(start_val, RatNumRef) else float(start_val)

                scheduled_meetings.append((order_val, p['name'], people[i]['loc'], start_minutes, end_minutes))
