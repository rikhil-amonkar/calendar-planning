        opt.add(If(meet[i],
                   And(S[i] >= data['avail_start'],
                       S[i] + data['min_duration'] <= data['avail_end'],

            )

            else:

    # Maximize number of meetings
    opt.maximize(Sum([If(meet[i], 1, 0) for i in range(n_real)]))
