    # Meeting variables for friends (index 0 to 9)
    met = [Bool(f'met_{i}') for i in range(10)]
    start = [Int(f'start_{i}') for i in range(10)]
    end = [Int(f'end_{i}') for i in range(10)]

    # Dummy meeting at start: Nob Hill at 9:00 (540 minutes)
    dummy_location = 'Nob Hill'
    dummy_start = 540
    dummy_end = 540

    # Constraints for each friend
    for i, (name, loc, avail_start, avail_end, min_dur) in enumerate(friends):
        opt.add(Implies(met[i], start[i] >= avail_start))
        opt.add(Implies(met[i], end[i] <= avail_end))
        opt.add(Implies(met[i], end[i] - start[i] >= min_dur))

    # Constraints for travel between meetings (including dummy)
    # For each pair of meetings (dummy and friends, and between friends)
    meetings = [(-1, dummy_location, dummy_start, dummy_end)] + list(enumerate(friends))
    for idx1, (i, loc1, s1, e1) in enumerate(meetings):
        for idx2, (j, loc2, s2, e2) in enumerate(meetings):
            if idx1 >= idx2:
                continue
            if i == -1 and j == -1:
                continue
            # For dummy and friend j
            if i == -1:
                # j is a friend index
                opt.add(Implies(met[j], start[j] >= dummy_end + travel_times[dummy_location][loc2]))

                # i is a friend index
                opt.add(Implies(met[i], end[i] + travel_times[loc1][dummy_location] <= dummy_start))

            else:
                # Both i and j are friends
                opt.add(Implies(And(met[i], met[j]),
                               Or(

                                   end[j] + travel_times[loc2][loc1] <= start[i]
                               )))

    # Maximize the number of meetings
    opt.maximize(Sum([If(met[i], 1, 0) for i in range(10)]))

        for i, (name, location, _, _, _) in enumerate(friends):

                end_str = f"{e_hours}:{e_minutes:02d}"

                    "person": name,

                    "end_time": end_str
                })
