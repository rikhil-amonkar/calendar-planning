    from z3 import *
    import json

    # Define travel times
    travel_times = { ... }

    # Friends data (without dummy)
    friends = [
        {'name': 'Mary', 'location': 'Golden Gate Park', 'available_start': -15, 'available_end': 165, 'min_duration': 45},
        {'name': 'Kevin', 'location': 'Haight-Ashbury', 'available_start': 75, 'available_end': 435, 'min_duration': 90},
        {'name': 'Deborah', 'location': 'Bayview', 'available_start': 360, 'available_end': 615, 'min_duration': 120},
        {'name': 'Stephanie', 'location': 'Presidio', 'available_start': 60, 'available_end': 495, 'min_duration': 120},
        {'name': 'Emily', 'location': 'Financial District', 'available_start': 150, 'available_end': 765, 'min_duration': 105}
    ]

    # Create solver and variables
    opt = Optimize()
    n = len(friends)

    # Meet variables for each friend
    meet = [Bool(f"meet_{i}") for i in range(n)]
    start = [Real(f"start_{i}") for i in range(n)]
    end = [Real(f"end_{i}") for i in range(n)]

    # Dummy meeting for the start
    dummy_location = 'Embarcadero'
    dummy_start = 0
    dummy_end = 0

    # For each friend, set constraints if meeting
    for i, f in enumerate(friends):
        # If we meet this friend, then the meeting must be within available time and have the minimum duration
        opt.add(Implies(meet[i], And(start[i] >= f['available_start'], end[i] <= f['available_end'], end[i] == start[i] + f['min_duration'])))
        # Also, the start time must be at least the travel time from Embarcadero to the friend's location
        opt.add(Implies(meet[i], start[i] >= travel_times[dummy_location][f['location']]))

    # For each pair of friends, if both are met, then they must not overlap and account for travel time
    for i in range(n):
        for j in range(i+1, n):
            # If both meetings i and j are held, then one must be after the other with travel time
            travel_ij = travel_times[friends[i]['location']][friends[j]['location']]
            travel_ji = travel_times[friends[j]['location']][friends[i]['location']]
            opt.add(Implies(And(meet[i], meet[j]),
                            Or(start[j] >= end[i] + travel_ij, start[i] >= end[j] + travel_ji)
                          ))

    # Maximize the number of meetings
    opt.maximize(Sum([If(meet[i], 1, 0) for i in range(n)]))

    # Check and get model
    if opt.check() == sat:
        m = opt.model()
        # Extract the meetings that are scheduled
        scheduled_meetings = []
        for i, f in enumerate(friends):
            if is_true(m.evaluate(meet[i])):
                start_val = m.evaluate(start[i])
                # Convert Z3 rational to float and then to integer minutes
                start_minutes = float(start_val.as_fraction())
                end_minutes = start_minutes + f['min_duration']
                scheduled_meetings.append({
                    'name': f['name'],
                    'location': f['location'],
                    'start': start_minutes,
                    'end': end_minutes
                })
        # Sort scheduled_meetings by start time
        scheduled_meetings.sort(key=lambda x: x['start'])

        # Convert minutes to time string (from 9:00 base)
        def minutes_to_time(minutes):
            # minutes is from 9:00
            total_minutes = int(round(minutes))
            hours = total_minutes // 60
            minutes = total_minutes % 60
            # Since we started at 9:00, we add hours to 9:00
            actual_hour = 9 + hours
            actual_minute = minutes
            # Format to string without leading zero
            return f"{actual_hour}:{actual_minute:02d}"

        itinerary = []
        for meeting in scheduled_meetings:
            itinerary.append({
                "action": "meet",
                "location": meeting['location'],
                "person": meeting['name'],
                "start_time": minutes_to_time(meeting['start']),
                "end_time": minutes_to_time(meeting['end'])
            })

        result = {"itinerary": itinerary}
        print(json.dumps(result, indent=2))
    else:
        print('{"itinerary": []}')
