        ('Golden Gate Park', 'The Castro'): 13,

    }
    
    friends = [
        {'name': 'Emily', 'location': 'Russian Hill', 'start_avail': 735, 'end_avail': 855, 'duration': 105},
        {'name': 'Mark', 'location': 'Presidio', 'start_avail': 885, 'end_avail': 1170, 'duration': 60},

        {'name': 'Steven', 'location': 'Golden Gate Park', 'start_avail': 675, 'end_avail': 1275, 'duration': 105}
    ]
    
    s = Optimize()
    meet_vars = [Bool(f"meet_{i}") for i in range(len(friends))]
    start_vars = [Int(f"start_{i}") for i in range(len(friends))]
    
    for i, friend in enumerate(friends):
        s.add(Implies(meet_vars[i], start_vars[i] >= friend['start_avail']))
        s.add(Implies(meet_vars[i], start_vars[i] + friend['duration'] <= friend['end_avail']))
        s.add(Implies(meet_vars[i], start_vars[i] >= 540 + travel_times[('Alamo Square', friend['location'])]))
    
    for i in range(len(friends)):
        for j in range(i + 1, len(friends)):
            s.add(Implies(And(meet_vars[i], meet_vars[j]),
                          Or(
                              start_vars[i] >= start_vars[j] + friends[j]['duration'] + travel_times[(friends[j]['location'], friends[i]['location'])],
                              start_vars[j] >= start_vars[i] + friends[i]['duration'] + travel_times[(friends[i]['location'], friends[j]['location'])]
                          )))
    
    s.maximize(Sum([If(meet_vars[i], 1, 0) for i in range(len(friends))]))
    
    if s.check() == sat:
        m = s.model()
        itinerary = []
        for i, friend in enumerate(friends):
            if m.eval(meet_vars[i]):
                start_val = m.eval(start_vars[i]).as_long()
                end_val = start_val + friend['duration']
                start_str = f"{start_val // 60}:{start_val % 60:02d}"
                end_str = f"{end_val // 60}:{end_val % 60:02d}"
                itinerary.append({
                    "action": "meet",
                    "location": friend['location'],
                    "person": friend['name'],
                    "start_time": start_str,
                    "end_time": end_str
                })
        itinerary.sort(key=lambda x: x['start_time'])
        print(json.dumps({"itinerary": itinerary}, indent=2))
    else:
        print('{"itinerary": []}')

if __name__ == "__main__":
    main()
