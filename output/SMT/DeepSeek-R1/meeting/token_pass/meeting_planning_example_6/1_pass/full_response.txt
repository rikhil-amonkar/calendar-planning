```python
import json
from z3 import *

def main():
    # Convert all times to minutes since midnight
    start_fw = 9 * 60  # 9:00 AM
    kenneth_start = 14 * 60 + 15  # 2:15 PM
    kenneth_end = 19 * 60 + 45  # 7:45 PM
    travel_time = 11  # minutes

    # Create Z3 variables
    meet_start = Int('meet_start')
    meet_duration = Int('meet_duration')
    leave_fw = Int('leave_fw')

    # Initialize solver with optimization
    opt = Optimize()
    
    # Add constraints
    opt.add(leave_fw >= start_fw)  # Can't leave before arrival
    opt.add(meet_start == leave_fw + travel_time)  # Arrival at Nob Hill
    opt.add(meet_start >= kenneth_start)  # Can't meet before Kenneth arrives
    opt.add(meet_start + meet_duration <= kenneth_end)  # Must finish before Kenneth leaves
    opt.add(meet_duration >= 90)  # Minimum meeting duration
    
    # Maximize meeting duration
    opt.maximize(meet_duration)
    
    # Check satisfiability
    if opt.check() == sat:
        m = opt.model()
        start_min = m.eval(meet_start).as_long()
        duration_min = m.eval(meet_duration).as_long()
        end_min = start_min + duration_min
        
        # Convert minutes to time strings
        def format_time(minutes):
            hours = minutes // 60
            mins = minutes % 60
            return f"{hours}:{mins:02d}"
        
        # Create itinerary
        itinerary = [{
            "action": "meet",
            "location": "Nob Hill",
            "person": "Kenneth",
            "start_time": format_time(start_min),
            "end_time": format_time(end_min)
        }]
        
        # Output JSON result
        result = {"itinerary": itinerary}
        print(json.dumps(result, indent=2))
    else:
        print('{"itinerary": []}')

if __name__ == "__main__":
    main()
```
