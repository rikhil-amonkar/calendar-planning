from z3 import *
import json

def time_to_minutes(time_str):
    parts = time_str.split(':')
    hours = int(parts[0])
    minutes = int(parts[1])
    return hours * 60 + minutes

def minutes_to_time(minutes):
    total_minutes = minutes
    hours = total_minutes // 60
    minutes = total_minutes % 60
    return f"{hours}:{minutes:02d}"

for f in friends:
    f['start_avail_min'] = time_to_minutes(f['start_avail']) - 9*60
    f['end_avail_min'] = time_to_minutes(f['end_avail']) - 9*60

meet = [Bool(f"meet_{i}") for i in range(len(friends))]
start = [Real(f"start_{i}") for i in range(len(friends))]
end = [Real(f"end_{i}") for i in range(len(friends))]

solver = Optimize()

for i, f in enumerate(friends):
    solver.add(Implies(meet[i], start[i] >= f['start_avail_min']))
    solver.add(Implies(meet[i], end[i] <= f['end_avail_min']))
    solver.add(Implies(meet[i], end[i] - start[i] >= f['min_duration']))
    solver.add(Implies(meet[i], start[i] >= travel_times['Richmond District'][f['location']]))

for i in range(len(friends)):
    for j in range(i+1, len(friends)):
        both_meet = And(meet[i], meet[j])
        time_ij = travel_times[friends[i]['location']][friends[j]['location']]
        time_ji = travel_times[friends[j]['location']][friends[i]['location']]
        constraint = Or(
            end[i] + time_ij <= start[j],
            end[j] + time_ji <= start[i]
        )
        solver.add(Implies(both_meet, constraint))

objective = Sum([If(meet[i], 1, 0) for i in range(len(friends))])
solver.maximize(objective)

if solver.check() == sat:
    model = solver.model()
    itinerary = []
    for i, f in enumerate(friends):
        if is_true(model.eval(meet[i])):
            start_val = model.eval(start[i])
            end_val = model.eval(end[i])
            start_min = int(str(start_val).split('.')[0])
            end_min = int(str(end_val).split('.')[0])
            start_abs = 9*60 + start_min
            end_abs = 9*60 + end_min
            start_time = minutes_to_time(start_abs)
            end_time = minutes_to_time(end_abs)
            itinerary.append({
                "action": "meet",
                "location": f['location'],
                "person": f['name'],
                "start_time": start_time,
                "end_time": end_time
            })
    itinerary.sort(key=lambda x: time_to_minutes(x['start_time']))
    print(json.dumps({"itinerary": itinerary}, indent=2))
else:
    print('{"itinerary": []}')
