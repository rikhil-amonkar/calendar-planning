    ]

    n = len(friends)
    s = Solver()

    # Decision variables for each friend
    meet = [Bool(f"meet_{i}") for i in range(n)]
    start = [Int(f"start_{i}") for i in range(n)]
    end = [Int(f"end_{i}") for i in range(n)]
    order = [Int(f"order_{i}") for i in range(n)]

    # Dummy start at The Castro at 9:00
    start_dummy = 9*60
    end_dummy = 9*60

    # Constraints for each friend
    for i, (name, loc, avail_start, avail_end, min_dur) in enumerate(friends):
        s.add(If(meet[i], start[i] >= avail_start, True))
        s.add(If(meet[i], end[i] <= avail_end, True))
        s.add(If(meet[i], end[i] == start[i] + min_dur, True))
        s.add(If(meet[i], order[i] >= 0, order[i] == -1))
        s.add(order[i] < n)

    # Distinct orders for meetings that are held
    for i in range(n):
        for j in range(i+1, n):
            s.add(If(And(meet[i], meet[j]), order[i] != order[j], True))

    # Travel constraints from dummy start and between meetings
    for i in range(n):
        # If meeting i is held and is first (order[i]==0), then start[i] >= end_dummy + travel from The Castro to loc
        s.add(If(And(meet[i], order[i] == 0), 
                 start[i] >= end_dummy + travel_time["The Castro"][friends[i][1]], 
                 True))
        
        for j in range(n):
            if i != j:
                # If both meetings are held and order[i] < order[j], then start[j] >= end[i] + travel time from i to j
                s.add(If(And(meet[i], meet[j], order[i] < order[j]),
                         start[j] >= end[i] + travel_time[friends[i][1]][friends[j][1]],
                         True))

    # Maximize the number of meetings
    opt = Optimize()
    opt.add(s.assertions())
    opt.maximize(Sum([If(meet[i], 1, 0) for i in range(n)]))
    
    if opt.check() == sat:
        m = opt.model()
        scheduled_meetings = []
        for i, (name, loc, _, _, _) in enumerate(friends):
            if is_true(m.evaluate(meet[i])):
                start_val = m.evaluate(start[i]).as_long()
                end_val = m.evaluate(end[i]).as_long()
                # Convert minutes to time string
                start_hour = start_val // 60
                start_min = start_val % 60
                end_hour = end_val // 60
                end_min = end_val % 60
                start_str = f"{start_hour}:{start_min:02d}"
                end_str = f"{end_hour}:{end_min:02d}"
                scheduled_meetings.append({
                    "action": "meet",
                    "location": loc,
                    "person": name,
                    "start_time": start_str,
                    "end_time": end_str
                })
        # Sort meetings by start time
        scheduled_meetings.sort(key=lambda x: x['start_time'])
        result = {"itinerary": scheduled_meetings}
        print(json.dumps(result, indent=2))
    else:
        print('{"itinerary": []}')
