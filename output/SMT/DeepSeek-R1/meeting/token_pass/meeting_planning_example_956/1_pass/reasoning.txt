     import z3
     from z3 import *

     # Define the list of friends
     friends = [
         # name, location, available_start (min from 9:00), available_end, required_duration
         ("William", "Alamo Square", ...),
         ...
     ]

     # Hardcode travel time dictionary
     travel_dict = {
         ("The Castro", "Alamo Square"): 8,
         ... # all pairs
     }

     # We'll create a function to get travel time between two locations
     def get_travel_time(loc1, loc2):
         # if loc1 == loc2, return 0
         if loc1 == loc2:
             return 0
         key = (loc1, loc2)
         if key in travel_dict:
             return travel_dict[key]
         # try reverse
         key2 = (loc2, loc1)
         if key2 in travel_dict:
             return travel_dict[key2]
         # else error
         raise ValueError(f"Travel time between {loc1} and {loc2} not found")

     # Initialize Z3 solver and variables
     opt = Optimize()
     n = len(friends)
     met = [Bool(f"met_{i}") for i in range(n)]
     start = [Real(f"start_{i}") for i in range(n)]
     end = [Real(f"end_{i}") for i in range(n)]

     # Add constraints for each friend
     for i in range(n):
         name, loc, avail_start, avail_end, dur = friends[i]
         # If met[i] is true, then meeting constraints
         opt.add(Implies(met[i], start[i] >= avail_start))
         opt.add(Implies(met[i], end[i] <= avail_end))
         opt.add(Implies(met[i], end[i] == start[i] + dur))

         # Constraint from the start location (The Castro) to this meeting
         travel_from_start = get_travel_time("The Castro", loc)
         opt.add(Implies(met[i], start[i] >= travel_from_start))

     # Add pairwise constraints for every two different friends
     for i in range(n):
         for j in range(i+1, n):
             # If both meetings are held, then one must be after the other with travel time
             loc_i = friends[i][1]
             loc_j = friends[j][1]
             travel_ij = get_travel_time(loc_i, loc_j)
             travel_ji = get_travel_time(loc_j, loc_i)

             # Condition for i before j
             cond1 = And(met[i], met[j], end[i] + travel_ij <= start[j])
             # Condition for j before i
             cond2 = And(met[i], met[j], end[j] + travel_ji <= start[i])
             # Either condition must hold
             opt.add(Or(cond1, cond2))

     # Maximize the number of meetings
     total_met = Sum([If(met[i], 1, 0) for i in range(n)])
     opt.maximize(total_met)

     # Check and get the model
     if opt.check() == sat:
         m = opt.model()
         # Extract the meetings that are met and their times
         itinerary = []
         for i in range(n):
             if is_true(m.evaluate(met[i])):
                 start_val = m.evaluate(start[i])
                 end_val = m.evaluate(end[i])
                 # Convert to minutes (as integers) from real numbers
                 # We assume the model gives integers
                 start_min = start_val.as_long() if isinstance(start_val, IntNumRef) else round(float(str(start_val)))
                 end_min = end_val.as_long() if isinstance(end_val, IntNumRef) else round(float(str(end_val)))
                 # Convert minutes to time string

                 end_str = minutes_to_time(end_min)

                     "end_time": end_str
                 })
         # Sort itinerary by start_time

         # Output as JSON
         import json
         print(json.dumps({"itinerary": itinerary}, indent=2))
     else:
         print('{"itinerary": []}')
