    start = [Real(f"start_{i}") for i in range(n)]
    end = [Real(f"end_{i}") for i in range(n)]
    
    # Base constraints for each friend
    for i in range(n):
        f = friends[i]
        opt.add(Implies(meet[i], start[i] >= f["start_avail"] - 540))  # Convert to minutes from 9:00

    # Constraints for travel time between meetings

            opt.add(Implies(And(meet[i], meet[j]),
                Or(end[i] + travel_ij <= start[j], end[j] + travel_ji <= start[i])))
    
    # Maximize number of meetings
    opt.maximize(Sum([If(meet[i], 1, 0) for i in range(n)]))
    
    if opt.check() == sat:

        for i in range(n):

                    "person": friends[i]["name"],
                    "start_time": start_str,
                    "end_time": end_str
                })
        itinerary.sort(key=lambda x: x["start_time"])
        print(json.dumps({"itinerary": itinerary}, indent=2))
    else:
        print('{"itinerary": []}')

if __name__ == "__main__":
    main()
