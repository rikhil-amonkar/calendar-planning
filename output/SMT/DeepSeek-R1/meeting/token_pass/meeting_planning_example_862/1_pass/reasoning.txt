    import z3

    # Precompute travel time matrix from the given data
    travel_time = {
        'Mission District': {
            'Alamo Square': 11,
            'Presidio': 25,
            ...},
        ...
    }

    # List of friends
    friends = [
        {'name': 'Laura', 'location': 'Alamo Square', 'avail_start': '14:30', 'avail_end': '16:15', 'min_duration': 75},
        ...]
    ]

    # Convert time strings to minutes from 9:00
    def time_str_to_minutes(time_str):
        h, m = time_str.split(':')
        return int(h)*60 + int(m) - 540

    # Precompute adjusted availability start and end for each friend
    for friend in friends:
        start_min = time_str_to_minutes(friend['avail_start'])
        end_min = time_str_to_minutes(friend['avail_end'])
        # Adjust availability start to be at least 0
        friend['avail_start_min'] = max(start_min, 0)
        friend['avail_end_min'] = end_min

    # Create a list of meetings including the dummy
    meetings = []
    # Dummy meeting
    meetings.append({
        'name': 'Start',
        'location': 'Mission District',
        'avail_start_min': 0,
        'avail_end_min': 0,
        'min_duration': 0,
        'is_dummy': True
    })
    for friend in friends:
        meetings.append({
            'name': friend['name'],
            'location': friend['location'],
            'avail_start_min': friend['avail_start_min'],
            'avail_end_min': friend['avail_end_min'],
            'min_duration': friend['min_duration'],
            'is_dummy': False
        })

    # Z3 variables
    n = len(meetings)
    starts = [z3.Int(f'start_{i}') for i in range(n)]
    ends = [z3.Int(f'end_{i}') for i in range(n)]
    happens = [z3.Bool(f'happens_{i}') for i in range(n)]

    # The dummy meeting always happens and is fixed
    solver = z3.Optimize()
    solver.add(starts[0] == 0)
    solver.add(ends[0] == 0)
    solver.add(happens[0] == True)

    # For each real meeting, if it happens, then constraints on availability and duration
    for i in range(1, n):
        m = meetings[i]
        solver.add(z3.Implies(happens[i], starts[i] >= m['avail_start_min']))
        solver.add(z3.Implies(happens[i], ends[i] <= m['avail_end_min']))

    for i in range(n):

            constraint = z3.Or(

                ends[j] + travel_ji <= starts[i]
            )
            solver.add(z3.Implies(z3.And(happens[i], happens[j]), constraint))

    # Maximize the number of real meetings that happen

    solver.maximize(z3.Sum([z3.If(h, 1, 0) for h in real_happens]))

    # Check and get model
    if solver.check() == z3.sat:

        # Extract the meetings that happen (excluding dummy) and their start and end times
        itinerary = []
        for i in range(1, n):
            if model.eval(happens[i]):

                # Convert minutes from 9:00 to time string

                start_h = start_total_minutes // 60
                start_m = start_total_minutes % 60
                end_h = end_total_minutes // 60

                time_str_start = f"{start_h}:{start_m:02d}"
                time_str_end = f"{end_h}:{end_m:02d}"
                itinerary.append({
                    'action': 'meet',
                    'location': meetings[i]['location'],
                    'person': meetings[i]['name'],
                    'start_time': time_str_start,
                    'end_time': time_str_end
                })
        # Sort itinerary by start time
        itinerary.sort(key=lambda x: x['start_time'])
        # Output as JSON
        import json
        print(json.dumps({'itinerary': itinerary}, indent=2))
    else:
        print('No solution found')
