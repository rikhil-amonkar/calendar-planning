      def check_schedule(order, travel_matrix, start_location=0):
          solver = z3.Solver()
          # The start_location is Russian Hill (index 0)
          n = len(order)
          starts = [z3.Real(f's_{i}') for i in range(n)]
          ends = [z3.Real(f'e_{i}') for i in range(n)]

          # First meeting travel from start_location to first meeting's location
          travel0 = travel_matrix[start_location][order[0]['location']]
          solver.add(starts[0] >= travel0)
          solver.add(starts[0] >= order[0]['available_start'])
          solver.add(ends[0] == starts[0] + order[0]['min_duration'])
          solver.add(ends[0] <= order[0]['available_end'])

          for i in range(1, n):
              travel_time = travel_matrix[order[i-1]['location']][order[i]['location']]
              solver.add(starts[i] >= ends[i-1] + travel_time)
              solver.add(starts[i] >= order[i]['available_start'])
              solver.add(ends[i] == starts[i] + order[i]['min_duration'])
              solver.add(ends[i] <= order[i]['available_end'])

          if solver.check() == z3.sat:
              model = solver.model()
              return [ (model.eval(starts[i]), model.eval(ends[i])) for i in range(n) ]
          else:
              return None
