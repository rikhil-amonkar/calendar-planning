      # Create the solver

      # Create variables for each meeting: a boolean for whether we meet, and a real for start time.
      met = [Bool(f'met_{i}') for i in range(4)]
      start = [Real(f'start_{i}') for i in range(4)]

      # Define the meetings data
      meetings = [
          {'name': 'Nancy', 'loc': 'Chinatown', 'start_avail': 30, 'end_avail': 270, 'dur': 90},
          {'name': 'Mary', 'loc': 'Alamo Square', 'start_avail': -120, 'end_avail': 720, 'dur': 75},
          {'name': 'Jessica', 'loc': 'Bayview', 'start_avail': 135, 'end_avail': 285, 'dur': 45},
          {'name': 'Rebecca', 'loc': 'Fisherman\'s Wharf', 'start_avail': -120, 'end_avail': -30, 'dur': 45}
      ]

      # For each meeting, if met, then constraints on start time and availability.
      for i in range(4):
          m = meetings[i]
          s.add(Implies(met[i], And(start[i] >= m['start_avail'], start[i] + m['dur'] <= m['end_avail'])))
          # Also, the start time must be at least the travel time from Financial District to this location.
          s.add(Implies(met[i], start[i] >= travel_time('Financial District', m['loc'])))

      # For each pair of meetings (i, j) with i != j, if both are met, then they must not overlap and have travel time.
      for i in range(4):
          for j in range(4):
              if i != j:
                  # The constraint: met[i] and met[j] implies ( (start[i] + meetings[i]['dur'] + travel_time(meetings[i]['loc'], meetings[j]['loc']) <= start[j]) OR (start[j] + meetings[j]['dur'] + travel_time(meetings[j]['loc'], meetings[i]['loc']) <= start[i]) )

      # We want to maximize the number of meetings that are met.
      s.maximize(Sum([If(met[i], 1, 0) for i in range(4)]))
