     from z3 import *

     # Precompute travel time matrix
     travel_dict = {
         'Financial District': {'Russian Hill': 10, 'Sunset District': 31, 'North Beach': 7, 'The Castro': 23, 'Golden Gate Park': 23},
         'Russian Hill': {'Financial District': 11, 'Sunset District': 23, 'North Beach': 5, 'The Castro': 21, 'Golden Gate Park': 21},
         'Sunset District': {'Financial District': 30, 'Russian Hill': 24, 'North Beach': 29, 'The Castro': 17, 'Golden Gate Park': 11},
         'North Beach': {'Financial District': 8, 'Russian Hill': 4, 'Sunset District': 27, 'The Castro': 22, 'Golden Gate Park': 22},
         'The Castro': {'Financial District': 20, 'Russian Hill': 18, 'Sunset District': 17, 'North Beach': 20, 'Golden Gate Park': 11},
         'Golden Gate Park': {'Financial District': 26, 'Russian Hill': 19, 'Sunset District': 10, 'North Beach': 24, 'The Castro': 13}
     }

     friends_info = [
         {'name': 'Ronald', 'location': 'Russian Hill', 'start_avail': 285, 'end_avail': 495, 'min_duration': 105},
         {'name': 'Patricia', 'location': 'Sunset District', 'start_avail': 15, 'end_avail': 780, 'min_duration': 60},
         {'name': 'Laura', 'location': 'North Beach', 'start_avail': 210, 'end_avail': 225, 'min_duration': 15},
         {'name': 'Emily', 'location': 'The Castro', 'start_avail': 435, 'end_avail': 570, 'min_duration': 60},
         {'name': 'Mary', 'location': 'Golden Gate Park', 'start_avail': 360, 'end_avail': 450, 'min_duration': 60}
     ]

     # Create the start meeting
     start_meeting = {'name': 'start', 'location': 'Financial District', 'start_avail': 0, 'end_avail': 0, 'min_duration': 0}

     all_meetings = [start_meeting] + friends_info

     # Create Z3 variables for each meeting (except start, which is fixed)
     met = [Bool(f'met_{i}') for i in range(1, len(all_meetings))]
     start_times = [Int(f'start_{i}') for i in range(len(all_meetings))]
     # For the start meeting, we fix met[0] to True and start_times[0] to 0.
     # For the friends, we have met[i] and start_times[i] for i from 1 to 5.

     s = Solver()

     # Fix the start meeting
     s.add(start_times[0] == 0)
     # We don't have a met variable for the start, but we consider it always met.

     # For each friend meeting (index i from 1 to 5)
     for i in range(1, len(all_meetings)):
         meeting = all_meetings[i]
         # If we meet this friend, then the start time must be within the availability and the meeting must last at least the minimum duration.
         s.add(Implies(met[i-1], start_times[i] >= meeting['start_avail']))
         s.add(Implies(met[i-1], start_times[i] + meeting['min_duration'] <= meeting['end_avail']))

     # Now, for every pair of meetings (i, j) with i != j, if both are met, then we require the disjunctive constraint.
     # Note: the start meeting is always met, so we don't condition on it.
     for i in range(len(all_meetings)):
         for j in range(i+1, len(all_meetings)):
             # For the start meeting (index0) we don't have a met variable, so we condition on the friends.
             # For i and j, if both are meetings that are happening (for friends, we use met[i-1] and met[j-1] for i>0 and j>0, but the start is always met).
             # Let's define conditions for whether meeting i and j are met:
             met_i = True if i == 0 else met[i-1]
             met_j = True if j == 0 else met[j-1]

             # If both are met, then we require:
             loc_i = all_meetings[i]['location']
             loc_j = all_meetings[j]['location']
             travel_ij = travel_dict[loc_i][loc_j]
             travel_ji = travel_dict[loc_j][loc_i]

             # The constraint: 
             #   start_i + dur_i + travel_ij <= start_j   OR
             #   start_j + dur_j + travel_ji <= start_i
             dur_i = all_meetings[i]['min_duration']
             dur_j = all_meetings[j]['min_duration']

             s.add(Implies(And(met_i, met_j),
                         Or(start_times[i] + dur_i + travel_ij <= start_times[j],
                            start_times[j] + dur_j + travel_ji <= start_times[i])))

     # We want to maximize the number of meetings (excluding the start)
     objective = Sum([If(m, 1, 0) for m in met])

     # We'll use Optimize to maximize the objective.
     opt = Optimize()
     opt.add(s.assertions())
     opt.maximize(objective)

     if opt.check() == sat:
         model = opt.model()
         # Extract the meetings that are met and their start times.
         scheduled_meetings = []
         for i in range(1, len(all_meetings)):
             if model.evaluate(met[i-1]):
                 start_val = model.evaluate(start_times[i]).as_long()
                 meeting = all_meetings[i]
                 scheduled_meetings.append({
                     'name': meeting['name'],
                     'location': meeting['location'],
                     'start': start_val,
                     'end': start_val + meeting['min_duration']
                 })

         # Sort scheduled_meetings by start time
         scheduled_meetings.sort(key=lambda x: x['start'])

         # Convert start and end times to time strings
         def minutes_to_time(minutes):
             # minutes is from 9:00 AM
             total_minutes = minutes
             hours = total_minutes // 60
             minutes = total_minutes % 60
             # The time is (9 + hours) but note: we started at 9:00, so the time of day is 9:00 + minutes
             hour_of_day = 9 + hours
             # But note: if hour_of_day becomes more than 12, we use 24-hour format.
             return f"{hour_of_day}:{minutes:02d}"

         itinerary = []
         for meeting in scheduled_meetings:
             itinerary.append({
                 "action": "meet",
                 "location": meeting['location'],
                 "person": meeting['name'],
                 "start_time": minutes_to_time(meeting['start']),
                 "end_time": minutes_to_time(meeting['end'])
             })

         # Output as JSON
         import json
         result = {"itinerary": itinerary}
         print(json.dumps(result, indent=2))

     else:
         print('No solution found')
