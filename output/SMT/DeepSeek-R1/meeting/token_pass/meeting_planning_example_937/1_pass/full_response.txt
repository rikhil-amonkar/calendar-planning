    start_times = [Int(f"start_{i}") for i in range(n)]

    # Constraints for each meeting
    for i in range(n):
        # If meeting is scheduled, it must be within availability window
        opt.add(Implies(meet_flags[i], start_times[i] >= meetings[i]['avail_start']))
        opt.add(Implies(meet_flags[i], start_times[i] + meetings[i]['min_dur'] <= meetings[i]['avail_end']))
        # Travel time from Russian Hill to meeting location
        opt.add(Implies(meet_flags[i], start_times[i] >= travel_times['Russian Hill'][meetings[i]['location']]))

    # Pairwise constraints for travel between meetings
    for i in range(n):
        for j in range(i+1, n):
            travel_ij = travel_times[meetings[i]['location']][meetings[j]['location']]
            travel_ji = travel_times[meetings[j]['location']][meetings[i]['location']]
            opt.add(Implies(And(meet_flags[i], meet_flags[j]),
                          Or(
                              start_times[i] + meetings[i]['min_dur'] + travel_ij <= start_times[j],
                              start_times[j] + meetings[j]['min_dur'] + travel_ji <= start_times[i]
                          )))

    # Maximize the number of meetings
    opt.maximize(Sum([If(flag, 1, 0) for flag in meet_flags]))
    
    if opt.check() == sat:
        model = opt.model()
        scheduled_meetings = []
        for i in range(n):
            if is_true(model.evaluate(meet_flags[i])):
                start_val = model.evaluate(start_times[i]).as_long()
                end_val = start_val + meetings[i]['min_dur']
                scheduled_meetings.append({
                    'action': 'meet',
                    'location': meetings[i]['location'],
                    'person': meetings[i]['person'],
                    'start_time': minutes_to_time(start_val),
                    'end_time': minutes_to_time(end_val)
                })
        scheduled_meetings.sort(key=lambda x: x['start_time'])
        result = {'itinerary': scheduled_meetings}
        print(json.dumps(result, indent=2))
    else:
        print('{"itinerary": []}')
