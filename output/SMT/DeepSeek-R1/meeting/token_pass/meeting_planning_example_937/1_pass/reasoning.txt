    from z3 import *

    # Define travel_dict and friends list as above.

    # Create a solver with optimization
    opt = Optimize()

    # Create a list for meetings (Z3 variables and also the meeting info)
    n = len(friends)
    meetings = [None] * (n+1)   # index 0 is the start, then 1..n for friends

    # Initialize meeting0 (start)
    meetings[0] = {
        'name': 'Start',
        'location': 'Russian Hill',
        'start': 0,
        'end': 0,
        'meet': True,   # always
        'z3_start': 0,
        'z3_end': 0
    }

    # For each friend, create Z3 variables and constraints
    for i, friend in enumerate(friends, start=1):
        meet_var = Bool(f"meet_{i}")
        start_var = Int(f"start_{i}")
        end_var = Int(f"end_{i}")

        # Constraints if we meet this friend
        opt.add(Implies(meet_var, start_var >= friend['start']))
        opt.add(Implies(meet_var, end_var <= friend['end']))
        opt.add(Implies(meet_var, end_var - start_var >= friend['min_duration']))

        meetings[i] = {
            'name': friend['name'],
            'location': friend['location'],
            'meet': meet_var,
            'z3_start': start_var,
            'z3_end': end_var
        }

    # Now, for every pair of meetings (i, j) with i != j, we add the disjunctive constraint if both are held.
    for i in range(n+1):
        for j in range(i+1, n+1):
            # If both meetings are held, then we require either i before j or j before i.
            loc_i = meetings[i]['location']
            loc_j = meetings[j]['location']
            travel_ij = travel_dict[loc_i][loc_j]
            travel_ji = travel_dict[loc_j][loc_i]

            # For meeting i and j, if both are held, then:
            condition = Or(
                meetings[i]['z3_end'] + travel_ij <= meetings[j]['z3_start'],
                meetings[j]['z3_end'] + travel_ji <= meetings[i]['z3_start']
            )

            # Only add if both are held
            if i == 0:
                # meeting0 is always held
                opt.add(Implies(meetings[j]['meet'], condition))
            elif j == 0:
                opt.add(Implies(meetings[i]['meet'], condition))
            else:
                opt.add(Implies(And(meetings[i]['meet'], meetings[j]['meet']), condition))

    # Maximize the number of meetings held (for i from 1 to n)
    opt.maximize(Sum([If(meetings[i]['meet'], 1, 0) for i in range(1, n+1)]))

    # Check and get the model
    if opt.check() == sat:
        model = opt.model()
        # Then we extract the meetings that are held and their times.
        # We need to sort the meetings by start time for the itinerary.
        held_meetings = []
        for i in range(1, n+1):
            if is_true(model[meetings[i]['meet']]):
                start_val = model[meetings[i]['z3_start']].as_long()
                end_val = model[meetings[i]['z3_end']].as_long()
                held_meetings.append({
                    'name': meetings[i]['name'],
                    'location': meetings[i]['location'],
                    'start': start_val,
                    'end': end_val
                })

        # Sort by start time
        held_meetings.sort(key=lambda x: x['start'])

        # Convert start and end times (minutes from 9:00AM) back to time strings.
        def minutes_to_time(minutes):
            total_minutes = minutes + 9*60   # convert to minutes from midnight
            hour = total_minutes // 60
            minute = total_minutes % 60
            # Format as "H:MM" without leading zero for hour, but minute always two digits?
            return f"{hour}:{minute:02d}"

        itinerary = []
        for meeting in held_meetings:
            itinerary.append({
                "action": "meet",
                "location": meeting['location'],
                "person": meeting['name'],
                "start_time": minutes_to_time(meeting['start']),
                "end_time": minutes_to_time(meeting['end'])
            })

        # Output as JSON
        import json
        result = {"itinerary": itinerary}
        print(json.dumps(result, indent=2))

    else:
        print('No solution found')
