    ends = [Int('end_%d' % i) for i in range(4)]

    # We'll try to schedule n meetings from 4 down to 1.
    found = False
    solution = None
    for n in range(4,0,-1):
        # Generate all subsets of size n

            # Generate all permutations of this subset
            for perm in itertools.permutations(subset):
                s = Solver()

                # For meetings not in the subset, we ignore (we don't set any constraints, but we also don't use them)
                # But note: we must not require anything for meetings not in the subset.

                for i in subset:

                first_meeting = perm[0]
                first_loc = locations[first_meeting]
                s.add(starts[first_meeting] >= travel_time[4][first_loc])

                # For consecutive meetings in the permutation
                for idx in range(len(perm)-1):
                    meeting_i = perm[idx]
                    meeting_j = perm[idx+1]
                    loc_i = locations[meeting_i]
                    loc_j = locations[meeting_j]
                    s.add(starts[meeting_j] >= ends[meeting_i] + travel_time[loc_i][loc_j])

                # Also, we must ensure that the meetings do not overlap arbitrarily (the travel constraints and order should take care of that, but we also have to ensure that the meetings are ordered as in the permutation? Actually, the travel constraints force the order.)

                # Check if satisfiable
                if s.check() == sat:

                    # We found a solution for n meetings
                    found = True
                    solution = (subset, perm, m)
                    break
            if found:

                "action": "meet",
                "location": location_names[i],
                "person": meeting_names[i],
                "start_time": start_time_str,
                "end_time": end_time_str

        # Output as JSON
