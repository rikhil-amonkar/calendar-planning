        s = Solver()

        house_name = {}
        names = ['Arnold', 'Peter', 'Eric']
        for n in names:
            house_name[n] = Int('house_'+n)

        house_height = {}
        heights = ['short', 'average', 'very short']
        for h in heights:
            house_height[h] = Int('house_'+h.replace(' ', '_'))

        # Also, we create variables for the name and height in each house
        name = [Int('name_%d' % i) for i in range(1,4)]
        height = [Int('height_%d' % i) for i in range(1,4)]

        # Each house's name and height must be in [0,2] (since we have 3 attributes for each)
        for i in range(3):
            s.add(name[i] >= 0, name[i] <= 2)
            s.add(height[i] >= 0, height[i] <= 2)

        # The house_name and house_height variables must be between 1 and 3
        for n in names:
            s.add(house_name[n] >= 1, house_name[n] <= 3)
        for h in heights:
            s.add(house_height[h] >= 1, house_height[h] <= 3)

        # Distinct constraints for the house of each name and each height
        s.add(Distinct([house_name[n] for n in names]))
        s.add(Distinct([house_height[h] for h in heights]))

        # Link the house_name and house_height to the per-house variables
        for i in range(3):  # i from 0 to 2, which corresponds to house i+1
            for idx, n in enumerate(names):
                s.add( (house_name[n] == i+1) == (name[i] == idx) )
            for idx, h in enumerate(heights):
                s.add( (house_height[h] == i+1) == (height[i] == idx) )

        s.add(house_height['short'] == 1)
