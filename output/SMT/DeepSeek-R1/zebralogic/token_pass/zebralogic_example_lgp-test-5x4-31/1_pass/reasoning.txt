   # Create sorts for each category

   VacationSort, (cruise, city, camping, beach, mountain) = EnumSort('Vacation', ['cruise', 'city', 'camping', 'beach', 'mountain'])
   ChildSort, (Bella, Samantha, Fred, Meredith, Timothy) = EnumSort('Child', ['Bella', 'Samantha', 'Fred', 'Meredith', 'Timothy'])
   NationalitySort, (dane, norwegian, brit, german, swede) = EnumSort('Nationality', ['dane', 'norwegian', 'brit', 'german', 'swede'])
