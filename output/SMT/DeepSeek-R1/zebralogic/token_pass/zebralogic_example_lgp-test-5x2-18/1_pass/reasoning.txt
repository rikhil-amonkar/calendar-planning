      from z3 import *

      s = Solver()

      names = ['Eric', 'Alice', 'Peter', 'Bob', 'Arnold']
      children = ['Timothy', 'Meredith', 'Samantha', 'Fred', 'Bella']

      # Create position variables for each name and each child.
      n_pos = {}
      for n in names:
          n_pos[n] = Int('n_'+n)
          s.add(And(n_pos[n] >= 1, n_pos[n] <= 5))

      s.add(Distinct([n_pos[n] for n in names]))

      c_pos = {}
      for c in children:
          c_pos[c] = Int('c_'+c)
          s.add(And(c_pos[c] >= 1, c_pos[c] <= 5))

      s.add(Distinct([c_pos[c] for c in children]))

      # Now add the clues:

      # 1. Bob is somewhere to the left of the person's child is named Samantha.
      s.add(n_pos['Bob'] < c_pos['Samantha'])

      # 2. The person who is the mother of Timothy is somewhere to the left of the person's child is named Samantha.
      # Note: the mother of Timothy is the person in the house where the child is Timothy.
      s.add(c_pos['Timothy'] < c_pos['Samantha'])

      # 3. The person's child is named Fred is in the second house.
      s.add(c_pos['Fred'] == 2)

      # 4. There is one house between Alice and the person's child is named Samantha.
      # This means: |n_pos['Alice'] - c_pos['Samantha']| == 2
      s.add(Or(n_pos['Alice'] - c_pos['Samantha'] == 2, c_pos['Samantha'] - n_pos['Alice'] == 2))

      # 5. Eric is not in the third house.
      s.add(n_pos['Eric'] != 3)

      # 6. Bob is not in the third house.
      s.add(n_pos['Bob'] != 3)

      # 7. The person's child is named Fred is directly left of the person's child is named Bella.
