      from z3 import *

      # Define the names and occupations
      names = ['Arnold', 'Eric', 'Peter', 'Alice']
      occupations = ['doctor', 'engineer', 'artist', 'teacher']

      # Create integer variables for the house of each name
      name_house = {}
      for n in names:
          name_house[n] = Int('name_house_' + n)

      # Similarly for occupations
      occ_house = {}
      for o in occupations:
          occ_house[o] = Int('occ_house_' + o)

      s = Solver()

      # Each name_house and occ_house must be between 1 and 4
      for n in names:
          s.add(And(name_house[n] >= 1, name_house[n] <= 4))
      for o in occupations:
          s.add(And(occ_house[o] >= 1, occ_house[o] <= 4))

      # The names are in distinct houses
      s.add(Distinct([name_house[n] for n in names]))
      s.add(Distinct([occ_house[o] for o in occupations]))

      # Now the clues:

      # 1. There are two houses between Eric and Peter.
      s.add(Abs(name_house['Eric'] - name_house['Peter']) == 3)

      # 2. The person who is a teacher is Peter.
      s.add(occ_house['teacher'] == name_house['Peter'])

      # 3. Peter is not in the first house.
      s.add(name_house['Peter'] != 1)

      # 4. There is one house between the person who is a doctor and Alice.
      s.add(Abs(occ_house['doctor'] - name_house['Alice']) == 2)

      # 5. The person who is an artist is Alice.
      s.add(occ_house['artist'] == name_house['Alice'])
