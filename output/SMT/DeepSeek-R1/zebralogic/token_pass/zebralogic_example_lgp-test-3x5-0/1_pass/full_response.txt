```python
import json
from z3 import *

def main():
    # Define the attributes
    names = ['Peter', 'Arnold', 'Eric']
    genres = ['science fiction', 'mystery', 'romance']
    smoothies = ['watermelon', 'desert', 'cherry']
    birthdays = ['april', 'jan', 'sept']
    heights = ['average', 'very short', 'short']
    
    # Create solver
    s = Solver()
    
    # Create Z3 enums for each attribute
    Name = Datatype('Name')
    for n in names:
        Name.declare(n)
    Name = Name.create()
    
    Genre = Datatype('Genre')
    for g in genres:
        Genre.declare(g)
    Genre = Genre.create()
    
    Smoothie = Datatype('Smoothie')
    for sm in smoothies:
        Smoothie.declare(sm)
    Smoothie = Smoothie.create()
    
    Birthday = Datatype('Birthday')
    for b in birthdays:
        Birthday.declare(b)
    Birthday = Birthday.create()
    
    Height = Datatype('Height')
    for h in heights:
        Height.declare(h)
    Height = Height.create()
    
    # Create variables for each house (1-3) for each attribute
    house_names = [Const(f'name_{i}', Name) for i in range(1,4)]
    house_genres = [Const(f'genre_{i}', Genre) for i in range(1,4)]
    house_smoothies = [Const(f'smoothie_{i}', Smoothie) for i in range(1,4)]
    house_birthdays = [Const(f'birthday_{i}', Birthday) for i in range(1,4)]
    house_heights = [Const(f'height_{i}', Height) for i in range(1,4)]
    
    # Add constraint: all attributes must be unique per category
    s.add(Distinct(house_names))
    s.add(Distinct(house_genres))
    s.add(Distinct(house_smoothies))
    s.add(Distinct(house_birthdays))
    s.add(Distinct(house_heights))
    
    # Clue 1: The person who likes Cherry smoothies is not in the second house.
    s.add(house_smoothies[1] != Smoothie.cherry)
    
    # Clue 2: Arnold is the person who loves mystery books.
    # We need to find the house where name is Arnold and set its genre to mystery
    for i in range(3):
        s.add(Implies(house_names[i] == Name.Arnold, house_genres[i] == Genre.mystery))
    
    # Clue 3: The person whose birthday is in January is not in the first house.
    s.add(house_birthdays[0] != Birthday.jan)
    
    # Clue 4: The person who is very short is the person who loves romance books.
    for i in range(3):
        s.add(Implies(house_heights[i] == Height.very_short, house_genres[i] == Genre.romance))
    
    # Clue 5: The person who loves mystery books is the person whose birthday is in September.
    for i in range(3):
        s.add(Implies(house_genres[i] == Genre.mystery, house_birthdays[i] == Birthday.sept))
    
    # Clue 6: The person who has an average height is the Desert smoothie lover.
    for i in range(3):
        s.add(Implies(house_heights[i] == Height.average, house_smoothies[i] == Smoothie.desert))
    
    # Clue 7: Eric is in the first house.
    s.add(house_names[0] == Name.Eric)
    
    # Clue 8: The Watermelon smoothie lover is the person who is short.
    for i in range(3):
        s.add(Implies(house_smoothies[i] == Smoothie.watermelon, house_heights[i] == Height.short))
    
    # Clue 9: The Watermelon smoothie lover is Eric.
    for i in range(3):
        s.add(Implies(house_smoothies[i] == Smoothie.watermelon, house_names[i] == Name.Eric))
    
    # Check and get the model
    if s.check() == sat:
        m = s.model()
        
        # Prepare results
        rows = []
        for i in range(3):
            name_val = m.eval(house_names[i])
            genre_val = m.eval(house_genres[i])
            smoothie_val = m.eval(house_smoothies[i])
            birthday_val = m.eval(house_birthdays[i])
            height_val = m.eval(house_heights[i])
            
            # Convert Z3 values to strings
            name_str = str(name_val).split('!')[0]
            genre_str = str(genre_val).split('!')[0]
            smoothie_str = str(smoothie_val).split('!')[0]
            birthday_str = str(birthday_val).split('!')[0]
            height_str = str(height_val).split('!')[0]
            
            rows.append([str(i+1), name_str, genre_str, smoothie_str, birthday_str, height_str])
        
        # Create JSON output
        output = {
            "solution": {
                "header": ["House", "Name", "BookGenre", "Smoothie", "Birthday", "Height"],
                "rows": rows
            }
        }
        print(json.dumps(output, indent=2))
    else:
        print("No solution found")

if __name__ == "__main__":
    main()
```
