   from z3 import *

   # Create enums for each category
   Name = Datatype('Name')
   for n in ['Eric', 'Peter', 'Alice', 'Arnold']:
       Name.declare(n)
   Name = Name.create()

   CarModel = Datatype('CarModel')
   for c in ['tesla_model_3', 'honda_civic', 'toyota_camry', 'ford_f150']:
       CarModel.declare(c)
   CarModel = CarModel.create()

   Birthday = Datatype('Birthday')
   for b in ['jan', 'april', 'sept', 'feb']:
       Birthday.declare(b)
   Birthday = Birthday.create()

   Hobby = Datatype('Hobby')
   for h in ['painting', 'cooking', 'gardening', 'photography']:
       Hobby.declare(h)
   Hobby = Hobby.create()

   # Now, for each house, we create a variable for each attribute.
   houses = [1, 2, 3, 4]
   name = [Const('name_%d' % i, Name) for i in houses]
   car = [Const('car_%d' % i, CarModel) for i in houses]
   birthday = [Const('birthday_%d' % i, Birthday) for i in houses]
   hobby = [Const('hobby_%d' % i, Hobby) for i in houses]

   s = Solver()

   # Each attribute must be unique per house for that category.
   s.add(Distinct(name))
   s.add(Distinct(car))
   s.add(Distinct(birthday))
   s.add(Distinct(hobby))

   # Now add the constraints from the clues.

   # Clue 1: The person whose birthday is in January is not in the second house.
   s.add(birthday[1] != Birthday.jan)   # house index 1 is house 1, but note: our list index 0 is house 1.

   s.add(birthday[1] != Birthday.jan)
