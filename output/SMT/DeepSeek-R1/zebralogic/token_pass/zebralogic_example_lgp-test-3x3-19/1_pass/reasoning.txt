      from z3 import *

      s = Solver()

      # Define the attributes as integers
      names = {'Eric':0, 'Arnold':1, 'Peter':2}
      smoothies = {'desert':0, 'watermelon':1, 'cherry':2}
      books = {'science fiction':0, 'romance':1, 'mystery':2}

      # Create variables for each house for each category
      name = [Int('name_%d' % (i+1)) for i in range(3)]
      smoothie = [Int('smoothie_%d' % (i+1)) for i in range(3)]
      book = [Int('book_%d' % (i+1)) for i in range(3)]

      # Each category must be distinct and in the range [0,2]
      for i in range(3):
          s.add(And(name[i] >= 0, name[i] <= 2))
          s.add(And(smoothie[i] >= 0, smoothie[i] <= 2))
          s.add(And(book[i] >= 0, book[i] <= 2))

      s.add(Distinct(name))
      s.add(Distinct(smoothie))
      s.add(Distinct(book))

      # Clue 5: Peter is in the first house.
      s.add(name[0] == names['Peter'])

      # Clue 2: Arnold is the person who loves mystery books.
      # So for the house i where name[i] == names['Arnold'], we must have book[i] == books['mystery']
      for i in range(3):
          s.add(If(name[i] == names['Arnold'], book[i] == books['mystery'], True))

      # Clue 3: The person who loves science fiction books is not in the first house.
      s.add(book[0] != books['science fiction'])

      # Clue 1: Cherry smoothie left of mystery book
      # We need to find the house for cherry smoothie and mystery book.
      cherry_house = Int('cherry_house')
      # We'll define cherry_house as the house index (1-based) where smoothie[i] == smoothies['cherry']
      s.add(cherry_house == If(smoothie[0] == smoothies['cherry'], 1, 
                            If(smoothie[1] == smoothies['cherry'], 2, 3)))

      mystery_house = Int('mystery_house')
      s.add(mystery_house == If(book[0] == books['mystery'], 1,
                             If(book[1] == books['mystery'], 2, 3)))

      s.add(cherry_house < mystery_house)

      # Clue 4: Desert smoothie directly left of mystery book
      desert_house = Int('desert_house')

                            If(smoothie[1] == smoothies['desert'], 2, 3)))

      s.add(desert_house + 1 == mystery_house)
