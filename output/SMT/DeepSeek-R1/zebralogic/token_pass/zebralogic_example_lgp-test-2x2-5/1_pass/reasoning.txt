   from z3 import *

   # Define the enums
   Name, (Eric, Arnold) = EnumSort('Name', ['Eric', 'Arnold'])
   Style, (victorian, colonial) = EnumSort('Style', ['victorian', 'colonial'])

   # Create variables for each house
   name1 = Const('name1', Name)
   name2 = Const('name2', Name)
   style1 = Const('style1', Style)
   style2 = Const('style2', Style)

   s = Solver()

   # Each house has a unique name and style, but note: the attributes are unique across houses.
   s.add(Distinct(name1, name2))
   s.add(Distinct(style1, style2))

   # Clue 1: Victorian is left of colonial.
   # We can define: the house with Victorian style has a lower house number than the one with colonial.
   # But note: we have only two houses, so we can say:
   #   (style1 == victorian) And (style2 == colonial)
   # Because if Victorian is in house1 and colonial in house2, then house1 is left of house2.

   s.add(And(style1 == victorian, style2 == colonial))

   # Clue 2: Eric is in the first house.
   s.add(name1 == Eric)

   # Check and get the model.

   if s.check() == sat:
