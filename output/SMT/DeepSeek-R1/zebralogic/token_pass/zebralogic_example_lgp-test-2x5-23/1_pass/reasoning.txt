   from z3 import *

   # Define the enums
   Name = Enum('Name', ['Arnold', 'Eric'])
   Education = Enum('Education', ['associate', 'high school'])
   Height = Enum('Height', ['short', 'very short'])
   Food = Enum('Food', ['grilled cheese', 'pizza'])
   Drink = Enum('Drink', ['tea', 'water'])

   # Create variables for each house for each attribute
   name1, name2 = Consts('name1 name2', Name)
   education1, education2 = Consts('education1 education2', Education)
   height1, height2 = Consts('height1 height2', Height)
   food1, food2 = Consts('food1 food2', Food)
   drink1, drink2 = Consts('drink1 drink2', Drink)

   s = Solver()

   # Each attribute category must have distinct values per house
   s.add(Distinct(name1, name2))
   s.add(Distinct(education1, education2))
   s.add(Distinct(height1, height2))
   s.add(Distinct(food1, food2))
   s.add(Distinct(drink1, drink2))

   # Now add the clues:

   # Clue 1: The person who is very short is the person who is a pizza lover.
   # We can express: (height1 is very short and food1 is pizza) or (height2 is very short and food2 is pizza)
   very_short = Height.variant(1)  # because 'very short' is the second element we passed
   pizza = Food.variant(1)          # similarly, 'pizza' is the second in Food
   s.add(Or(And(height1 == very_short, food1 == pizza),
            And(height2 == very_short, food2 == pizza)))

   # Clue 2: The person who loves eating grilled cheese is in the second house.
   grilled_cheese = Food.variant(0)
   s.add(food2 == grilled_cheese)

   # Clue 3: The person with a high school diploma is the person who is a pizza lover.
   high_school = Education.variant(1)
   s.add(Or(And(education1 == high_school, food1 == pizza),
            And(education2 == high_school, food2 == pizza)))

   # Clue 4: The tea drinker is the person who loves eating grilled cheese.
   tea = Drink.variant(0)
   s.add(Or(And(drink1 == tea, food1 == grilled_cheese),
            And(drink2 == tea, food2 == grilled_cheese)))

   # Clue 5: Arnold is the person who is a pizza lover.
   arnold = Name.variant(0)
   s.add(Or(And(name1 == arnold, food1 == pizza),
            And(name2 == arnold, food2 == pizza)))
