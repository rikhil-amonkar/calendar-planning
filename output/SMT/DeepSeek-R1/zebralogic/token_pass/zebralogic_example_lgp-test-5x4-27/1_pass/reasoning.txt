   from z3 import *
   import json

   s = Solver()

   # Define enums
   NameSort, (Peter, Alice, Eric, Bob, Arnold) = EnumSort('Name', ['Peter','Alice','Eric','Bob','Arnold'])
   BirthdaySort, (april, feb, mar, jan, sept) = EnumSort('Birthday', ['april','feb','mar','jan','sept'])
   CigarSort, (pall_mall, prince, dunhill, blends, blue_master) = EnumSort('Cigar', ['pall mall','prince','dunhill','blends','blue master'])
   DrinkSort, (water, coffee, tea, milk, root_beer) = EnumSort('Drink', ['water','coffee','tea','milk','root beer'])

   # Create variables for each house for each attribute
   names = [Const('name_%d' % i, NameSort) for i in range(1,6)]
   birthdays = [Const('birthday_%d' % i, BirthdaySort) for i in range(1,6)]
   cigars = [Const('cigar_%d' % i, CigarSort) for i in range(1,6)]
   drinks = [Const('drink_%d' % i, DrinkSort) for i in range(1,6)]

   # Distinct constraints
   s.add(Distinct(names))
   s.add(Distinct(birthdays))
   s.add(Distinct(cigars))
   s.add(Distinct(drinks))

   # Auxiliary house variables for relative position constraints
   house_root_beer = Int('house_root_beer')
   house_peter = Int('house_peter')
   house_jan = Int('house_jan')
   house_arnold = Int('house_arnold')
   house_tea = Int('house_tea')
   house_coffee = Int('house_coffee')

   # Constrain auxiliary variables to be in [1,5]
   s.add(house_root_beer >= 1, house_root_beer <= 5)
   s.add(house_peter >= 1, house_peter <= 5)
   s.add(house_jan >= 1, house_jan <= 5)
   s.add(house_arnold >= 1, house_arnold <= 5)
   s.add(house_tea >= 1, house_tea <= 5)
   s.add(house_coffee >= 1, house_coffee <= 5)

   # Link auxiliary variables to the attributes
   for i in range(5):
        # i from 0 to 4, representing house i+1
        s.add(Implies(drinks[i]==root_beer, house_root_beer == i+1))
        s.add(Implies(names[i]==Peter, house_peter == i+1))
        s.add(Implies(birthdays[i]==jan, house_jan == i+1))
        s.add(Implies(names[i]==Arnold, house_arnold == i+1))
        s.add(Implies(drinks[i]==tea, house_tea == i+1))
        s.add(Implies(drinks[i]==coffee, house_coffee == i+1))

   # Now the clues:

   # 1. The root beer lover is Eric.
   for i in range(5):
        s.add(Implies(names[i]==Eric, drinks[i]==root_beer))

   # 2. The person partial to Pall Mall is in the third house.
   s.add(cigars[2] == pall_mall)   # house3 is index2

   # 3. The person whose birthday is in April is Bob.
   for i in range(5):
        s.add(Implies(birthdays[i]==april, names[i]==Bob))

   # 4. The Dunhill smoker is the person whose birthday is in March.
   for i in range(5):
        s.add(Implies(cigars[i]==dunhill, birthdays[i]==mar))

   # 5. Peter is somewhere to the right of the root beer lover.
   s.add(house_peter > house_root_beer)

   # 6. There is one house between the person whose birthday is in January and Peter.
   s.add(Abs(house_jan - house_peter) == 2)

   # 7. The person who smokes many unique blends is the person whose birthday is in February.
   for i in range(5):
        s.add(Implies(cigars[i]==blends, birthdays[i]==feb))

   # 8. The person whose birthday is in February is in the second house.
   s.add(birthdays[1] == feb)   # house2 is index1

   # 9. Arnold is directly left of Peter.
   s.add(house_arnold + 1 == house_peter)

   # 10. The person who likes milk is not in the fifth house.
   s.add(drinks[4] != milk)   # house5 is index4

   # 11. The person who smokes Blue Master is the coffee drinker.
   for i in range(5):
        s.add(Implies(cigars[i]==blue_master, drinks[i]==coffee))

   # 12. There is one house between the tea drinker and the coffee drinker.
   s.add(Abs(house_tea - house_coffee) == 2)

   # 13. Eric is in the third house.
   s.add(names[2] == Eric)   # house3 is index2
