   PhoneSort, (oneplus9, google_pixel6, samsung_galaxy_s21, iphone13, huawei_p50, xiaomi_mi11) = EnumSort('Phone', ['oneplus9','google_pixel6','samsung_galaxy_s21','iphone13','huawei_p50','xiaomi_mi11'])
