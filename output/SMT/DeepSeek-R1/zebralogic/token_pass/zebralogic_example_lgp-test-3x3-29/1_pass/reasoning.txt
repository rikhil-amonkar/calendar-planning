   For clue 1: we require that there exists an i and j such that:
        (food_i == 3) and (name_j == 2) and (|i - j| == 1)

        house1 adjacent to house2

   from z3 import *

   # Define the attributes as integers
   names = {'Eric': 1, 'Peter': 2, 'Arnold': 3}
   mothers = {'Holly': 1, 'Aniya': 2, 'Janelle': 3}
   foods = {'pizza': 1, 'grilled cheese': 2, 'spaghetti': 3}

   # We'll create variables for each house for name, mother, food.
   n1, n2, n3 = Ints('n1 n2 n3')
   m1, m2, m3 = Ints('m1 m2 m3')
   f1, f2, f3 = Ints('f1 f2 f3')

   s = Solver()

   # Each variable is between 1 and 3
   s.add([n1>=1, n1<=3, n2>=1, n2<=3, n3>=1, n3<=3])
   s.add([m1>=1, m1<=3, m2>=1, m2<=3, m3>=1, m3<=3])
   s.add([f1>=1, f1<=3, f2>=1, f2<=3, f3>=1, f3<=3])

   # Distinct constraints
   s.add(Distinct(n1, n2, n3))
   s.add(Distinct(m1, m2, m3))
   s.add(Distinct(f1, f2, f3))

   # Clue 1: The person who loves the spaghetti eater and Peter are next to each other.
   # We break it down by which house has spaghetti.
   s.add(
       Or(
           And(f1 == foods['spaghetti'], n2 == names['Peter']),   # spaghetti in 1 -> Peter in 2
           And(f2 == foods['spaghetti'], Or(n1 == names['Peter'], n3 == names['Peter'])), # spaghetti in 2 -> Peter in 1 or 3
           And(f3 == foods['spaghetti'], n2 == names['Peter'])    # spaghetti in 3 -> Peter in 2
       )
   )

   # Clue 2: The person who loves eating grilled cheese is directly left of The person whose mother's name is Aniya.
   s.add(
       Or(
           And(f1 == foods['grilled cheese'], m2 == mothers['Aniya']),
           And(f2 == foods['grilled cheese'], m3 == mothers['Aniya'])
       )
   )

   # Clue 3: The person who loves eating grilled cheese is Eric.
   s.add(
       Or(
           And(f1 == foods['grilled cheese'], n1 == names['Eric']),
           And(f2 == foods['grilled cheese'], n2 == names['Eric']),
           And(f3 == foods['grilled cheese'], n3 == names['Eric'])
       )
   )

   # Clue 4: Peter is The person whose mother's name is Holly.
   s.add(
       Or(
           And(n1 == names['Peter'], m1 == mothers['Holly']),
           And(n2 == names['Peter'], m2 == mothers['Holly']),
           And(n3 == names['Peter'], m3 == mothers['Holly'])
       )
   )
