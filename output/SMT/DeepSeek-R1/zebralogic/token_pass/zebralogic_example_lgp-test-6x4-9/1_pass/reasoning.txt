   from z3 import *

   # Define the lists
   names = ['Carol', 'Bob', 'Alice', 'Arnold', 'Eric', 'Peter']
   phones = ['samsung_galaxy_s21', 'google_pixel_6', 'iphone_13', 'huawei_p50', 'oneplus_9', 'xiaomi_mi_11']
   nationalities = ['swede', 'chinese', 'norwegian', 'dane', 'german', 'brit']
   colors = ['blue', 'red', 'yellow', 'green', 'white', 'purple']

   # Create a solver
   s = Solver()

   # Create variables for each attribute value (house number for that value)
   # For names
   name_vars = {}
   for n in names:
       name_vars[n] = Int(f'{n}_house')
       s.add(And(name_vars[n] >= 1, name_vars[n] <= 6))

   # Similarly for phones, nationalities, colors
   phone_vars = {}
   for p in phones:
       phone_vars[p] = Int(f'{p}_house')
       s.add(And(phone_vars[p] >= 1, phone_vars[p] <= 6))

   nation_vars = {}
   for nat in nationalities:
       nation_vars[nat] = Int(f'{nat}_house')
       s.add(And(nation_vars[nat] >= 1, nation_vars[nat] <= 6))

   color_vars = {}
   for c in colors:
       color_vars[c] = Int(f'{c}_house')
       s.add(And(color_vars[c] >= 1, color_vars[c] <= 6))

   # Now, add distinct constraints for each category
   s.add(Distinct([name_vars[n] for n in names]))
   s.add(Distinct([phone_vars[p] for p in phones]))
   s.add(Distinct([nation_vars[nat] for nat in nationalities]))
   s.add(Distinct([color_vars[c] for c in colors]))

   # Now, add the clues

   # 1. Carol is not in the third house.
   s.add(name_vars['Carol'] != 3)

   # 2. There is one house between the Dane and the British person.
   s.add(Abs(nation_vars['dane'] - nation_vars['brit']) == 2)

   s.add(phone_vars['huawei_p50'] != 3)

   # 8. The person who uses a Samsung Galaxy S21 is in the fifth house.
   s.add(phone_vars['samsung_galaxy_s21'] == 5)

   # 9. The person who loves white is somewhere to the right of the person whose favorite color is red.
   s.add(color_vars['white'] > color_vars['red'])

   # 10. The person who uses a Samsung Galaxy S21 is Bob.
   s.add(phone_vars['samsung_galaxy_s21'] == name_vars['Bob'])

   # 11. The Dane is the person who loves yellow.
   s.add(nation_vars['dane'] == color_vars['yellow'])

   # 12. The person who uses a Samsung Galaxy S21 is somewhere to the left of Peter.
   s.add(phone_vars['samsung_galaxy_s21'] < name_vars['Peter'])

   # 13. The person who loves blue is Peter.
   s.add(color_vars['blue'] == name_vars['Peter'])

   # 14. Peter is the British person.
   s.add(name_vars['Peter'] == nation_vars['brit'])

   s.add(phone_vars['xiaomi_mi_11'] == nation_vars['chinese'])

            house_index = m[var].as_long() - 1
            # We want to store the original string with spaces? But note: the output requires the original phone model string.

            # Actually, we can create a mapping from the variable name to the original string, but we didn't.
