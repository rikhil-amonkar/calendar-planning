```python
import json
from z3 import *

def main():
    # Create solver
    solver = Solver()

    # Define the house numbers
    houses = [1, 2, 3, 4, 5]

    # Define attributes using EnumSort
    Name, (Arnold, Eric, Bob, Peter, Alice) = EnumSort('Name', ['Arnold', 'Eric', 'Bob', 'Peter', 'Alice'])
    Smoothie, (desert, watermelon, lime, cherry, dragonfruit) = EnumSort('Smoothie', ['desert', 'watermelon', 'lime', 'cherry', 'dragonfruit'])
    Nationality, (german, swede, norwegian, dane, brit) = EnumSort('Nationality', ['german', 'swede', 'norwegian', 'dane', 'brit'])

    # Create variables for each house
    names = [Const(f'name_{i}', Name) for i in houses]
    smoothies = [Const(f'smoothie_{i}', Smoothie) for i in houses]
    nationalities = [Const(f'nationality_{i}', Nationality) for i in houses]

    # Each attribute is a permutation (all different)
    solver.add(Distinct(names))
    solver.add(Distinct(smoothies))
    solver.add(Distinct(nationalities))

    # Add clues
    # Clue 2: The Dragonfruit smoothie lover is in the second house.
    solver.add(smoothies[1] == dragonfruit)

    # Clue 10: Alice is in the third house.
    solver.add(names[2] == Alice)

    # Clue 11: The Watermelon smoothie lover is in the third house.
    solver.add(smoothies[2] == watermelon)

    # Clue 9: Alice is the Norwegian.
    solver.add(nationalities[2] == norwegian)

    # Clue 8: Bob is the Dane.
    # We'll find Bob's house and set nationality to dane
    for i in houses:
        solver.add(Implies(names[i-1] == Bob, nationalities[i-1] == dane))

    # Clue 3: Peter is not in the first house.
    solver.add(names[0] != Peter)

    # Clue 5: The Desert smoothie lover is not in the fifth house.
    solver.add(smoothies[4] != desert)

    # Clue 4: The Dane and the British person are next to each other.
    for i in range(4):
        solver.add(Or(
            And(nationalities[i] == dane, nationalities[i+1] == brit),
            And(nationalities[i] == brit, nationalities[i+1] == dane)
        ))

    # Clue 6: The Swedish person is somewhere to the left of the Dragonfruit smoothie lover.
    # Since dragonfruit is in house 2, swede must be in house 1
    solver.add(nationalities[0] == swede)

    # Clue 7: There are two houses between the person who drinks Lime smoothies and the Dane.
    # Find the house of the Dane (Bob) and then lime must be 3 away
    for i in houses:
        for j in houses:
            solver.add(Implies(And(nationalities[i-1] == dane, smoothies[j-1] == lime), Abs(i - j) == 3))

    # Clue 1: The Dragonfruit smoothie lover is somewhere to the left of Eric.
    # Since dragonfruit is in house 2, Eric must be in house 3,4, or 5
    # We'll enforce that Eric's house > 2
    for i in houses:
        solver.add(Implies(names[i-1] == Eric, i > 2))

    # Check and get solution
    if solver.check() == sat:
        model = solver.model()
        result = []
        for i in houses:
            idx = i - 1
            name_val = model.eval(names[idx])
            smoothie_val = model.eval(smoothies[idx])
            nationality_val = model.eval(nationalities[idx])
            result.append([
                str(i),
                str(name_val).split('[')[1].split(']')[0] if '[' in str(name_val) else str(name_val),
                str(smoothie_val).split('[')[1].split(']')[0] if '[' in str(smoothie_val) else str(smoothie_val),
                str(nationality_val).split('[')[1].split(']')[0] if '[' in str(nationality_val) else str(nationality_val)
            ])
        
        output = {
            "solution": {
                "header": ["House", "Name", "Smoothie", "Nationality"],
                "rows": result
            }
        }
        print(json.dumps(output, indent=2))
    else:
        print("No solution found")

if __name__ == "__main__":
    main()
```
