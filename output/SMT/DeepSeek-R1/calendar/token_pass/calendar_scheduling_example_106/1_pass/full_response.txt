from z3 import Solver, Int, Or, And, sat

def main():
    # Initialize solver
    s = Solver()
    
    # Define meeting duration in minutes
    duration = 60
    
    # Total minutes in workday (9:00 to 17:00)
    total_minutes = 480
    start_time = Int('start_time')
    s.add(start_time >= 0)
    s.add(start_time <= total_minutes - duration)
    
    # Convert busy intervals to minutes from 9:00
    # Olivia: 12:30-13:30, 14:30-15:00, 16:30-17:00
    olivia_busy = [(210, 270), (330, 360), (450, 480)]
    # Virginia: 9:00-10:00, 11:30-16:00, 16:30-17:00
    virginia_busy = [(0, 60), (150, 420), (450, 480)]
    # Paul: 9:00-9:30, 11:00-11:30, 13:00-14:00, 14:30-16:00, 16:30-17:00
    paul_busy = [(0, 30), (120, 150), (240, 300), (330, 420), (450, 480)]
    
    # Combine all constraints (Anna has no meetings)
    all_busy = olivia_busy + virginia_busy + paul_busy
    
    # Add constraints for each busy interval
    for busy_start, busy_end in all_busy:
        s.add(Or(
            start_time + duration <= busy_start,
            start_time >= busy_end
        ))
    
    # Check for solution
    if s.check() == sat:
        m = s.model()
        start_minutes = m[start_time].as_long()
        
        # Calculate start and end times
        start_hour = 9 + start_minutes // 60
        start_minute = start_minutes % 60
        end_minutes = start_minutes + duration
        end_hour = 9 + end_minutes // 60
        end_minute = end_minutes % 60
        
        # Format output
        print(f"{start_hour:02d}:{start_minute:02d}:{end_hour:02d}:{end_minute:02d}")
        print("Monday")
    else:
        print("No solution found")

if __name__ == "__main__":
    main()
