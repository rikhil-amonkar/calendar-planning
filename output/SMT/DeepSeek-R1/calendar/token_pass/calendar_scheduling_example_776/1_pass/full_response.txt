from z3 import *

def main():
    # Define variables
    day = Int('day')
    start = Int('start')
    
    s = Solver()
    
    # Day constraints: 0=Monday, 1=Tuesday, 2=Wednesday
    s.add(day >= 0, day <= 2)
    
    # Work hours: 9:00 to 17:00 (540 to 1020 minutes)
    s.add(start >= 540)
    s.add(start + 30 <= 1020)
    
    # John's constraints: avoid Tuesday and Wednesday, and on Monday avoid after 14:30 (870 minutes)
    s.add(day != 1)  # No Tuesday
    s.add(day != 2)  # No Wednesday
    s.add(If(day == 0, start + 30 <= 870, True))  # On Monday, meeting must end by 14:30
    
    # Jennifer's busy intervals (start and end in minutes from 00:00)
    busy_intervals = {
        0: [(540, 660), (690, 780), (810, 870), (900, 1020)],  # Monday
        1: [(540, 690), (720, 1020)],  # Tuesday
        2: [(540, 690), (720, 750), (780, 840), (870, 960), (990, 1020)]  # Wednesday
    }
    
    # For the selected day, ensure no overlap with busy intervals
    for day_num, intervals in busy_intervals.items():
        day_constraints = []
        for bus_start, bus_end in intervals:
            # Meeting must not overlap: it must end before busy start or start after busy end
            day_constraints.append(Or(start + 30 <= bus_start, start >= bus_end))
        s.add(If(day == day_num, And(day_constraints), True))
    
    # Check for a solution
    if s.check() == sat:
        m = s.model()
        d = m[day].as_long()
        st = m[start].as_long()
        
        # Convert minutes to time strings
        start_hr = st // 60
        start_min = st % 60
        end_minutes = st + 30
        end_hr = end_minutes // 60
        end_min = end_minutes % 60
        
        day_str = ["Monday", "Tuesday", "Wednesday"][d]
        time_str = f"{start_hr:02d}:{start_min:02d}:{end_hr:02d}:{end_min:02d}"
        print(f"{day_str} {time_str}")
    else:
        print("No solution found")

if __name__ == "__main__":
    main()
