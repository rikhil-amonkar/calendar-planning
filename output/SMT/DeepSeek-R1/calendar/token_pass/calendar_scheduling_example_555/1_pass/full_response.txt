```python
from z3 import Int, Solver, Or, And, sat

def main():
    # Initialize the solver
    solver = Solver()
    
    # Define the start time in minutes from 9:00 (0 minutes = 9:00)
    S = Int('S')
    
    # Total working minutes in the day (9:00 to 17:00 is 8 hours = 480 minutes)
    total_minutes = 480
    meeting_duration = 30
    
    # Base constraints: meeting must be within work hours
    solver.add(S >= 0)
    solver.add(S + meeting_duration <= total_minutes)
    
    # Evelyn's constraint: does not want to meet after 13:00 (13:00 is 4 hours from 9:00 = 240 minutes)
    solver.add(S + meeting_duration <= 240)
    
    # Randy's constraints: blocked periods in minutes from 9:00
    # Blocked: 9:00-10:30 (0 to 90), 11:00-15:30 (120 to 390), 16:00-17:00 (420 to 480)
    # Available slots: 10:30-11:00 (90 to 120) and 15:30-16:00 (390 to 420)
    solver.add(Or(
        And(S >= 90, S + meeting_duration <= 120),
        And(S >= 390, S + meeting_duration <= 420)
    ))
    
    # Check for a solution
    if solver.check() == sat:
        model = solver.model()
        start_time_minutes = model.eval(S).as_long()
        
        # Convert start time to HH:MM format
        start_hour = 9 + start_time_minutes // 60
        start_minute = start_time_minutes % 60
        start_str = f"{start_hour:02d}:{start_minute:02d}"
        
        # Calculate end time
        end_time_minutes = start_time_minutes + meeting_duration
        end_hour = 9 + end_time_minutes // 60
        end_minute = end_time_minutes % 60
        end_str = f"{end_hour:02d}:{end_minute:02d}"
        
        # Output the result
        print(f"Monday {start_str}:{end_str}")
    else:
        print("No solution found")

if __name__ == "__main__":
    main()
```
