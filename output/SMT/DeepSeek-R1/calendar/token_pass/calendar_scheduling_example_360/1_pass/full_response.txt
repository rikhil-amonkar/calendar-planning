To solve this meeting scheduling problem, we use the Z3 SMT solver to find a 30-minute time slot between 9:00 and 17:00 on Monday where all participants are available. The solution involves converting time to minutes past 9:00, defining constraints to avoid busy intervals, and finding a valid start time.

```python
from z3 import *

def main():
    # Initialize solver
    s = Solver()
    
    # Meeting duration in minutes
    meeting_duration = 30
    
    # Total available time window (9:00 to 17:00) in minutes
    total_minutes = 480
    start_min = 0
    end_min = total_minutes - meeting_duration
    
    # Start time variable in minutes from 9:00
    start = Int('start')
    s.add(start >= start_min, start <= end_min)
    
    # Busy intervals for each participant in minutes from 9:00
    busy_intervals = [
        # Emily's busy intervals
        [(60, 90), (420, 450)],
        # Maria's busy intervals
        [(90, 120), (300, 330)],
        # Carl's busy intervals
        [(30, 60), (90, 210), (270, 300), (330, 390), (420, 480)],
        # David's busy intervals
        [(30, 120), (150, 180), (210, 270), (300, 360), (420, 480)],
        # Frank's busy intervals
        [(30, 90), (120, 150), (210, 270), (330, 480)]
    ]
    
    # Add constraints for each participant's busy intervals
    for intervals in busy_intervals:
        for (busy_start, busy_end) in intervals:
            # Meeting must not overlap with busy interval
            s.add(Or(start + meeting_duration <= busy_start, start >= busy_end))
    
    # Check for a solution
    if s.check() == sat:
        m = s.model()
        start_val = m.evaluate(start).as_long()
        
        # Convert start time to HH:MM format
        meeting_start_hr = 9 + start_val // 60
        meeting_start_min = start_val % 60
        meeting_end_val = start_val + meeting_duration
        meeting_end_hr = 9 + meeting_end_val // 60
        meeting_end_min = meeting_end_val % 60
        
        # Format output
        print(f"{meeting_start_hr:02d}:{meeting_start_min:02d}:{meeting_end_hr:02d}:{meeting_end_min:02d}")
        print("Monday")
    else:
        print("No solution found")

if __name__ == "__main__":
    main()
```
