from z3 import *

def main():
    # Initialize the optimizer
    opt = Optimize()
    
    # Define the day variable (0=Monday, 1=Tuesday, 2=Wednesday, 3=Thursday, 4=Friday)
    day = Int('day')
    # Define the start time variable (in minutes from 00:00)
    start_time = Int('start_time')
    
    # Meeting duration in minutes
    meeting_duration = 60
    meeting_end = start_time + meeting_duration
    
    # Work hours: 9:00 to 17:00 (540 to 1020 minutes)
    opt.add(start_time >= 540)
    opt.add(meeting_end <= 1020)
    opt.add(day >= 0, day <= 4)
    
    # Busy intervals for Brian and Julia (in minutes from 00:00)
    brian_busy = {
        0: [(570, 600), (750, 870), (930, 960)],   # Monday
        1: [(540, 570)],                           # Tuesday
        2: [(750, 840), (990, 1020)],              # Wednesday
        3: [(660, 690), (780, 810), (990, 1020)],  # Thursday
        4: [(570, 600), (630, 660), (780, 810), (900, 960), (990, 1020)]  # Friday
    }
    
    julia_busy = {
        0: [(540, 600), (660, 690), (750, 780), (930, 960)],  # Monday
        1: [(780, 840), (960, 990)],                          # Tuesday
        2: [(540, 690), (720, 750), (780, 1020)],             # Wednesday
        3: [(540, 630), (660, 1020)],                         # Thursday
        4: [(540, 600), (630, 690), (750, 840), (870, 900), (930, 960)]  # Friday
    }
    
    # Constraint: Meeting must not overlap with any busy interval for the selected day
    def no_overlap(busy_intervals, day_val, start, end):
        constraints = []
        for interval in busy_intervals.get(day_val, []):
            busy_start, busy_end = interval
            # Meeting must end before busy start or start after busy end
            constraints.append(Or(end <= busy_start, start >= busy_end))
        return And(constraints)
    
    # For each possible day, add constraints if that day is selected
    for d in range(5):
        brian_free = no_overlap(brian_busy, d, start_time, meeting_end)
        julia_free = no_overlap(julia_busy, d, start_time, meeting_end)
        opt.add(Implies(day == d, And(brian_free, julia_free)))
    
    # Preference: Avoid Monday if possible (assign higher cost to Monday)
    cost = If(day == 0, 100000, day * 1000 + start_time)
    opt.minimize(cost)
    
    # Check for a solution
    if opt.check() == sat:
        m = opt.model()
        d = m[day].as_long()
        start_minutes = m[start_time].as_long()
        
        # Convert minutes to time string
        start_h = start_minutes // 60
        start_m = start_minutes % 60
        end_minutes = start_minutes + meeting_duration
        end_h = end_minutes // 60
        end_m = end_minutes % 60
        
        days = ["Monday", "Tuesday", "Wednesday", "Thursday", "Friday"]
        day_str = days[d]
        time_str = f"{start_h:02d}:{start_m:02d}:{end_h:02d}:{end_m:02d}"
        print(f"{day_str} {time_str}")
    else:
        print("No solution found")

if __name__ == "__main__":
    main()
