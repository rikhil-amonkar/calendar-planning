    # Define variables
    day = Int('day')
    start = Int('start')

    # Create an optimizer
    opt = Optimize()

    # Check for a solution
    if opt.check() == sat:

        day_val = m[day].as_long()
        start_val = m[start].as_long()

        day_str = days[day_val]
