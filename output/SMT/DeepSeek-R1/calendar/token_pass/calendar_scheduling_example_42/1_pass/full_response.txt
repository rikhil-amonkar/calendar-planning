from z3 import *

def main():
    # Initialize solver
    s = Solver()
    
    # Work hours: 9:00 to 17:00 (8 hours = 480 minutes)
    total_minutes = 480
    meeting_duration = 60
    
    # Start time variable (in minutes from 9:00)
    start = Int('start')
    s.add(start >= 0)
    s.add(start + meeting_duration <= total_minutes)
    
    # Busy intervals in minutes from 9:00
    julie_busy = [(0, 30), (120, 150), (180, 210), (270, 300), (420, 480)]
    sean_busy = [(0, 30), (240, 270), (360, 390), (420, 450)]
    lori_busy = [(60, 90), (120, 240), (390, 480)]
    
    # Add constraints for each participant
    for (busy_start, busy_end) in julie_busy:
        s.add(Or(start + meeting_duration <= busy_start, start >= busy_end))
    
    for (busy_start, busy_end) in sean_busy:
        s.add(Or(start + meeting_duration <= busy_start, start >= busy_end))
    
    for (busy_start, busy_end) in lori_busy:
        s.add(Or(start + meeting_duration <= busy_start, start >= busy_end))
    
    # Check for solution
    if s.check() == sat:
        m = s.model()
        start_minutes = m.evaluate(start).as_long()
        
        # Convert start time to HH:MM format
        start_hour = 9 + start_minutes // 60
        start_minute = start_minutes % 60
        end_minutes = start_minutes + meeting_duration
        end_hour = 9 + end_minutes // 60
        end_minute = end_minutes % 60
        
        # Format output
        print(f"{start_hour:02d}:{start_minute:02d}:{end_hour:02d}:{end_minute:02d}")
        print("Monday")
    else:
        print("No solution found")

if __name__ == "__main__":
    main()
