```python
from z3 import *

def main():
    # Initialize solver
    s = Solver()
    
    # Work hours: 9:00 to 17:00 (9*60 to 17*60 minutes)
    work_start = 9 * 60
    work_end = 17 * 60
    meeting_duration = 60
    
    # Convert blocked times to minutes since midnight for easier calculation
    # Kayla's blocked intervals: 10:00-10:30, 14:30-16:00
    kayla_blocked = [(10*60, 10*60+30), (14*60+30, 16*60)]
    # Rebecca's blocked intervals: 9:00-13:00, 13:30-15:00, 15:30-16:00
    rebecca_blocked = [(9*60, 13*60), (13*60+30, 15*60), (15*60+30, 16*60)]
    
    # Meeting start time variable (in minutes from midnight)
    start = Int('start')
    
    # Constraint: Meeting must be within work hours
    s.add(start >= work_start)
    s.add(start + meeting_duration <= work_end)
    
    # Function to add blocked interval constraints
    def add_blocked_constraints(intervals):
        for block_start, block_end in intervals:
            # Meeting must not overlap with blocked interval
            s.add(Or(start + meeting_duration <= block_start, start >= block_end))
    
    # Add constraints for both participants
    add_blocked_constraints(kayla_blocked)
    add_blocked_constraints(rebecca_blocked)
    
    # Check for a solution
    if s.check() == sat:
        m = s.model()
        start_val = m.evaluate(start).as_long()
        
        # Convert start and end times to HH:MM format
        start_hour = start_val // 60
        start_minute = start_val % 60
        end_val = start_val + meeting_duration
        end_hour = end_val // 60
        end_minute = end_val % 60
        
        # Format the time range
        time_range = f"{start_hour:02d}:{start_minute:02d}:{end_hour:02d}:{end_minute:02d}"
        print(f"Monday {time_range}")
    else:
        print("No solution found")

if __name__ == "__main__":
    main()
```
