    #   We can express: if day==0, then the meeting must not overlap with any of these intervals.

    # Let's create a dictionary for the busy intervals:

    # Format: { day_index: { participant: list of (start_minute, end_minute) } }

    busy_schedule = {
        0: {  # Monday
            "Betty": [(60,90), (150,210), (420,450)],
            "Megan": [(0,480)]
        },
