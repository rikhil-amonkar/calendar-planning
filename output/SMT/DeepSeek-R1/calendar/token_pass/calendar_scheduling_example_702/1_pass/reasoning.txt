    from z3 import *

    # Define the variables
    day = Int('day')
    start = Int('start')

    # Define the day range and start range
    day_start_min = 540
    day_end_max = 1020
    meeting_duration = 30

    # Robert's busy intervals (by day)
    robert_busy = {
        0: [(660, 690), (840, 870), (930, 960)],   # Monday
        1: [(630, 660), (900, 930)],                # Tuesday
        2: [(600, 660), (690, 720), (750, 780), (810, 840), (900, 930), (960, 990)]  # Wednesday
    }

    # Ralph's busy intervals (by day)
    ralph_busy = {
        0: [(600, 810), (840, 870), (900, 1020)],
        1: [(540, 570), (600, 630), (660, 690), (720, 780), (840, 930), (960, 1020)],
        2: [(630, 660), (690, 720), (780, 870), (990, 1020)]
    }

    # Create solver and optimizer
    opt = Optimize()

    # Constraints for day and start
    opt.add(day >= 0, day <= 2)
    opt.add(start >= day_start_min, start <= day_end_max - meeting_duration)

    # For Robert: for each day and each busy interval in that day, add non-overlap constraint
    for d in range(3):
        for (s, e) in robert_busy[d]:
            # If the meeting is on day d, then it must not overlap with (s, e)
            opt.add(Or(day != d, start >= e, start + meeting_duration <= s))

        for (s, e) in ralph_busy[d]:
            opt.add(Or(day != d, start >= e, start + meeting_duration <= s))

    # Define the objective: avoid Monday if possible, then minimize start time
    penalty = If(day == 0, 100000, 0)
    objective = penalty + start
    opt.minimize(objective)

    # Check and get the solution
    if opt.check() == sat:
        m = opt.model()
        d_val = m[day].as_long()
        start_val = m[start].as_long()
        # Convert start_val to time string
        hours = start_val // 60
        minutes = start_val % 60
        end_val = start_val + meeting_duration
        end_hours = end_val // 60
        end_minutes = end_val % 60
        # Format the time range as HH:MM:HH:MM
        time_range = f"{hours:02d}:{minutes:02d}:{end_hours:02d}:{end_minutes:02d}"
        days = ['Monday', 'Tuesday', 'Wednesday']
        day_name = days[d_val]
        print(f"{day_name} {time_range}")
    else:
        print("No solution found")
