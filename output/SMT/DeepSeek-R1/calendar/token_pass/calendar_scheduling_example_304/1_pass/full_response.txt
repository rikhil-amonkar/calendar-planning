from z3 import *

def main():
    # Initialize solver
    s = Solver()
    
    # Meeting duration in minutes
    meeting_duration = 30
    
    # Work hours: 9:00 to 17:00 (9 AM to 5 PM) in minutes from 9:00 as 0
    work_start = 0
    work_end = 8 * 60  # 8 hours * 60 minutes = 480 minutes (17:00)
    
    # Start time variable (in minutes from 9:00)
    S = Int('S')
    
    # Constraints for work hours
    s.add(S >= work_start)
    s.add(S + meeting_duration <= work_end)
    
    # Janice's preference: meeting should end by 13:00 (240 minutes from 9:00)
    s.add(S + meeting_duration <= 240)
    
    # Christine's busy intervals (in minutes from 9:00)
    christine_busy = [(30, 90), (180, 210), (240, 270), (330, 360), (420, 450)]
    for start, end in christine_busy:
        s.add(Or(S + meeting_duration <= start, S >= end))
    
    # Bobby's busy intervals
    bobby_busy = [(180, 210), (330, 360)]
    for start, end in bobby_busy:
        s.add(Or(S + meeting_duration <= start, S >= end))
    
    # Elizabeth's busy intervals
    elizabeth_busy = [(0, 30), (150, 240), (270, 300), (360, 390), (420, 480)]
    for start, end in elizabeth_busy:
        s.add(Or(S + meeting_duration <= start, S >= end))
    
    # Tyler's busy intervals
    tyler_busy = [(0, 120), (180, 210), (240, 270), (390, 420), (450, 480)]
    for start, end in tyler_busy:
        s.add(Or(S + meeting_duration <= start, S >= end))
    
    # Edward's busy intervals
    edward_busy = [(0, 30), (60, 120), (150, 300), (330, 390), (420, 480)]
    for start, end in edward_busy:
        s.add(Or(S + meeting_duration <= start, S >= end))
    
    # Check for a solution
    if s.check() == sat:
        m = s.model()
        start_minutes = m.eval(S).as_long()
        
        # Convert start minutes to time string
        start_hour = 9 + start_minutes // 60
        start_minute = start_minutes % 60
        end_minutes = start_minutes + meeting_duration
        end_hour = 9 + end_minutes // 60
        end_minute = end_minutes % 60
        
        # Format the time range as HH:MM:HH:MM
        time_range = f"{start_hour:02d}:{start_minute:02d}:{end_hour:02d}:{end_minute:02d}"
        print(f"Monday {time_range}")
    else:
        print("No solution found")

if __name__ == "__main__":
    main()
