        from z3 import *

        required_days = [4,5,4,3,3,4]
        edges = [(0,4), (4,0), (4,1), (1,4), (1,3), (3,1), (0,5), (5,0), (0,2), (2,0), (5,4), (4,5)]

        s = Solver()
        order = [Int('order_%i' % i) for i in range(6)]

        # Domain constraints
        for i in range(6):
            s.add(And(order[i] >= 0, order[i] <= 5))
        s.add(Distinct(order))
        s.add(order[5] == 3)   # Porto last

        # Flight constraints for consecutive cities
        for i in range(5):
            s.add(Or([And(order[i] == a, order[i+1] == b) for (a,b) in edges]))

        if s.check() == sat:
            m = s.model()
            order_val = [m.evaluate(order[i]).as_long() for i in range(6)]
            # Then compute the day ranges for each city in the order.
            start_days = []
            end_days = []
            current_day = 1
            for i in range(6):
                city = order_val[i]
                duration = required_days[city]
                end_day = current_day + duration - 1
                start_days.append(current_day)
                end_days.append(end_day)
                current_day = end_day   # because the next city starts on the same day (travel day)

            # Now, create the itinerary list of dictionaries
            itinerary = []
            for i in range(6):
                city_index = order_val[i]
                city_names = ['Helsinki', 'Valencia', 'Dubrovnik', 'Porto', 'Prague', 'Reykjavik']
                place = city_names[city_index]
                start = start_days[i]
                end = end_days[i]
                day_range = f"Day {start}-{end}"
                itinerary.append({"day_range": day_range, "place": place})

            # Output as JSON
            import json
            print(json.dumps({"itinerary": itinerary}))
        else:
            print("No solution found")
