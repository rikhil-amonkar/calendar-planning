from z3 import *

def main():
    s = Solver()
    
    S = 15
    cities = ['Salzburg', 'Venice', 'Bucharest', 'Brussels', 'Hamburg', 'Copenhagen', 'Nice', 'Zurich', 'Naples']
    req_days = [2, 5, 4, 2, 4, 4, 3, 5, 4]
    
    city_seg = [Int(f"city_seg_{i}") for i in range(S)]
    days_seg = [Int(f"days_seg_{i}") for i in range(S)]
    start_seg = [Int(f"start_seg_{i}") for i in range(S)]
    
    s.add(start_seg[0] == 1)
    s.add(days_seg[0] >= 1)
    s.add(city_seg[0] >= 0, city_seg[0] <= 8)
    
    for i in range(1, S):
        s.add(start_seg[i] == start_seg[i-1] + days_seg[i-1] - 1)
        s.add(days_seg[i] >= 0)
        s.add(If(days_seg[i] > 0, And(city_seg[i] >= 0, city_seg[i] <= 8), True))
    
    s.add(start_seg[S-1] + days_seg[S-1] - 1 == 25)
    
    for c in range(9):
        total = Sum([If(And(days_seg[i] > 0, city_seg[i] == c), days_seg[i], 0) for i in range(S)])
        s.add(total == req_days[c])
    
    edges = [
        (0,4), (2,5), (1,3), (6,7), (4,6), (7,8), (4,2), (7,5), (2,3), (4,3),
        (1,8), (1,5), (2,8), (4,5), (6,8), (6,5), (1,7), (3,8), (5,3), (1,6),
        (0,4), (7,2), (3,8), (5,3), (1,6), (6,3), (4,1), (5,8), (6,8), (4,7),
        (0,4), (7,2), (3,8), (5,3), (1,6), (6,5)
    ]
    graph_set = set()
    for a, b in edges:
        graph_set.add((a, b))
        graph_set.add((b, a))
    
    for i in range(1, S):
        condition = And(days_seg[i] > 0, days_seg[i-1] > 0)
        allowed_flights = Or([And(city_seg[i-1] == a, city_seg[i] == b) for a, b in graph_set])
        s.add(If(condition, allowed_flights, True))
    
    def covers(i, d):
        return And(d >= start_seg[i], d <= start_seg[i] + days_seg[i] - 1)
    
    for d in [21, 22]:
        s.add(Or([And(city_seg[i] == 3, covers(i, d)) for i in range(S)]))
    for d in [18, 19, 20, 21]:
        s.add(Or([And(city_seg[i] == 5, covers(i, d)) for i in range(S)]))
    for d in [9, 10, 11]:
        s.add(Or([And(city_seg[i] == 6, covers(i, d)) for i in range(S)]))
    for d in [22, 23, 24, 25]:
        s.add(Or([And(city_seg[i] == 8, covers(i, d)) for i in range(S)]))
    
    if s.check() == sat:
        m = s.model()
        itinerary = []
        for i in range(S):
            days_val = m.evaluate(days_seg[i]).as_long()
            if days_val > 0:
                start_val = m.evaluate(start_seg[i]).as_long()
                city_val = m.evaluate(city_seg[i]).as_long()
                end_val = start_val + days_val - 1
                day_range = f"Day {start_val}-{end_val}"
                itinerary.append({"day_range": day_range, "place": cities[city_val]})
        print({"itinerary": itinerary})
    else:
        print({"itinerary": []})

if __name__ == "__main__":
    main()
