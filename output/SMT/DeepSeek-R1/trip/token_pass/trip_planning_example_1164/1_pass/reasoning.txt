     flights = [

     for (a,b) in flights:
         allowed_pairs.add( (a, b) )
         allowed_pairs.add( (b, a) )

     # Create the Z3 solver and variables
     x = [ Int('x_{}'.format(i)) for i in range(1, n_days+1) ]
     solver = Solver()

     # Constraint: each x[i] is between 0 and 7
     for i in range(n_days):
         solver.add( x[i] >= 0, x[i] < n_cities )

     # Constraint for the required days for each city
     for c in range(n_cities):
         total = 0
         for d in range(1, n_days+1):   # d from 1 to 17
             # Morning of day d
             total += If(x[d-1] == c, 1, 0)
             if d < n_days:   # evening travel for day d (if we travel on day d to c)
                 total += If(And(x[d-1] != x[d], x[d] == c), 1, 0)
