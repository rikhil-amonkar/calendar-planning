        from z3 import *

        # City indices: 0-Split, 1-Helsinki, 2-Reykjavik, 3-Vilnius, 4-Geneva
        req_array = [2,2,3,3,6]
        city_names = ['Split', 'Helsinki', 'Reykjavik', 'Vilnius', 'Geneva']

        allowed_flights = set([(0,1), (1,0), (0,4), (4,0), (1,4), (4,1), (1,2), (2,1), (1,3), (3,1), (0,3), (3,0)])

        # Create order variables
        order0 = Int('order0')
        order1 = Int('order1')
        order2 = Int('order2')
        order3 = Int('order3')
        order4 = Int('order4')
        orders = [order0, order1, order2, order3, order4]

        s = Solver()

        # Each order variable between0 and4
        for o in orders:
            s.add(o >=0, o<=4)

        s.add(Distinct(orders))

        # Compute start days for each segment
        start0 = 1
        start1 = req_array[order0]
        start2 = req_array[order0] + req_array[order1] - 1
        start3 = req_array[order0] + req_array[order1] + req_array[order2] - 2
        start4 = req_array[order0] + req_array[order1] + req_array[order2] + req_array[order3] - 3

        # Reykjavik is city2, Vilnius is city3
        # For Reykjavik: find which segment it is in and get its start day.
        reykjavik_start = If(order0==2, start0, 
                          If(order1==2, start1,
                          If(order2==2, start2,
                          If(order3==2, start3,
                          start4))))

        s.add(reykjavik_start >=8, reykjavik_start<=10)

        vilnius_start = If(order0==3, start0,
                        If(order1==3, start1,
                        If(order2==3, start2,
                        If(order3==3, start3,
                        start4))))

        s.add(vilnius_start>=5, vilnius_start<=9)

        # Direct flights between consecutive segments
        for i in range(4):
            # Check (orders[i], orders[i+1]) is in allowed_flights
            constraints = []
            for (a,b) in allowed_flights:
                constraints.append(And(orders[i]==a, orders[i+1]==b))
            s.add(Or(constraints))

        # Check if sat
        if s.check() == sat:
            m = s.model()
            # Get the order of the cities
            order_values = [m.evaluate(o).as_long() for o in orders]
            # Now, we have the order of the cities by index.
            # Then, we can compute the start day for each segment.
            start_days = [1]
            start_days.append(req_array[order_values[0]])
            start_days.append(req_array[order_values[0]] + req_array[order_values[1]] - 1)
            start_days.append(req_array[order_values[0]] + req_array[order_values[1]] + req_array[order_values[2]] - 2)
            start_days.append(req_array[order_values[0]] + req_array[order_values[1]] + req_array[order_values[2]] + req_array[order_values[3]] - 3)

            # The end day of each segment is start day + req_array[city] - 1
            segments = []
            for i in range(5):
                city_index = order_values[i]
                city_name = city_names[city_index]
                start = start_days[i]
                end = start + req_array[city_index] - 1
                segments.append((start, end, city_name))

            # Now, sort the segments by start day to output the itinerary in chronological order.
            segments.sort(key=lambda x: x[0])

            # Create the itinerary list of dictionaries
            itinerary = []
            for seg in segments:
                start, end, city = seg
                if start == end:
                    day_range = f"Day {start}"
                else:
                    day_range = f"Day {start}-{end}"
                itinerary.append({"day_range": day_range, "place": city})

            # Output as JSON
            import json
            print(json.dumps({"itinerary": itinerary}))
        else:
            print("No solution found")
