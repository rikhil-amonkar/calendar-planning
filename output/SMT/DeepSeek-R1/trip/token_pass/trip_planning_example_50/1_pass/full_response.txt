from z3 import *
import json

def main():
    n_days = 12
    V = [Bool('V_%d' % i) for i in range(n_days)]
    M = [Bool('M_%d' % i) for i in range(n_days)]
    K = [Bool('K_%d' % i) for i in range(n_days)]
    
    s = Solver()
    
    for i in range(n_days):
        s.add(Or(V[i], M[i], K[i]))
        s.add(Not(And(V[i], M[i], K[i])))
        s.add(Not(And(V[i], K[i])))
    
    s.add(Sum([If(V[i], 1, 0) for i in range(n_days)]) == 4)
    s.add(Sum([If(M[i], 1, 0) for i in range(n_days)]) == 3)
    s.add(Sum([If(K[i], 1, 0) for i in range(n_days)]) == 7)
    
    for i in range(n_days - 1):
        s.add(Or(And(V[i], V[i+1]), And(M[i], M[i+1]), And(K[i], K[i+1])))
    
    s.add(V[0])
    for i in range(n_days - 1):
        s.add(Implies(Not(V[i]), Not(V[i+1])))
    
    if s.check() == sat:
        m = s.model()
        city_sets = []
        for i in range(n_days):
            cities = []
            if is_true(m[V[i]]):
                cities.append("Vilnius")
            if is_true(m[M[i]]):
                cities.append("Munich")
            if is_true(m[K[i]]):
                cities.append("Mykonos")
            city_sets.append(cities)
        
        segments = []
        start = 0
        for i in range(1, n_days):
            if city_sets[i] != city_sets[i-1]:
                end = i - 1
                if start == end:
                    day_range = f"Day {start+1}"
                else:
                    day_range = f"Day {start+1}-{end+1}"
                segments.append({"day_range": day_range, "place": " and ".join(city_sets[start])})
                start = i
        if start < n_days:
            if start == n_days - 1:
                day_range = f"Day {start+1}"
            else:
                day_range = f"Day {start+1}-{n_days}"
            segments.append({"day_range": day_range, "place": " and ".join(city_sets[start])})
        
        result = {"itinerary": segments}
        print(json.dumps(result))
    else:
        print('{"itinerary": []}')

if __name__ == "__main__":
    main()
