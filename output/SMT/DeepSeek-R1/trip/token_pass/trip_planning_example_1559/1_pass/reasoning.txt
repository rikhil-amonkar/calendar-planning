      s = Solver()

      # Constraints on order: each element is between 0 and 9, and distinct
      for i in range(10):
          s.add(And(order[i] >= 0, order[i] < 10))
      s.add(Distinct(order))

      # Constraints for start and end of each city
      for i in range(10):
          # Duration constraint
          s.add(end[i] - start[i] + 1 == required_days[i])
          # If the city is fixed, set start and end
          if fixed_start[i] is not None:
              s.add(start[i] == fixed_start[i])
          if fixed_end[i] is not None:
              s.add(end[i] == fixed_end[i])

      # Constraints for the sequence
      # First city starts at 1
      s.add(start[order[0]] == 1)
      # Last city ends at 25
      s.add(end[order[9]] == 25)
      # For consecutive cities: end[i] = start[i+1]
      for i in range(9):
          s.add(end[order[i]] == start[order[i+1]])

      # Constraints for direct flights between consecutive cities
      for i in range(9):
          city_i = order[i]
          city_j = order[i+1]
          # Check if (city_i, city_j) is in allowed_edges
          s.add(Or([And(city_i == a, city_j == b) for (a,b) in allowed_edges]))

      # Check if the solution is satisfiable
      if s.check() == sat:
          m = s.model()
          # Extract the order
          order_val = [m.evaluate(order[i]).as_long() for i in range(10)]
          # Extract start and end for each city
          start_val = [m.evaluate(start[i]).as_long() for i in range(10)]
          end_val = [m.evaluate(end[i]).as_long() for i in range(10)]

          # Now, we need to output the itinerary in the order of days. But note: the days are continuous and each day is covered by one or two cities (on flight days).

          # However, the output should be a list of day-place mappings for contiguous days in the same city.

          # We will create a list of segments: for each city in the order, we have a segment from start day to end day.

          # But note: the same city might appear non-contiguously? We assumed each city appears exactly once.

          itinerary = []
          for i in range(10):
              city_index = order_val[i]
              city_name = cities[city_index]
              s_day = start_val[city_index]
              e_day = end_val[city_index]
              itinerary.append( (s_day, e_day, city_name) )

          # Now, sort by start day?
          # Actually, the order of the segments is already by the travel order, but the days are continuous so we can also output by day.

          # The output format: [{"day_range": "Day 1-5", "place": "Helsinki"}, ...]

          # We will create a list of segments for each city visit.

          segments = []
          for (s, e, city) in itinerary:
              segments.append( (s, e, city) )

          # Sort by start day
          segments.sort(key=lambda x: x[0])

          # Now, create the JSON list
          result = []
          for seg in segments:
              s, e, city = seg
              if s == e:
                  day_range = "Day {}".format(s)
              else:
                  day_range = "Day {}-{}".format(s, e)
              result.append({"day_range": day_range, "place": city})

          # Output as JSON
          import json
          print(json.dumps({"itinerary": result}))
