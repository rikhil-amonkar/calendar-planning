from z3 import *
import json

def main():
    # Define the cities and required days
    cities = ['Amsterdam', 'Bucharest', 'Frankfurt', 'London', 'Riga', 'Stockholm', 'Vilnius']
    required_days = {
        'Amsterdam': 2,
        'Bucharest': 4,
        'Frankfurt': 3,
        'London': 2,
        'Riga': 2,
        'Stockholm': 3,
        'Vilnius': 5
    }
    
    # Define direct flights as undirected edges
    direct_flights = [
        ('London', 'Amsterdam'),
        ('Vilnius', 'Frankfurt'),
        ('Riga', 'Vilnius'),
        ('Riga', 'Stockholm'),
        ('London', 'Bucharest'),
        ('Amsterdam', 'Stockholm'),
        ('Amsterdam', 'Frankfurt'),
        ('Frankfurt', 'Stockholm'),
        ('Bucharest', 'Riga'),
        ('Amsterdam', 'Riga'),
        ('Amsterdam', 'Bucharest'),
        ('Riga', 'Frankfurt'),
        ('Bucharest', 'Frankfurt'),
        ('London', 'Frankfurt'),
        ('London', 'Stockholm'),
        ('Amsterdam', 'Vilnius')
    ]
    # Create a set of connected city pairs (sorted)
    connected = set()
    for a, b in direct_flights:
        connected.add(tuple(sorted((a, b))))
    
    n_days = 15
    s = Solver()
    
    # Create a grid of variables: presence[day][city]
    presence = [[Bool(f"day_{day}_{city}") for city in cities] for day in range(1, n_days+1)]
    
    # Constraint: Each day must have at least one city and at most two cities
    for day in range(n_days):
        s.add(Or(presence[day]))  # At least one city per day
        # At most two cities per day
        for i in range(len(cities)):
            for j in range(i+1, len(cities)):
                for k in range(j+1, len(cities)):
                    s.add(Not(And(presence[day][i], presence[day][j], presence[day][k])))
    
    # Constraint: If two cities on same day, they must be connected
    for day in range(n_days):
        for i in range(len(cities)):
            for j in range(i+1, len(cities)):
                city_i = cities[i]
                city_j = cities[j]
                if tuple(sorted((city_i, city_j))) not in connected:
                    s.add(Not(And(presence[day][i], presence[day][j])))
    
    # Constraint: Total days per city must match requirements
    for c_idx, city in enumerate(cities):
        total = 0
        for day in range(n_days):
            total += If(presence[day][c_idx], 1, 0)
        s.add(total == required_days[city])
    
    # Specific constraints: Amsterdam between day 2 and 3 (1-indexed days 1 and 2 in zero-indexed)
    s.add(Or(presence[1][cities.index('Amsterdam')], presence[2][cities.index('Amsterdam')]))
    # Vilnius between day 7 and 11 (zero-indexed 6 to 10)
    vilnius_idx = cities.index('Vilnius')
    s.add(Or([presence[i][vilnius_idx] for i in range(6, 11)]))
    # Stockholm between day 13 and 15 (zero-indexed 12 to 14)
    stockholm_idx = cities.index('Stockholm')
    s.add(Or([presence[i][stockholm_idx] for i in range(12, 15)]))
    
    # Check and get model
    if s.check() == sat:
        m = s.model()
        # Decode the presence matrix
        itinerary = []
        current_set = None
        start_day = 1
        for day in range(n_days):
            day_cities = []
            for c_idx, city in enumerate(cities):
                if is_true(m.evaluate(presence[day][c_idx])):
                    day_cities.append(city)
            day_cities.sort()
            current_str = ", ".join(day_cities)
            if current_set is None:
                current_set = current_str
                start_day = day+1
            else:
                if current_str != current_set:
                    itinerary.append({
                        "day_range": f"Day {start_day}-{day}",
                        "place": current_set
                    })
                    current_set = current_str
                    start_day = day+1
        # Add the last segment
        itinerary.append({
            "day_range": f"Day {start_day}-{n_days}",
            "place": current_set
        })
        
        # Output as JSON
        print(json.dumps({"itinerary": itinerary}))
    else:
        print('{"itinerary": []}')

if __name__ == "__main__":
    main()
