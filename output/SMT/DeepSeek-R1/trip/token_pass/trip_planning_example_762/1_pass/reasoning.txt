        from z3 import *

        # Define the cities
        cities = ['Dublin', 'Madrid', 'Oslo', 'London', 'Vilnius', 'Berlin']
        cityEnum = Enum('City', cities)
        req = [3,2,3,2,3,5]  # corresponding to cities order

        # Direct flights: list of pairs of city names, then convert to indices
        direct_flights = [('London','Madrid'), ('Oslo','Vilnius'), ('Berlin','Vilnius'), ('Madrid','Oslo'), ('Madrid','Dublin'), ('London','Oslo'), ('Madrid','Berlin'), ('Berlin','Oslo'), ('Dublin','Oslo'), ('London','Dublin'), ('London','Berlin'), ('Berlin','Dublin')]
        # Convert to indices using the cities list
        direct_flights_indices = []
        for a,b in direct_flights:
            ia = cities.index(a)
            ib = cities.index(b)
            direct_flights_indices.append((ia, ib))
            direct_flights_indices.append((ib, ia)) # make symmetric

        # Create a set of connected pairs (as tuples with sorted indices to avoid duplicates) but we will use a list and check both orders.
        # We can create a function that checks if two city indices are connected.

        def is_connected(a, b):
            return Or([And(a == i, b == j) for i,j in direct_flights_indices])

        # Create an array city for 0..13
        city = [Const('city_{}'.format(i), cityEnum) for i in range(14)]

        s = Solver()

        # Constraint: total days for each city must meet the requirement.
        for c_idx, c in enumerate(cities):
            total = 0
            # First part: count city[i] for i in [0,12]
            for i in range(0, 13): # i from 0 to 12
                total = total + If(city[i] == cityEnum[c_idx], 1, 0)
            # Second part: for i from 1 to 13, if city[i] != city[i-1] and city[i] == c_idx, then add 1
            for i in range(1,14):
                total = total + If(And(city[i] != city[i-1], city[i] == cityEnum[c_idx]), 1, 0)
            s.add(total == req[c_idx])

        # Constraint: for each travel day (i from 1 to 13), if city[i] != city[i-1], then the flight must be direct.
        for i in range(1,14):
            s.add(If(city[i] != city[i-1], is_connected(city[i-1], city[i]), True))

        # Specific day constraints:
        dublin_idx = cities.index('Dublin')
        madrid_idx = cities.index('Madrid')
        berlin_idx = cities.index('Berlin')

        # Dublin between day7 and day9: for day d in [7,8,9] (d is the day index from 1 to 13)
        # On day d, we are in Dublin if city[d-1] is Dublin or city[d] is Dublin.
        for d in [7,8,9]:
            s.add(Or(city[d-1] == cityEnum[dublin_idx], city[d] == cityEnum[dublin_idx]))

        # Madrid between day2 and day3
        for d in [2,3]:
            s.add(Or(city[d-1] == cityEnum[madrid_idx], city[d] == cityEnum[madrid_idx]))

        # Berlin between day3 and day7
        for d in [3,4,5,6,7]:
            s.add(Or(city[d-1] == cityEnum[berlin_idx], city[d] == cityEnum[berlin_idx]))

        # Check and get the model
        if s.check() == sat:
            m = s.model()
            # Get the values for city[0] to city[13]
            city_values = [m.evaluate(city[i]) for i in range(14)]
            # Convert to city names
            city_names = [str(city_values[i]) for i in range(14)]
            # Now, create the itinerary segments from city[1] to city[13] (for days 1 to 13)
            segments = []
            start_day = 1
            current_city = city_names[1]
            for day in range(2, 14): # day corresponds to city[1] is day1, city[2] is day2, ... city[13] is day13
                if city_names[day] != current_city:
                    segments.append((start_day, day, current_city))
                    start_day = day
                    current_city = city_names[day]
            segments.append((start_day, 13, current_city))

            # Now format the segments as day ranges
            itinerary = []
            for seg in segments:
                start, end, city = seg
                if start == end:
                    day_range = "Day {}".format(start)
                else:
                    day_range = "Day {}-{}".format(start, end)
                itinerary.append({"day_range": day_range, "place": city})

            # Output as JSON
            import json
            print(json.dumps({"itinerary": itinerary}))
        else:
            print("No solution found")
