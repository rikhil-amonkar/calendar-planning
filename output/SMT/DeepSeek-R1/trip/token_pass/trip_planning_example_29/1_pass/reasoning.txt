   from z3 import *

   # Create solver
   s = Solver()

   # We have 10 days, indexed 0 to 9 (representing day1 to day10)
   n_days = 10
   days = range(n_days)

   # For each day, we have three booleans for each city
   in_K = [Bool(f'in_K_{i}') for i in days]
   in_D = [Bool(f'in_D_{i}') for i in days]
   in_F = [Bool(f'in_F_{i}') for i in days]

   # Constraints for each day
   for i in days:
       # At least one city
       s.add(Or(in_K[i], in_D[i], in_F[i]))

       # At most two cities
       # We can say: not all three are true
       s.add(Not(And(in_K[i], in_D[i], in_F[i])))

       # If two cities are true, then they must be connected
       # There are three possible pairs: (K,D), (K,F), (D,F)
       # For (K,D): not allowed -> so if both are true, then constraint violated
       # For (K,F): allowed
       # For (D,F): allowed

       # So we add: if both K and D are true, then false -> so we forbid (K and D)
       s.add(Implies(And(in_K[i], in_D[i]), False))

       # Note: the other two pairs are allowed.

   # Consecutive days must share at least one city
   for i in range(n_days-1):
       # Day i and i+1 must share at least one city
       s.add(Or(
           And(in_K[i], in_K[i+1]),
           And(in_D[i], in_D[i+1]),
           And(in_F[i], in_F[i+1])
       ))

   # Total days constraints
   s.add(Sum([If(in_K[i], 1, 0) for i in days]) == 2)
   s.add(Sum([If(in_D[i], 1, 0) for i in days]) == 7)
   s.add(Sum([If(in_F[i], 1, 0) for i in days]) == 3)

   # Wedding constraints: must be in Krakow on day9 and day10 (which are index 8 and 9)
   s.add(in_K[8])
   s.add(in_K[9])

   # Check satisfiability
   if s.check() == sat:
       m = s.model()
       # Now, we need to get the value for each day and each city
       # We'll create a list of the set of cities for each day
       itinerary = []
       for i in days:
           cities = []
           if is_true(m.evaluate(in_K[i])):
               cities.append('Krakow')
           if is_true(m.evaluate(in_D[i])):
               cities.append('Dubrovnik')
           if is_true(m.evaluate(in_F[i])):
               cities.append('Frankfurt')
           # Sort the cities for consistent representation
           cities.sort()
           itinerary.append(tuple(cities))
       # Now group consecutive days with the same set
       segments = []
       start = 0
       current_set = itinerary[0]
       for i in range(1, n_days):
           if itinerary[i] == current_set:
               continue
           else:
               # Output segment from start to i-1
               seg_length = i - start
               if seg_length == 1:
                   day_range = f"Day {start+1}"
               else:
                   day_range = f"Day {start+1}-{i}"
               place = current_set[0] if len(current_set)==1 else " and ".join(current_set)
               segments.append({"day_range": day_range, "place": place})
               start = i
               current_set = itinerary[i]
       # Add the last segment
       seg_length = n_days - start
       if seg_length == 1:
           day_range = f"Day {start+1}"
       else:
           day_range = f"Day {start+1}-{n_days}"
       place = current_set[0] if len(current_set)==1 else " and ".join(current_set)
       segments.append({"day_range": day_range, "place": place})

       # Output as JSON
       import json
       result = {"itinerary": segments}
       print(json.dumps(result, indent=2))
   else:
       print('No solution found')
