        from z3 import *
        import json

        # Define cities and mapping
        cities = ['Vienna', 'Stockholm', 'Nice', 'Split']
        city_dict = {0: 'Vienna', 1: 'Stockholm', 2: 'Nice', 3: 'Split'}

        # Create solver
        s = Solver()

        # Create variables
        c = [Int('c_%d' % i) for i in range(9)]
        travel = [Bool('travel_%d' % i) for i in range(8)]

        # Each c[i] must be between 0 and 3
        for i in range(9):
            s.add(And(0 <= c[i], c[i] <= 3))

        # Define travel constraints
        for i in range(8):
            s.add(travel[i] == (c[i] != c[i+1]))

        # Define allowed direct flights (both orders)
        allowed_pairs = [(0,1), (1,0), (0,2), (2,0), (0,3), (3,0), (1,3), (3,1), (1,2), (2,1)]
        for i in range(8):
            # If travel[i] is true, then (c[i], c[i+1]) must be in allowed_pairs
            s.add(Implies(travel[i], Or([And(c[i] == a, c[i+1] == b) for (a,b) in allowed_pairs])))

        # Events constraints
        s.add(c[0] == 0)   # Vienna on day1
        s.add(Or(c[5] == 3, c[6] == 3))   # Split on day7
        s.add(Or(c[7] == 3, c[8] == 3))   # Split on day9

        # Count constraints
        counts = [Int('count_%d' % i) for i in range(4)]
        for city in range(4):
            total = If(c[0] == city, 1, 0)
            for i in range(1,9):   # i from 1 to 8 (index in c for day i+1)
                total = total + If(travel[i-1],
                                  If(c[i-1] == city, 1, 0) + If(c[i] == city, 1, 0),
                                  If(c[i] == city, 1, 0))
            s.add(counts[city] == total)

        s.add(counts[0] == 2)
        s.add(counts[1] == 5)
        s.add(counts[2] == 2)
        s.add(counts[3] == 3)

        # Check and get model
        if s.check() == sat:
            m = s.model()
            # Get the values for c
            c_values = [m.evaluate(c[i]).as_long() for i in range(9)]
            # Now, group consecutive days with the same city
            itinerary = []
            start = 0
            current_city = c_values[0]
            for i in range(1,9):
                if c_values[i] != current_city:
                    end_day = i   # the last day of the current segment is day i (0-indexed, so day number i+1) but we are at the change at day i (so the previous segment ends at day i-1, and the new segment starts at day i)
                    # Actually, our segment from start to i-1 (inclusive) is the same city.
                    day_range = f"Day {start+1}-{i}" if start+1 != i else f"Day {start+1}"
                    itinerary.append({"day_range": day_range, "place": city_dict[current_city]})
                    start = i
                    current_city = c_values[i]
            # Add the last segment
            day_range = f"Day {start+1}-9" if start+1 != 9 else f"Day {start+1}"
            itinerary.append({"day_range": day_range, "place": city_dict[current_city]})

            # Output as JSON
            print(json.dumps({"itinerary": itinerary}))
        else:
            print('No solution found')
