import json
from z3 import *

def main():
    # City indices
    cities = {
        "Santorini": 0,
        "Krakow": 1,
        "Paris": 2,
        "Vilnius": 3,
        "Munich": 4,
        "Geneva": 5,
        "Amsterdam": 6,
        "Budapest": 7,
        "Split": 8
    }
    city_names = ["Santorini", "Krakow", "Paris", "Vilnius", "Munich", "Geneva", "Amsterdam", "Budapest", "Split"]
    
    required_days = [5, 5, 5, 3, 5, 2, 4, 5, 4]
    
    # Fixed constraints
    fixed_s = [25, 18, 11, None, None, None, None, None, None]
    fixed_e = [29, 22, 15, None, None, None, None, None, None]
    
    # Direct flights (undirected edges)
    edges = [
        (1, 2), (2, 6), (2, 8), (3, 4), (2, 5), (5, 6), (4, 8), (1, 8), (4, 6),
        (6, 7), (5, 8), (3, 8), (4, 5), (1, 4), (1, 3), (3, 6), (2, 7), (1, 6),
        (2, 3), (5, 7), (6, 8), (0, 5), (0, 6), (4, 7), (2, 4)
    ]
    allowed_edges = set()
    for a, b in edges:
        allowed_edges.add((a, b))
        allowed_edges.add((b, a))
    
    s = [Int(f"s_{i}") for i in range(9)]
    e = [Int(f"e_{i}") for i in range(9)]
    order = [Int(f"order_{i}") for i in range(9)]
    
    solver = Solver()
    
    # Fixed city constraints
    for i in range(3):
        solver.add(s[i] == fixed_s[i])
        solver.add(e[i] == fixed_e[i])
    
    # Duration constraints
    for i in range(9):
        solver.add(e[i] - s[i] + 1 == required_days[i])
        solver.add(s[i] >= 1)
        solver.add(e[i] <= 30)
        solver.add(s[i] <= e[i])
    
    # Permutation constraints
    solver.add(Distinct(order))
    for i in range(9):
        solver.add(order[i] >= 0)
        solver.add(order[i] < 9)
    
    # First and last city constraints
    solver.add(s[order[0]] == 1)
    solver.add(e[order[8]] == 30)
    
    # Consecutive city constraints
    for i in range(8):
        solver.add(e[order[i]] == s[order[i+1]])
        solver.add((order[i], order[i+1]) in allowed_edges)
    
    if solver.check() == sat:
        model = solver.model()
        order_val = [model.evaluate(order[i]).as_long() for i in range(9)]
        s_val = [model.evaluate(s[i]).as_long() for i in range(9)]
        e_val = [model.evaluate(e[i]).as_long() for i in range(9)]
        
        itinerary = []
        for city_index in order_val:
            start = s_val[city_index]
            end = e_val[city_index]
            itinerary.append({
                "day_range": f"Day {start}-{end}",
                "place": city_names[city_index]
            })
        
        print(json.dumps({"itinerary": itinerary}))
    else:
        print('{"itinerary": []}')

if __name__ == "__main__":
    main()
