from z3 import *
import json

def main():
    # City codes
    cities = {
        0: "Bucharest",
        1: "Venice",
        2: "Prague",
        3: "Frankfurt",
        4: "Zurich",
        5: "Florence",
        6: "Tallinn"
    }
    
    # Direct flights (undirected)
    direct_flights = set()
    connections = [
        (2, 6), (6, 2),  # Prague and Tallinn
        (2, 4), (4, 2),  # Prague and Zurich
        (5, 2), (2, 5),  # Florence and Prague
        (3, 0), (0, 3),  # Frankfurt and Bucharest
        (3, 1), (1, 3),  # Frankfurt and Venice
        (2, 0), (0, 2),  # Prague and Bucharest
        (0, 4), (4, 0),  # Bucharest and Zurich
        (6, 3), (3, 6),  # Tallinn and Frankfurt
        (4, 5), (5, 4),  # Zurich to Florence
        (3, 4), (4, 3),  # Frankfurt and Zurich
        (4, 1), (1, 4),  # Zurich and Venice
        (5, 3), (3, 5),  # Florence and Frankfurt
        (2, 3), (3, 2),  # Prague and Frankfurt
        (6, 4), (4, 6)   # Tallinn and Zurich
    ]
    for conn in connections:
        direct_flights.add(conn)
    
    n_days = 26
    n_cities = 7
    
    # Z3 variables
    start_city = [Int(f"start_city_{i}") for i in range(n_days)]
    travel = [Bool(f"travel_{i}") for i in range(n_days - 1)]
    
    solver = Solver()
    
    # Constraint: start_city within [0, 6]
    for i in range(n_days):
        solver.add(And(start_city[i] >= 0, start_city[i] < n_cities))
    
    # Constraints for travel and flight connections
    for i in range(n_days - 1):
        # If travel, then cities must be different and connected
        solver.add(Implies(travel[i], 
                           And(start_city[i] != start_city[i+1],
                              Or([And(start_city[i] == a, start_city[i+1] == b) for (a,b) in direct_flights])
                           )))
        # If not travel, then same city
        solver.add(Implies(Not(travel[i]), start_city[i] == start_city[i+1]))
    
    # Total travel days = 6
    solver.add(Sum([If(travel[i], 1, 0) for i in range(n_days-1)]) == 6)
    
    # Constraints for total days per city
    req_days = {
        0: 3,  # Bucharest
        1: 5,  # Venice
        2: 4,  # Prague
        3: 5,  # Frankfurt
        4: 5,  # Zurich
        5: 5,  # Florence
        6: 5   # Tallinn
    }
    
    for city in range(n_cities):
        count = 0
        # Count days starting in the city
        for i in range(n_days):
            count += If(start_city[i] == city, 1, 0)
        # Count travel days arriving at the city
        for i in range(n_days-1):
            count += If(And(travel[i], start_city[i+1] == city), 1, 0)
        solver.add(count == req_days[city])
    
    # Event constraints
    # Frankfurt show: days 12 to 16 (0-indexed: 11 to 15) must have start_city = Frankfurt
    for d in range(11, 16):
        solver.add(start_city[d] == 3)
    
    # Venice wedding: must be in Venice between day 22 and 26 (0-indexed: 21 to 25)
    solver.add(Or([start_city[d] == 1 for d in range(21, 26)]))
    
    # Tallinn meeting: must be in Tallinn between day 8 and 12 (0-indexed: 7 to 11)
    solver.add(Or([start_city[d] == 6 for d in range(7, 12)]))
    
    # Check and get model
    if solver.check() == sat:
        model = solver.model()
        start_city_val = [model.evaluate(start_city[i]).as_long() for i in range(n_days)]
        
        # Group into blocks
        blocks = []
        current_city = start_city_val[0]
        start_day = 1
        for i in range(1, n_days):
            if start_city_val[i] != current_city:
                end_day = i
                blocks.append({
                    "day_range": f"Day {start_day}-{end_day}",
                    "place": cities[current_city]
                })
                start_day = end_day
                current_city = start_city_val[i]
        blocks.append({
            "day_range": f"Day {start_day}-{n_days}",
            "place": cities[current_city]
        })
        
        # Output as JSON
        result = {"itinerary": blocks}
        print(json.dumps(result, indent=2))
    else:
        print('{"itinerary": []}')

if __name__ == "__main__":
    main()
