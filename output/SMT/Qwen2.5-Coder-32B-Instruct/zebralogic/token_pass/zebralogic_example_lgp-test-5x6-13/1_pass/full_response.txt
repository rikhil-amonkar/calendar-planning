from z3 import *

# Define the domain size
num_houses = 5

# Create variables
names = [Int(f'name_{i}') for i in range(num_houses)]
foods = [Int(f'food_{i}') for i in range(num_houses)]
car_models = [Int(f'car_model_{i}') for i in range(num_houses)]
phone_models = [Int(f'phone_model_{i}') for i in range(num_houses)]
occupations = [Int(f'occupation_{i}') for i in range(num_houses)]
drinks = [Int(f'drink_{i}') for i in range(num_houses)]

# Define mappings
name_map = {'Eric': 0, 'Peter': 1, 'Arnold': 2, 'Alice': 3, 'Bob': 4}
food_map = {'stir fry': 0, 'spaghetti': 1, 'stew': 2, 'grilled cheese': 3, 'pizza': 4}
car_model_map = {'ford f150': 0, 'tesla model 3': 1, 'bmw 3 series': 2, 'toyota camry': 3, 'honda civic': 4}
phone_model_map = {'iphone 13': 0, 'google pixel 6': 1, 'samsung galaxy s21': 2, 'oneplus 9': 3, 'huawei p50': 4}
occupation_map = {'teacher': 0, 'lawyer': 1, 'doctor': 2, 'artist': 3, 'engineer': 4}
drink_map = {'tea': 0, 'milk': 1, 'water': 2, 'root beer': 3, 'coffee': 4}

# Reverse mappings for output
reverse_name_map = {v: k for k, v in name_map.items()}
reverse_food_map = {v: k for k, v in food_map.items()}
reverse_car_model_map = {v: k for k, v in car_model_map.items()}
reverse_phone_model_map = {v: k for k, v in phone_model_map.items()}
reverse_occupation_map = {v: k for k, v in occupation_map.items()}
reverse_drink_map = {v: k for k, v in drink_map.items()}

# Create solver
solver = Solver()

# Add constraints for uniqueness
solver.add(Distinct(names))
solver.add(Distinct(foods))
solver.add(Distinct(car_models))
solver.add(Distinct(phone_models))
solver.add(Distinct(occupations))
solver.add(Distinct(drinks))

# Clue 1: The root beer lover is the person who owns a Honda Civic.
solver.add(drinks[car_model_map['honda civic']] == drink_map['root beer'])

# Clue 2: The person who likes milk is directly left of the person who loves eating grilled cheese.
for i in range(num_houses - 1):
    solver.add(Implies(drinks[i] == drink_map['milk'], foods[i + 1] == food_map['grilled cheese']))

# Clue 3: Alice is the person who uses a Samsung Galaxy S21.
solver.add(phone_models[name_map['Alice']] == phone_model_map['samsung galaxy s21'])

# Clue 4: Alice is the person who loves stir fry.
solver.add(foods[name_map['Alice']] == food_map['stir fry'])

# Clue 5: The tea drinker is not in the fifth house.
solver.add(drinks[num_houses - 1] != drink_map['tea'])

# Clue 6: The person who owns a BMW 3 Series is somewhere to the left of the tea drinker.
for i in range(num_houses - 1):
    solver.add(Implies(car_models[i] == car_model_map['bmw 3 series'], drinks[i + 1] == drink_map['tea']))

# Clue 7: The person who is a doctor is Arnold.
solver.add(occupations[name_map['Arnold']] == occupation_map['doctor'])

# Clue 8: The person who uses an iPhone 13 is the coffee drinker.
solver.add(phone_models[drink_map['coffee']] == phone_model_map['iphone 13'])

# Clue 9: The person who is an engineer is the person who owns a BMW 3 Series.
solver.add(occupations[car_model_map['bmw 3 series']] == occupation_map['engineer'])

# Clue 10: The person who loves the stew is the person who uses an iPhone 13.
solver.add(foods[phone_model_map['iphone 13']] == food_map['stew'])

# Clue 11: The person who is a doctor is directly left of the person who uses a OnePlus 9.
for i in range(num_houses - 1):
    solver.add(Implies(occupations[i] == occupation_map['doctor'], phone_models[i + 1] == phone_model_map['oneplus 9']))

# Clue 12: The person who owns a Honda Civic is directly left of the person who loves the spaghetti eater.
for i in range(num_houses - 1):
    solver.add(Implies(car_models[i] == car_model_map['honda civic'], foods[i + 1] == food_map['spaghetti']))

# Clue 13: The person who uses a Google Pixel 6 is the tea drinker.
solver.add(phone_models[drink_map['tea']] == phone_model_map['google pixel 6'])

# Clue 14: Alice is the person who is an artist.
solver.add(occupations[name_map['Alice']] == occupation_map['artist'])

# Clue 15: There is one house between Alice and the person who owns a Ford F-150.
for i in range(num_houses):
    if i > 0 and i < num_houses - 1:
        solver.add(Or(And(names[i - 1] == name_map['Alice'], car_models[i + 1] == car_model_map['ford f150']),
                      And(names[i + 1] == name_map['Alice'], car_models[i - 1] == car_model_map['ford f150'])))

    for i in range(num_houses):

          '"rows": ' + str(solution) +

    print("No solution found")
