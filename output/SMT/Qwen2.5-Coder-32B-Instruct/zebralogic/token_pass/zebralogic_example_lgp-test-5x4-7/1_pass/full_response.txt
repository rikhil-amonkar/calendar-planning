```python
from z3 import *

# Define the domain of possible values
houses = [1, 2, 3, 4, 5]
names = ["Alice", "Peter", "Bob", "Eric", "Arnold"]
smoothies = ["lime", "dragonfruit", "desert", "watermelon", "cherry"]
animals = ["horse", "dog", "bird", "fish", "cat"]
nationalities = ["german", "swede", "norwegian", "brit", "dane"]

# Create dictionaries to map variables to their respective domains
name_vars = {house: Int(f"name_{house}") for house in houses}
smoothie_vars = {house: Int(f"smoothie_{house}") for house in houses}
animal_vars = {house: Int(f"animal_{house}") for house in houses}
nationality_vars = {house: Int(f"nationality_{house}") for house in houses}

# Create a solver instance
solver = Solver()

# Add constraints for uniqueness within each category
for var_dict, domain in [(name_vars, names), (smoothie_vars, smoothies), (animal_vars, animals), (nationality_vars, nationalities)]:
    solver.add(Distinct(var_dict.values()))
    for house, var in var_dict.items():
        solver.add(Or([var == i for i in range(len(domain))]))

# Map indices to actual values for easier interpretation
def index_to_value(index, domain):
    return domain[index]

# Encode clues into constraints
# Clue 1: The Swedish person is directly left of the dog owner.
solver.add(ForAll([Int('h')], Implies(nationality_vars[Int('h')] == nationalities.index("swede"), animal_vars[Int('h') + 1] == animals.index("dog")))

# Clue 2: There are two houses between the dog owner and the British person.
solver.add(Or([And(animal_vars[h] == animals.index("dog"), nationality_vars[h+3] == nationalities.index("brit")) for h in range(1, 3)]))

# Clue 3: The Dane is the person who keeps horses.
solver.add(nationality_vars[3] == nationalities.index("dane"))
solver.add(animal_vars[3] == animals.index("horse"))

# Clue 4: The bird keeper is somewhere to the right of the cat lover.
solver.add(Or([And(animal_vars[h1] == animals.index("cat"), animal_vars[h2] == animals.index("bird")) for h1 in range(5) for h2 in range(h1+1, 5)]))

# Clue 5: The dog owner is directly left of the person who drinks Lime smoothies.
solver.add(ForAll([Int('h')], Implies(animal_vars[Int('h')] == animals.index("dog"), smoothie_vars[Int('h') + 1] == smoothies.index("lime"))))

# Clue 6: Eric is the cat lover.
solver.add(name_vars[h] == names.index("Eric") for h in range(5) if animal_vars[h] == animals.index("cat"))

# Clue 7: Bob is the bird keeper.
solver.add(name_vars[h] == names.index("Bob") for h in range(5) if animal_vars[h] == animals.index("bird"))

# Clue 8: The person who likes Cherry smoothies is directly left of Peter.
solver.add(ForAll([Int('h')], Implies(smoothie_vars[Int('h')] == smoothies.index("cherry"), name_vars[Int('h') + 1] == names.index("Peter"))))

# Clue 9: The bird keeper is the Watermelon smoothie lover.
solver.add(smoothie_vars[h] == smoothies.index("watermelon") for h in range(5) if animal_vars[h] == animals.index("bird"))

# Clue 10: The Desert smoothie lover is the dog owner.
solver.add(smoothie_vars[h] == smoothies.index("desert") for h in range(5) if animal_vars[h] == animals.index("dog"))

# Clue 11: The person who keeps horses is in the third house.
solver.add(animal_vars[3] == animals.index("horse"))

# Clue 12: The Norwegian is Alice.
solver.add(nationality_vars[h] == nationalities.index("norwegian") for h in range(5) if name_vars[h] == names.index("Alice"))

# Solve the problem
if solver.check() == sat:
    model = solver.model()
    solution = []
    for house in houses:
        name_val = index_to_value(model[name_vars[house]].as_long(), names)
        smoothie_val = index_to_value(model[smoothie_vars[house]].as_long(), smoothies)
        animal_val = index_to_value(model[animal_vars[house]].as_long(), animals)
        nationality_val = index_to_value(model[nationality_vars[house]].as_long(), nationalities)
        solution.append([str(house), name_val, smoothie_val, animal_val, nationality_val])
    
    print({
        "solution": {
            "header": ["House", "Name", "Smoothie", "Animal", "Nationality"],
            "rows": solution
        }
    })
else:
    print("No solution found")
```
