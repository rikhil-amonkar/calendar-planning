from z3 import *

# Define the domain for each attribute
names = ["Alice", "Peter", "Eric", "Bob", "Arnold", "Carol"]
cigars = ["pall mall", "yellow monster", "dunhill", "blue master", "prince", "blends"]
music_genres = ["hip hop", "jazz", "country", "pop", "classical", "rock"]
drinks = ["water", "milk", "boba tea", "tea", "root beer", "coffee"]
mothers = ["Kailyn", "Penny", "Janelle", "Holly", "Sarah", "Aniya"]
foods = ["soup", "pizza", "spaghetti", "stir fry", "stew", "grilled cheese"]

# Create variables for each house
houses = range(1, 7)
name_vars = [Int(f"name_{i}") for i in houses]
cigar_vars = [Int(f"cigar_{i}") for i in houses]
music_genre_vars = [Int(f"music_genre_{i}") for i in houses]
drink_vars = [Int(f"drink_{i}") for i in houses]
mother_vars = [Int(f"mother_{i}") for i in houses]
food_vars = [Int(f"food_{i}") for i in houses]

# Create a solver instance
solver = Solver()

# Add constraints for uniqueness within each attribute
for vars in [name_vars, cigar_vars, music_genre_vars, drink_vars, mother_vars, food_vars]:
    solver.add(Distinct(vars))

# Add constraints based on the clues
# Clue 1: Carol is directly left of the person who loves eating grilled cheese.
solver.add(name_vars[0] == names.index("Carol"))
solver.add(food_vars[1] == foods.index("grilled cheese"))

solver.add(food_vars[i] == foods.index("grilled cheese") > music_genre_vars[j] == music_genres.index("rock") for i in range(1, 6) for j in range(i))

# Clue 5: Eric is directly left of Carol.
solver.add(name_vars[0] == names.index("Eric"))
solver.add(name_vars[1] == names.index("Carol"))

# Clue 6: The person who loves pop music is not in the third house.
solver.add(music_genre_vars[2] != music_genres.index("pop"))

# Clue 7: Eric is the person who loves country music.
solver.add(name_vars[0] == names.index("Eric"))
solver.add(music_genre_vars[0] == music_genres.index("country"))

solver.add(music_genre_vars[5] == music_genres.index("classical"))

solver.add(drink_vars[i] == drinks.index("coffee") == name_vars[i] == names.index("Bob") for i in houses)

solver.add(food_vars[4] != foods.index("stew"))

solver.add(Or(
    And(mother_vars[i] == mothers.index("Sarah"), cigar_vars[i+3] == cigars.index("yellow monster")) for i in range(3)
))

# Clue 14: Eric is the tea drinker.
solver.add(name_vars[0] == names.index("Eric"))

# Clue 16: The person who loves the soup is Bob.
solver.add(food_vars[i] == foods.index("soup") == name_vars[i] == names.index("Bob") for i in houses)

solver.add(music_genre_vars[i] == music_genres.index("hip hop") == mother_vars[i+1] == mothers.index("Kailyn") for i in range(5))

# Clue 18: Arnold is somewhere to the right of The person whose mother's name is Kailyn.
solver.add(name_vars[i] == names.index("Arnold") > mother_vars[j] == mothers.index("Kailyn") for i in range(1, 6) for j in range(i))
