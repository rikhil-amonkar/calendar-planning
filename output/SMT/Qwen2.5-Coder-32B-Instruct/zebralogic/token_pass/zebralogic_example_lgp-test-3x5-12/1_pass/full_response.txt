```python
from z3 import *

# Define the variables
names = ['Eric', 'Peter', 'Arnold']
cigars = ['blue master', 'prince', 'pall mall']
hobbies = ['photography', 'gardening', 'cooking']
educations = ['high school', 'associate', 'bachelor']
drinks = ['tea', 'milk', 'water']

# Create symbolic variables for each attribute in each house
house_names = [Int(f'house{i}_name') for i in range(1, 4)]
house_cigars = [Int(f'house{i}_cigar') for i in range(1, 4)]
house_hobbies = [Int(f'house{i}_hobby') for i in range(1, 4)]
house_educations = [Int(f'house{i}_education') for i in range(1, 4)]
house_drinks = [Int(f'house{i}_drink') for i in range(1, 4)]

# Create a solver instance
solver = Solver()

# Add constraints for unique values within each attribute category
solver.add(Distinct(house_names))
solver.add(Distinct(house_cigars))
solver.add(Distinct(house_hobbies))
solver.add(Distinct(house_educations))
solver.add(Distinct(house_drinks))

# Map each symbolic variable to its corresponding attribute value
name_map = {i: j for i, j in enumerate(names)}
cigar_map = {i: j for i, j in enumerate(cigars)}
hobby_map = {i: j for i, j in enumerate(hobbies)}
education_map = {i: j for i, j in enumerate(educations)}
drink_map = {i: j for i, j in enumerate(drinks)}

# Add constraints based on the clues
# Clue 1: The person partial to Pall Mall is Peter.
solver.add(house_cigars[house_names.index(1)] == cigars.index('pall mall'))

# Clue 2: The person who likes milk is directly left of the person with a high school diploma.
solver.add(house_drinks.index(drinks.index('milk')) + 1 == house_educations.index(educations.index('high school')))

# Clue 3: Eric is the tea drinker.
solver.add(house_names[house_drinks.index(drinks.index('tea'))] == names.index('Eric'))

# Clue 4: Arnold and the Prince smoker are next to each other.
solver.add(Or(
    Abs(house_names.index(names.index('Arnold')) - house_cigars.index(cigars.index('prince'))) == 1,
    Abs(house_cigars.index(cigars.index('prince')) - house_names.index(names.index('Arnold'))) == 1
))

# Clue 5: The person who enjoys gardening is somewhere to the left of the Prince smoker.
solver.add(house_hobbies.index(hobbies.index('gardening')) < house_cigars.index(cigars.index('prince')))

# Clue 6: The person who likes milk is the person with an associate's degree.
solver.add(house_drinks.index(drinks.index('milk')) == house_educations.index(educations.index('associate')))

# Clue 7: The person with a bachelor's degree is directly left of the photography enthusiast.
solver.add(house_educations.index(educations.index('bachelor')) + 1 == house_hobbies.index(hobbies.index('photography')))

# Check if the constraints are satisfiable
if solver.check() == sat:
    model = solver.model()
    solution = {
        "solution": {
            "header": ["House", "Name", "Cigar", "Hobby", "Education", "Drink"],
            "rows": []
        }
    }
    for i in range(1, 4):
        name_val = name_map[model[house_names[i-1]].as_long()]
        cigar_val = cigar_map[model[house_cigars[i-1]].as_long()]
        hobby_val = hobby_map[model[house_hobbies[i-1]].as_long()]
        education_val = education_map[model[house_educations[i-1]].as_long()]
        drink_val = drink_map[model[house_drinks[i-1]].as_long()]
        solution["solution"]["rows"].append([str(i), name_val, cigar_val, hobby_val, education_val, drink_val])
    print(solution)
else:
    print("No solution found")
```
