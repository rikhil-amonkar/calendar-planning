from z3 import *

# Define the domains
names = ['Eric', 'Arnold', 'Peter', 'Alice']
hair_colors = ['blonde', 'black', 'brown', 'red']
music_genres = ['pop', 'jazz', 'rock', 'classical']

# Create variables for each house
houses = [1, 2, 3, 4]
name_vars = {house: Int(f'name_{house}') for house in houses}
hair_color_vars = {house: Int(f'hair_color_{house}') for house in houses}
music_genre_vars = {house: Int(f'music_genre_{house}') for house in houses}

# Create solvers
solver = Solver()

# Add constraints for unique values within each category
for var_dict, domain in [(name_vars, names), (hair_color_vars, hair_colors), (music_genre_vars, music_genres)]:
    # Ensure each value appears exactly once
    for val in range(len(domain)):
        solver.add(Or([var_dict[house] == val for house in houses]))
        for i in range(len(houses)):
            for j in range(i + 1, len(houses)):
                solver.add(var_dict[houses[i]] != var_dict[houses[j]])

# Encode the clues
# 1. Eric is the person who has red hair.
solver.add(name_vars[1] == names.index('Eric') == hair_color_vars[1] == hair_colors.index('red'))

# 2. The person who loves classical music is directly left of the person who has blonde hair.
solver.add(music_genre_vars[1] == music_genres.index('classical') == hair_color_vars[2] == hair_colors.index('blonde'))

# 3. The person who has brown hair is not in the first house.
solver.add(hair_color_vars[1] != hair_colors.index('brown'))

# 4. The person who loves pop music is not in the third house.
solver.add(music_genre_vars[3] != music_genres.index('pop'))

# 5. The person who loves classical music is in the first house.
solver.add(music_genre_vars[1] == music_genres.index('classical'))

# 6. The person who loves jazz music is the person who has red hair.
solver.add(music_genre_vars[1] == music_genres.index('jazz') == hair_color_vars[1] == hair_colors.index('red'))

# 7. The person who loves rock music is Arnold.
solver.add(music_genre_vars[2] == music_genres.index('rock') == name_vars[2] == names.index('Arnold'))

# 8. Peter is somewhere to the right of the person who loves rock music.
solver.add(name_vars[3] == names.index('Peter') | name_vars[4] == names.index('Peter'))

# Solve the problem
if solver.check() == sat:
    model = solver.model()
    
    # Map the integer values back to the original strings
    solution = {
        "solution": {
            "header": ["House", "Name", "HairColor", "MusicGenre"],
            "rows": []
        }
    }
    
    for house in houses:
        name = names[model.evaluate(name_vars[house]).as_long()]
        hair_color = hair_colors[model.evaluate(hair_color_vars[house]).as_long()]
        music_genre = music_genres[model.evaluate(music_genre_vars[house]).as_long()]
        solution["solution"]["rows"].append([str(house), name, hair_color, music_genre])
    
    print(solution)
else:
    print("No solution found")
