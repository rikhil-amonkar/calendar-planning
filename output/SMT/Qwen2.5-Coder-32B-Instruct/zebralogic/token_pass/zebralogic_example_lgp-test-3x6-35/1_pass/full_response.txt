from z3 import *

# Define Enumerations for each characteristic
Name = EnumSort('Name', ['Eric', 'Arnold', 'Peter'])
Vacation = EnumSort('Vacation', ['mountain', 'city', 'beach'])
Height = EnumSort('Height', ['very_short', 'average', 'short'])
Flower = EnumSort('Flower', ['carnations', 'daffodils', 'lilies'])
HairColor = EnumSort('HairColor', ['brown', 'black', 'blonde'])
Education = EnumSort('Education', ['associate', 'bachelor', 'high_school'])

# Unpack the sorts and their members
Name, (Eric, Arnold, Peter) = Name
Vacation, (mountain, city, beach) = Vacation
Height, (very_short, average, short) = Height
Flower, (carnations, daffodils, lilies) = Flower
HairColor, (brown, black, blonde) = HairColor
Education, (associate, bachelor, high_school) = Education

# Create variables for each characteristic in each house
house1_name, house2_name, house3_name = Consts('house1_name house2_name house3_name', Name)
house1_vacation, house2_vacation, house3_vacation = Consts('house1_vacation house2_vacation house3_vacation', Vacation)
house1_height, house2_height, house3_height = Consts('house1_height house2_height house3_height', Height)

house1_education, house2_education, house3_education = Consts('house1_education house2_education house3_education', Education)

# Create a solver instance
solver = Solver()

# Add constraints based on the clues
solver.add(Distinct(house1_name, house2_name, house3_name))
solver.add(Distinct(house1_vacation, house2_vacation, house3_vacation))
solver.add(Distinct(house1_height, house2_height, house3_height))
solver.add(Distinct(house1_flower, house2_flower, house3_flower))

solver.add(house3_education == high_school)

# Clue 6: The person who is short is somewhere to the right of the person who is very short.
solver.add(Or(house2_height == short, house3_height == short))
solver.add(Or(house1_height == very_short, house2_height == very_short))

# Clue 7: The person who loves the bouquet of lilies is Eric.
solver.add(house1_flower != lilies, house2_flower != lilies, house3_flower == lilies)
solver.add(house1_name != Eric, house2_name != Eric, house3_name == Eric)

# Clue 8: The person who loves the bouquet of lilies is the person with a bachelor's degree.
solver.add(house1_education != bachelor, house2_education != bachelor, house3_education == bachelor)

# Clue 9: The person who prefers city breaks is somewhere to the right of Peter.
solver.add(Or(house2_vacation == city, house3_vacation == city))
solver.add(Or(house1_name == Peter, house2_name == Peter))

# Clue 10: The person who has blonde hair is in the third house.
solver.add(house3_hair_color == blonde)

# Clue 11: The person who loves beach vacations is the person who has brown hair.
