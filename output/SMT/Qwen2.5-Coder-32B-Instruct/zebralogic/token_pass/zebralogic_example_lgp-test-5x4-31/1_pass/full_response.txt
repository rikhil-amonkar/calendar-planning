from z3 import *

# Create solver instance
solver = Solver()

# Define variables for each house
houses = [1, 2, 3, 4, 5]
names = ['Alice', 'Bob', 'Arnold', 'Eric', 'Peter']
vacations = ['cruise', 'city', 'camping', 'beach', 'mountain']
children = ['Bella', 'Samantha', 'Fred', 'Meredith', 'Timothy']
nationalities = ['dane', 'norwegian', 'brit', 'german', 'swede']

# Create dictionaries for each attribute
name_vars = {house: Int(f'name_{house}') for house in houses}
vacation_vars = {house: Int(f'vacation_{house}') for house in houses}
child_vars = {house: Int(f'child_{house}') for house in houses}
nationality_vars = {house: Int(f'nationality_{house}') for house in houses}

# Map string values to integers for constraints
name_map = {name: i for i, name in enumerate(names)}
vacation_map = {vacation: i for i, vacation in enumerate(vacations)}
child_map = {child: i for i, child in enumerate(children)}
nationality_map = {nationality: i for i, nationality in enumerate(nationalities)}

# Add constraints for unique values in each category
for house in houses:
    solver.add(name_vars[house] >= 0, name_vars[house] < len(names))
    solver.add(vacation_vars[house] >= 0, vacation_vars[house] < len(vacations))
    solver.add(child_vars[house] >= 0, child_vars[house] < len(children))
    solver.add(nationality_vars[house] >= 0, nationality_vars[house] < len(nationalities))

solver.add(Distinct([name_vars[house] for house in houses]))
solver.add(Distinct([vacation_vars[house] for house in houses]))
solver.add(Distinct([child_vars[house] for house in houses]))
solver.add(Distinct([nationality_vars[house] for house in houses]))

# Add specific clues as constraints
# Clue 1: The Norwegian is Peter.
solver.add(nationality_vars[houses[1]] == nationality_map['norwegian'])
solver.add(name_vars[houses[1]] == name_map['Peter'])

# Clue 2: The Swedish person is the person's child is named Bella.
solver.add(And([Implies(nationality_vars[house] == nationality_map['swede'], child_vars[house] == child_map['Bella']) for house in houses]))

# Clue 3: The person who loves beach vacations is directly left of the person's child is named Samantha.
solver.add(Or([And(vacation_vars[house] == vacation_map['beach'], child_vars[house + 1] == child_map['Samantha']) for house in range(1, 5)]))

# Clue 4: The person's child is named Bella is not in the second house.
solver.add(child_vars[houses[1]] != child_map['Bella'])

# Clue 5: Alice is the British person.
solver.add(name_vars[houses[2]] == name_map['Alice'])
solver.add(nationality_vars[houses[2]] == nationality_map['brit'])

# Clue 6: The person who likes going on cruises is in the first house.
solver.add(vacation_vars[houses[0]] == vacation_map['cruise'])

# Clue 7: The person's child is named Meredith is in the fourth house.
solver.add(child_vars[houses[3]] == child_map['Meredith'])

# Clue 8: Eric is not in the fifth house.
solver.add(name_vars[houses[4]] != name_map['Eric'])

# Clue 9: The Swedish person is somewhere to the right of the Norwegian.
solver.add(Or([nationality_vars[house] == nationality_map['swede'] for house in range(2, 6)]))

# Clue 10: There is one house between the person's child is named Fred and the person who prefers city breaks.
solver.add(Or(
    And(child_vars[houses[0]] == child_map['Fred'], vacation_vars[houses[2]] == vacation_map['city']),
    And(child_vars[houses[1]] == child_map['Fred'], vacation_vars[houses[3]] == vacation_map['city']),
    And(child_vars[houses[2]] == child_map['Fred'], vacation_vars[houses[4]] == vacation_map['city']),
    And(child_vars[houses[3]] == child_map['Fred'], vacation_vars[houses[1]] == vacation_map['city']),
    And(child_vars[houses[4]] == child_map['Fred'], vacation_vars[houses[2]] == vacation_map['city'])
))

# Clue 11: Bob is the person who enjoys camping trips.
solver.add(name_vars[houses[2]] == name_map['Bob'])
solver.add(vacation_vars[houses[2]] == vacation_map['camping'])

# Clue 12: The Dane is in the fifth house.
solver.add(nationality_vars[houses[4]] == nationality_map['dane'])

# Clue 13: The person who enjoys camping trips is not in the fifth house.
solver.add(vacation_vars[houses[4]] != vacation_map['camping'])

# Solve the problem
if solver.check() == sat:
    model = solver.model()
    
    # Prepare the result in JSON format
    result = {
        "solution": {
            "header": ["House", "Name", "Vacation", "Children", "Nationality"],
            "rows": []
        }
    }
    
    for house in houses:
        name_val = names[model.evaluate(name_vars[house]).as_long()]
        vacation_val = vacations[model.evaluate(vacation_vars[house]).as_long()]
        child_val = children[model.evaluate(child_vars[house]).as_long()]
        nationality_val = nationalities[model.evaluate(nationality_vars[house]).as_long()]
        result["solution"]["rows"].append([str(house), name_val, vacation_val, child_val, nationality_val])
    
    print(result)
else:
    print("No solution found")
