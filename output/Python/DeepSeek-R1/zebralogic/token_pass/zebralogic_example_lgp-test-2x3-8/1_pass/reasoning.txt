   import itertools

   names = ['Arnold', 'Eric']
   birthdays = ['april', 'sept']
   mothers = ['Aniya', 'Holly']

   # We'll generate all permutations for each attribute
   name_perms = list(itertools.permutations(names))
   birthday_perms = list(itertools.permutations(birthdays))
   mother_perms = list(itertools.permutations(mothers))

   # Now iterate over every combination of these permutations
   for name_perm in name_perms:
      for birthday_perm in birthday_perms:
          for mother_perm in mother_perms:
             # Assign to houses: house1 gets the first of each permutation, house2 gets the second.
             house1 = {'Name': name_perm[0], 'Birthday': birthday_perm[0], 'Mother': mother_perm[0]}
             house2 = {'Name': name_perm[1], 'Birthday': birthday_perm[1], 'Mother': mother_perm[1]}

             # Check clue2: house1 must have birthday 'april'
             if house1['Birthday'] != 'april':
                 continue

             # Check clue1: find Eric's house and the house with mother Holly
             eric_house = None
             holly_house = None
             if house1['Name'] == 'Eric':
                 eric_house = 1
             elif house2['Name'] == 'Eric':
                 eric_house = 2

             if house1['Mother'] == 'Holly':
                 holly_house = 1
             elif house2['Mother'] == 'Holly':
                 holly_house = 2

             # If Eric is not found or Holly is not found, skip? Actually, every attribute must be assigned, so they must be found.
             if eric_house is None or holly_house is None:
