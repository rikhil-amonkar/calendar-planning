        def apply_same_house(attr1, value1, attr2, value2):

                    if value2 not in possibilities[house][attr2]:

                if value2 in possibilities[house][attr2]:

        def apply_right_of(attr1, value1, attr2, value2):

            # attr1 is right of attr2
