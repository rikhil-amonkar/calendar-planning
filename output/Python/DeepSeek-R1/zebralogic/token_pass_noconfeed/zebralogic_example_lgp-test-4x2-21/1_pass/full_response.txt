import itertools
import json

def main():
    # Define the fixed attributes for house 2
    house2_name = "Alice"
    house2_style = "craftsman"
    
    # Remaining names and styles
    remaining_names = ["Eric", "Arnold", "Peter"]
    remaining_styles = ["colonial", "ranch", "victorian"]
    
    # Generate all permutations for the remaining houses (1, 3, 4)
    name_perms = list(itertools.permutations(remaining_names))
    style_perms = list(itertools.permutations(remaining_styles))
    
    # Iterate over all combinations of name and style permutations
    for name_perm in name_perms:
        for style_perm in style_perms:
            # Initialize assignment for all houses (index 0=house1, 1=house2, 2=house3, 3=house4)
            names = [None] * 4
            styles = [None] * 4
            
            # Assign house 2
            names[1] = house2_name
            styles[1] = house2_style
            
            # Assign remaining houses (1, 3, 4)
            names[0] = name_perm[0]  # house1 name
            names[2] = name_perm[1]  # house3 name
            names[3] = name_perm[2]  # house4 name
            
            styles[0] = style_perm[0]  # house1 style
            styles[2] = style_perm[1]  # house3 style
            styles[3] = style_perm[2]  # house4 style
            
            # Check constraints
            # Constraint 2: Victorian directly left of Peter
            victorian_index = None
            peter_index = None
            for i in range(4):
                if styles[i] == "victorian":
                    victorian_index = i
                if names[i] == "Peter":
                    peter_index = i
            if victorian_index is None or peter_index != victorian_index + 1:
                continue
            
            # Constraint 3: Peter right of ranch
            ranch_index = None
            for i in range(4):
                if styles[i] == "ranch":
                    ranch_index = i
            if ranch_index is None or peter_index <= ranch_index:
                continue
            
            # Constraint 4: Arnold right of craftsman (house2)
            arnold_index = None
            for i in range(4):
                if names[i] == "Arnold":
                    arnold_index = i
            if arnold_index is None or arnold_index <= 1:
                continue
            
            # All constraints satisfied, build solution
            solution = {
                "header": ["House", "Name", "HouseStyle"],
                "rows": [
                    ["1", names[0], styles[0]],
                    ["2", names[1], styles[1]],
                    ["3", names[2], styles[2]],
                    ["4", names[3], styles[3]]
                ]
            }
            
            # Output JSON
            print(json.dumps({"solution": solution}))
            return
    
    # If no solution found (should not happen)
    print(json.dumps({"solution": None}))

if __name__ == "__main__":
    main()
