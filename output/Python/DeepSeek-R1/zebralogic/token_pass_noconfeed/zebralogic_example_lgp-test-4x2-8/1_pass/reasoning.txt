 import itertools
 import json

 names = ['Peter', 'Arnold', 'Alice', 'Eric']
 colors = ['yellow', 'green', 'red', 'white']

 # Fixed assignments
 fixed_name_house1 = 'Peter'
 fixed_color_house3 = 'green'

 # The remaining names and colors
 remaining_names = [n for n in names if n != fixed_name_house1]
 remaining_colors = [c for c in colors if c != fixed_color_house3]

 # Generate permutations for the remaining names (for houses 2,3,4) and remaining colors (for houses 1,2,4)
 for name_perm in itertools.permutations(remaining_names):
    full_names = [fixed_name_house1] + list(name_perm)   # for house1,2,3,4

    for color_perm in itertools.permutations(remaining_colors):
        full_colors = [None, None, fixed_color_house3, None]
        # Assign color_perm[0] to house1 (index0), color_perm[1] to house2 (index1), color_perm[2] to house4 (index3)
        full_colors[0] = color_perm[0]
        full_colors[1] = color_perm[1]
        full_colors[3] = color_perm[2]

        # Check constraint 3: |red index - yellow index| = 2 (in 1-indexed, which is the same as 0-indexed indices difference of 2)
        # Find the 0-indexed indices of red and yellow in full_colors
        try:
            red_index = full_colors.index('red')
            yellow_index = full_colors.index('yellow')
        except ValueError:
            continue

        if abs(red_index - yellow_index) != 2:
            continue

        # Check constraint 4: Arnold is directly left of Eric (so Arnold's index + 1 = Eric's index)
        try:
            arnold_index = full_names.index('Arnold')
            eric_index = full_names.index('Eric')
        except ValueError:
            continue

        if arnold_index + 1 != eric_index:
            continue

        # Check constraint 5: Eric has color yellow -> the color at eric_index must be yellow
        if full_colors[eric_index] != 'yellow':
            continue

        # If we passed all, we found the solution.
        solution = {
            "solution": {
                "header": ["House", "Name", "Color"],
                "rows": []
            }
        }
        for i in range(4):
            solution["solution"]["rows"].append([str(i+1), full_names[i], full_colors[i]])

        # Output the solution as JSON
        print(json.dumps(solution, indent=2))
        exit(0)
