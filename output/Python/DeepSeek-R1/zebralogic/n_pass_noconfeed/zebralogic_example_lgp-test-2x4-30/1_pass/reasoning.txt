 Constraints as functions:

 def check_constraints(house1, house2):
   # Constraint 1: The Desert smoothie lover is Arnold.
   # Check: if a house has smoothie 'desert', then name must be 'Arnold'
   for house in [house1, house2]:
        if house[4] == 'desert' and house[1] != 'Arnold':
            return False
        # Also, if a house has name 'Arnold', then smoothie must be 'desert'? Not necessarily, but clue says the desert smoothie lover is Arnold, meaning that Arnold is the one who loves desert. So if Arnold is in a house, that house must have desert smoothie.
        if house[1] == 'Arnold' and house[4] != 'desert':
            return False

   # Constraint 2: The person who has brown hair is the person who loves basketball.
   for house in [house1, house2]:
        if house[2] == 'brown' and house[3] != 'basketball':
            return False
        if house[3] == 'basketball' and house[2] != 'brown':
            return False

   # Constraint 3: Arnold is somewhere to the left of the person who has black hair.
   # Find Arnold's house number and black hair's house number.
   arnold_house = None
   black_hair_house = None
   for house in [house1, house2]:
        if house[1] == 'Arnold':
            arnold_house = house[0]
        if house[2] == 'black':
            black_hair_house = house[0]

   if arnold_house is None or black_hair_house is None:
        return False
   if arnold_house >= black_hair_house:
        return False

   return True
