import json

def time_to_minutes(time_str):
    hours, minutes = map(int, time_str.split(':'))
    return hours * 60 + minutes

def minutes_to_time(minutes):
    hours = minutes // 60
    mins = minutes % 60
    return f"{hours}:{mins:02d}"

def main():
    travel_times = {
        'Golden Gate Park': {
            'Fisherman\'s Wharf': 24,
            'Bayview': 23,
            'Mission District': 17,
            'Embarcadero': 25,
            'Financial District': 26
        },
        'Fisherman\'s Wharf': {
            'Golden Gate Park': 25,
            'Bayview': 26,
            'Mission District': 22,
            'Embarcadero': 8,
            'Financial District': 11
        },
        'Bayview': {
            'Golden Gate Park': 22,
            'Fisherman\'s Wharf': 25,
            'Mission District': 13,
            'Embarcadero': 19,
            'Financial District': 19
        },
        'Mission District': {
            'Golden Gate Park': 17,
            'Fisherman\'s Wharf': 22,
            'Bayview': 15,
            'Embarcadero': 19,
            'Financial District': 17
        },
        'Embarcadero': {
            'Golden Gate Park': 25,
            'Fisherman\'s Wharf': 6,
            'Bayview': 21,
            'Mission District': 20,
            'Financial District': 5
        },
        'Financial District': {
            'Golden Gate Park': 23,
            'Fisherman\'s Wharf': 10,
            'Bayview': 19,
            'Mission District': 17,
            'Embarcadero': 4
        }
    }

    friends = [
        {'name': 'Joseph', 'location': 'Fisherman\'s Wharf', 'avail_start': time_to_minutes('8:00'), 'avail_end': time_to_minutes('17:30'), 'min_duration': 90},
        {'name': 'Jeffrey', 'location': 'Bayview', 'avail_start': time_to_minutes('17:30'), 'avail_end': time_to_minutes('21:30'), 'min_duration': 60},
        {'name': 'Kevin', 'location': 'Mission District', 'avail_start': time_to_minutes('11:15'), 'avail_end': time_to_minutes('15:15'), 'min_duration': 30},
        {'name': 'David', 'location': 'Embarcadero', 'avail_start': time_to_minutes('8:15'), 'avail_end': time_to_minutes('9:00'), 'min_duration': 30},
        {'name': 'Barbara', 'location': 'Financial District', 'avail_start': time_to_minutes('10:30'), 'avail_end': time_to_minutes('16:30'), 'min_duration': 15}
    ]

    current_time = time_to_minutes('9:00')
    current_location = 'Golden Gate Park'
    itinerary = []

    # David is impossible to meet due to time constraints, so we skip
    meet_order = ['Barbara', 'Kevin', 'Joseph', 'Jeffrey']
    
    for person_name in meet_order:
        friend = next(f for f in friends if f['name'] == person_name)
        travel_time = travel_times[current_location][friend['location']]
        arrival_time = current_time + travel_time
        start_time = max(arrival_time, friend['avail_start'])
        
        if person_name == 'Joseph':
            # Leave Joseph by 17:00 to reach Jeffrey on time
            end_time = min(start_time + friend['min_duration'], time_to_minutes('17:00'))
        else:
            end_time = start_time + friend['min_duration']
        
        # Ensure meeting doesn't exceed availability
        end_time = min(end_time, friend['avail_end'])
        
        if start_time >= end_time:
            continue  # Skip if no valid meeting time
            
        itinerary.append({
            'action': 'meet',
            'location': friend['location'],
            'person': friend['name'],
            'start_time': minutes_to_time(start_time),
            'end_time': minutes_to_time(end_time)
        })
        
        current_time = end_time
        current_location = friend['location']

    result = {'itinerary': itinerary}
    print(json.dumps(result, indent=2))

if __name__ == '__main__':
    main()
