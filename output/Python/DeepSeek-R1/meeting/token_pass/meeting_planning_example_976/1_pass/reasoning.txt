        time_str = time_str.replace('AM', '').replace('PM', '')
        parts = time_str.split(':')
        hour = int(parts[0])
        minute = int(parts[1])
        if 'PM' in time_str and hour != 12:
            hour += 12
        if 'AM' in time_str and hour == 12:
            hour = 0
        return hour * 60 + minute
