import itertools

def convert_time(time_str):
    if "AM" in time_str:
        time_str = time_str.replace("AM", "")
        parts = time_str.split(":")
        hour = int(parts[0])
        if hour == 12:
            hour = 0
        minute = int(parts[1])
        return hour * 60 + minute
    else:
        time_str = time_str.replace("PM", "")
        parts = time_str.split(":")
        hour = int(parts[0])
        if hour != 12:
            hour += 12
        minute = int(parts[1])
        return hour * 60 + minute

def minutes_to_time(minutes):
    hours = minutes // 60

    return f"{hours}:{mins:02d}"
