def time_to_minutes(time_str):
    time_str = time_str.strip()
    parts = time_str.split(':')
    hour = int(parts[0])
    rest = parts[1]
    if 'AM' in rest:
        minutes = int(rest.replace('AM', ''))
        if hour == 12:
            hour = 0
        return hour * 60 + minutes
    else:
        minutes = int(rest.replace('PM', ''))
        if hour != 12:
            hour += 12
        return hour * 60 + minutes

def minutes_to_time(minutes):
    hours = minutes // 60
    mins = minutes % 60
    return f"{hours}:{mins:02d}"
