import json

def time_str_to_minutes(s):
    time, period = s[:-2], s[-2:]
    if ':' in time:
        hours, minutes = time.split(':')
    else:
        hours = time
        minutes = '00'
    hours = int(hours)
    minutes = int(minutes)
    if period == 'PM' and hours != 12:
        hours += 12
    if period == 'AM' and hours == 12:
        hours = 0
    return hours * 60 + minutes

def minutes_to_time_str(m):
    hours = m // 60
    minutes = m % 60
    return f"{hours}:{minutes:02d}"
