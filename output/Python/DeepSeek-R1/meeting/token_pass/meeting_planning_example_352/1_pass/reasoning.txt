    friends = [

        {"person": "Joseph", "location": "Haight-Ashbury", "avail_start": 12*60+30, "avail_end": 19*60+45, "min_duration": 90},
        {"person": "Sandra", "location": "Chinatown", "avail_start": 7*60+15, "avail_end": 19*60+15, "min_duration": 75},
        {"person": "Nancy", "location": "Marina District", "avail_start": 11*60, "avail_end": 20*60+15, "min_duration": 105}
    ]

    start_time = 9*60  # 9:00 AM in minutes from midnight

    best_count = 0
    best_itinerary = None

    for order in itertools.permutations(friends):
        current_loc = 'Union Square'
        current_time = start_time
        itinerary = []
        count = 0
        for friend in order:
            # Get travel time from current_loc to friend's location
            travel_time = travel_times[current_loc][friend['location']]
            arrival = current_time + travel_time
            # Calculate meeting start time: cannot start before friend's avail_start and must start at or after arrival
            meeting_start = max(arrival, friend['avail_start'])
            meeting_end = meeting_start + friend['min_duration']
            # Check if meeting can be held within friend's availability
            if meeting_end > friend['avail_end']:
                break  # cannot meet this friend in this order

            # Meeting is possible
            itinerary.append({
                'action': 'meet',
                'location': friend['location'],
                'person': friend['person'],
                'start_time': meeting_start,
                'end_time': meeting_end
            })
            count += 1
            current_loc = friend['location']
            current_time = meeting_end

        if count > best_count:
            best_count = count
            best_itinerary = itinerary

    # Now convert best_itinerary to the required JSON format, converting times to string
    # Function to convert minutes to time string without leading zero (e.g., 540 -> '9:00', 630 -> '10:30')
    def minutes_to_time(minutes):
        hours = minutes // 60
        minutes = minutes % 60
        return f"{hours}:{minutes:02d}"

    result = {
        "itinerary": [
            {
                "action": "meet",
                "location": item['location'],
                "person": item['person'],
                "start_time": minutes_to_time(item['start_time']),
                "end_time": minutes_to_time(item['end_time'])
            } for item in best_itinerary
        ]
    }

    print(json.dumps(result, indent=2))
