    dp = [[INF] * num_districts for _ in range(1<<n)]
    parent = [[None] * num_districts for _ in range(1<<n)]

    # Initialize starting state

    # Iterate over masks
    for mask in range(1<<n):
        for loc in range(num_districts):
            if dp[mask][loc] == INF:
                continue
            for i in range(n):
                if mask & (1<<i):
                    continue
                f = friends[i]
                to_loc = district_index[f['location']]
                t_time = travel_time[loc][to_loc]
                arrival = dp[mask][loc] + t_time
                start = max(arrival, f['start'])
                end = start + f['min_duration']
                if end <= f['end']:
                    new_mask = mask | (1<<i)
                    if end < dp[new_mask][to_loc]:
                        dp[new_mask][to_loc] = end
                        parent[new_mask][to_loc] = (mask, loc, i, start, end)

    # Find the best mask
    best_mask = 0
    best_loc = 0
    best_count = 0
    for mask in range(1<<n):
        for loc in range(num_districts):
            if dp[mask][loc] < INF:
                count = bin(mask).count('1')
                if count > best_count:
                    best_count = count
                    best_mask = mask
                    best_loc = loc
