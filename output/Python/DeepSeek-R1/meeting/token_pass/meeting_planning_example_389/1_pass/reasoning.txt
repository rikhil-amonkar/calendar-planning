    def simulate_schedule(afternoon_meetings, include_helen, travel_times, helen_meeting, start_time, start_location):
        current_time = start_time
        current_location = start_location
        schedule = []

        for meeting in afternoon_meetings:
            loc = meeting['location']
            travel_time = travel_times[current_location][loc]
            arrival = current_time + travel_time
            start_meeting = max(arrival, meeting['start_avail'])
            if start_meeting + meeting['min_duration'] > meeting['end_avail']:
                return [], 0   # cannot schedule this meeting

            end_meeting = start_meeting + meeting['min_duration']
            schedule.append({
                'action': 'meet',
                'location': loc,
                'person': meeting['name'],
                'start_time': minutes_to_time(start_meeting),
                'end_time': minutes_to_time(end_meeting)
            })
            current_time = end_meeting
            current_location = loc

        if include_helen:
            travel_time_helen = travel_times[current_location][helen_meeting['location']]
            arrival_helen = current_time + travel_time_helen
            start_helen = max(arrival_helen, helen_meeting['start_avail'])
            if start_helen + helen_meeting['min_duration'] <= helen_meeting['end_avail']:
                end_helen = start_helen + helen_meeting['min_duration']

                    'end_time': minutes_to_time(end_helen)
                })
                current_time = end_helen
                current_location = helen_meeting['location']

        return schedule, len(schedule)
