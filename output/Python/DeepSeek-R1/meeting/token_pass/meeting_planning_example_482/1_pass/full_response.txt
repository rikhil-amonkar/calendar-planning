import itertools
import json

def main():
    travel_times = {
        'Haight-Ashbury': {
            'Mission District': 11,
            'Bayview': 18,
            'Pacific Heights': 12,
            'Russian Hill': 17,
            'Fisherman\'s Wharf': 23
        },
        'Mission District': {
            'Haight-Ashbury': 12,
            'Bayview': 15,
            'Pacific Heights': 16,
            'Russian Hill': 15,
            'Fisherman\'s Wharf': 22
        },
        'Bayview': {
            'Haight-Ashbury': 19,
            'Mission District': 13,
            'Pacific Heights': 23,
            'Russian Hill': 23,
            'Fisherman\'s Wharf': 25
        },
        'Pacific Heights': {
            'Haight-Ashbury': 11,
            'Mission District': 15,
            'Bayview': 22,
            'Russian Hill': 7,
            'Fisherman\'s Wharf': 13
        },
        'Russian Hill': {
            'Haight-Ashbury': 17,
            'Mission District': 16,
            'Bayview': 23,
            'Pacific Heights': 7,
            'Fisherman\'s Wharf': 7
        },
        'Fisherman\'s Wharf': {
            'Haight-Ashbury': 22,
            'Mission District': 22,
            'Bayview': 26,
            'Pacific Heights': 12,
            'Russian Hill': 7
        }
    }

    class Friend:
        def __init__(self, name, location, start, end, min_duration):
            self.name = name
            self.location = location
            self.start = start
            self.end = end
            self.min_duration = min_duration

    def minutes_to_time(minutes):
        hours = minutes // 60
        mins = minutes % 60
        return f"{hours}:{mins:02d}"

    friends = [
        Friend('Stephanie', 'Mission District', 495, 825, 90),
        Friend('Sandra', 'Bayview', 780, 1170, 15),
        Friend('Brian', 'Russian Hill', 735, 960, 120),
        Friend('Jason', 'Fisherman\'s Wharf', 510, 1065, 60)
    ]

    start_time = 540
    start_location = 'Haight-Ashbury'
    best_itinerary = []
    max_friends = 0

    for n in range(len(friends), 0, -1):
        for perm in itertools.permutations(friends, n):
            current_time = start_time
            current_location = start_location
            itinerary = []
            valid = True
            for friend in perm:
                travel_time = travel_times[current_location][friend.location]
                arrival_time = current_time + travel_time
                meeting_start = max(arrival_time, friend.start)
                if meeting_start + friend.min_duration > friend.end:
                    valid = False
                    break
                itinerary.append({
                    'action': 'meet',
                    'location': friend.location,
                    'person': friend.name,
                    'start_time': minutes_to_time(meeting_start),
                    'end_time': minutes_to_time(meeting_start + friend.min_duration)
                })
                current_time = meeting_start + friend.min_duration
                current_location = friend.location
            if valid and len(perm) > max_friends:
                max_friends = len(perm)
                best_itinerary = itinerary
        if best_itinerary:
            break

    result = {'itinerary': best_itinerary}
    print(json.dumps(result, indent=2))

if __name__ == '__main__':
    main()
