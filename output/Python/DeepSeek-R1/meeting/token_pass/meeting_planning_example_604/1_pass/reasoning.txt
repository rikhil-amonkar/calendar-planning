 for perm in itertools.permutations(meetings):

    scheduled = []
    for meeting in perm:
        loc = meeting['location']
        travel_duration = travel_time[current_location][loc]
        arrival = current_time + travel_duration
        start_meeting = max(arrival, meeting['window_start'])
        end_meeting = start_meeting + meeting['min_duration']
        if end_meeting > meeting['window_end']:
            break  # cannot schedule this meeting
        scheduled.append({
            'action': 'meet',
            'location': loc,
            'person': meeting['person'],
            'start_time': start_meeting,
            'end_time': end_meeting
        })
        current_time = end_meeting
        current_location = loc
    else:
        # we scheduled all meetings in the permutation
        if len(scheduled) > best_count:
            best_count = len(scheduled)
            best_itinerary = scheduled
        continue

    # if we broke out, then we scheduled only part of the permutation
    if len(scheduled) > best_count:
        best_count = len(scheduled)
