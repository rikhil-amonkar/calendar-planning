     import itertools
     import json

     days_req = {
         'Brussels': 4,
         'Bucharest': 3,
         'Stuttgart': 4,
         'Mykonos': 2,
         'Madrid': 2,
         'Helsinki': 5,
         'Split': 3,
         'London': 5
     }

     graph = {
         'Helsinki': {'London', 'Madrid', 'Split', 'Brussels'},
         'London': {'Helsinki', 'Madrid', 'Brussels', 'Bucharest', 'Mykonos', 'Split'},
         'Madrid': {'Helsinki', 'London', 'Mykonos', 'Brussels', 'Bucharest', 'Split'},
         'Brussels': {'London', 'Bucharest', 'Helsinki', 'Madrid'},
         'Bucharest': {'London', 'Brussels', 'Madrid'},
         'Mykonos': {'Madrid', 'London'},
         'Stuttgart': {'London', 'Split'},
         'Split': {'Madrid', 'Helsinki', 'London', 'Stuttgart'}
     }

     cities_without_madrid = ['Brussels', 'Bucharest', 'Stuttgart', 'Mykonos', 'Helsinki', 'Split', 'London']

     found = False
     valid_permutation = None

     for perm in itertools.permutations(cities_without_madrid):
        # Check consecutive flights in the permutation
        valid = True
        # Check from first to last within the permutation
        for i in range(len(perm)-1):
            if perm[i+1] not in graph[perm[i]]:
                valid = False
                break
        if not valid:
            continue

        # Check the last city in the permutation to Madrid
        if 'Madrid' not in graph[perm[-1]]:
            continue

        # Check Stuttgart constraint
        cumulative = 0
        stuttgart_index = None
        for idx, city in enumerate(perm):
            if city == 'Stuttgart':
                stuttgart_index = idx
                break
        # If we found Stuttgart, compute the start day
        if stuttgart_index is not None:
            # The start day of Stuttgart is 1 + cumulative sum of (days_req[city]-1) for all cities before it
            start_day_stuttgart = 1
            for j in range(stuttgart_index):
                start_day_stuttgart += (days_req[perm[j]] - 1)
            if start_day_stuttgart > 4:
                valid = False
            else:
                valid = True
        else:
            # This should not happen because Stuttgart is in the list
            valid = False

        if valid:
            valid_permutation = perm
            found = True
            break

     if found:
        # Build the itinerary
        itinerary = []
        current_start = 1
        for city in valid_permutation:
            end_day = current_start + days_req[city] - 1
            itinerary.append({"day_range": f"Day {current_start}-{end_day}", "place": city})
            current_start = end_day
        # Add Madrid
        itinerary.append({"day_range": "Day 20-21", "place": "Madrid"})
        result = {"itinerary": itinerary}
        print(json.dumps(result))
     else:
        # Print an error message in JSON format?
        print(json.dumps({"error": "No valid itinerary found"}))
