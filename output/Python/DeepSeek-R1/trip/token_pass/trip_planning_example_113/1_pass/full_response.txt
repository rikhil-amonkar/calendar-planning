import json

def main():
    total_days = 12
    naples_days = 3
    seville_days = 4
    milan_days = 7
    seville_show_start = 9
    seville_show_end = 12
    
    # Calculate travel days needed (each travel day counts for two cities)
    total_city_days = naples_days + seville_days + milan_days
    travel_days = total_city_days - total_days  # Should be 2
    
    # Determine feasible route based on direct flights
    # Only possible route with direct flights: Naples -> Milan -> Seville
    naples_start = 1
    naples_end = naples_start + naples_days - 1
    
    milan_start = naples_end
    milan_end = milan_start + milan_days - 1
    
    seville_start = milan_end
    seville_end = seville_start + seville_days - 1
    
    # Verify constraints
    assert seville_start <= seville_show_start and seville_end >= seville_show_end, "Seville show constraints not satisfied"
    assert seville_end == total_days, "Itinerary does not match total days"
    
    itinerary = [
        {"day_range": f"Day {naples_start}-{naples_end}", "place": "Naples"},
        {"day_range": f"Day {milan_start}-{milan_end}", "place": "Milan"},
        {"day_range": f"Day {seville_start}-{seville_end}", "place": "Seville"}
    ]
    
    result = {"itinerary": itinerary}
    print(json.dumps(result))

if __name__ == "__main__":
    main()
