        import json

        geneva_days = 4
        munich_days = 7
        valencia_days = 6
        stuttgart_days = 2
        bucharest_days = 2

        start_day = 1
        itinerary = []

        # Geneva
        end_day = start_day + geneva_days - 1
        itinerary.append({"day_range": f"Day {start_day}-{end_day}", "place": "Geneva"})
        start_day = end_day

        # Munich
        end_day = start_day + munich_days - 1
        itinerary.append({"day_range": f"Day {start_day}-{end_day}", "place": "Munich"})
        start_day = end_day

        # Bucharest
        end_day = start_day + bucharest_days - 1
        itinerary.append({"day_range": f"Day {start_day}-{end_day}", "place": "Bucharest"})
        start_day = end_day

        # Valencia
        end_day = start_day + valencia_days - 1
        itinerary.append({"day_range": f"Day {start_day}-{end_day}", "place": "Valencia"})
        start_day = end_day

        # Stuttgart
        end_day = start_day + stuttgart_days - 1
        itinerary.append({"day_range": f"Day {start_day}-{end_day}", "place": "Stuttgart"})
