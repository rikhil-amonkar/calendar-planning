   import itertools

   graph = { ... }

   days = { ... }

   cities_list = list(days.keys())

   for perm in itertools.permutations(cities_list):
        # Check consecutive cities in perm are connected in graph
        valid_route = True
        for i in range(len(perm)-1):
            if perm[i+1] not in graph[perm[i]]:
                valid_route = False
                break
        if not valid_route:
            continue

        # Compute start days for each city in this permutation
        start_days = [1]  # start day of first city
        for i in range(len(perm)-1):
            start_days.append(start_days[-1] + days[perm[i]] - 1)

        # Compute end days
        end_days = [start_days[i] + days[city] - 1 for i, city in enumerate(perm)]

        # Check Porto constraint: find index of Porto in perm
        porto_index = None
        for idx, city in enumerate(perm):
            if city == 'Porto':
                porto_index = idx
                break
        if porto_index is None:
            continue

        # Check if Porto's stay overlaps with [16,18]
        porto_start = start_days[porto_index]
        porto_end = end_days[porto_index]
        if porto_end < 16 or porto_start > 18:
            continue

        # If we get here, we found a valid itinerary
        itinerary = []
        for idx, city in enumerate(perm):
            start = start_days[idx]
            end = end_days[idx]
            day_range_str = f"Day {start}-{end}" if start != end else f"Day {start}"
            itinerary.append({"day_range": day_range_str, "place": city})

        # We break after the first valid one
        break
