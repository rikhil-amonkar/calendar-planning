          def is_connected(city1, city2):
             return (city1, city2) in connections or (city2, city1) in connections

          def is_path_valid(path):
             for i in range(len(path)-1):
                 if not is_connected(path[i], path[i+1]):
                     return False
             return True

          start_city = "Warsaw"
          other_cities = list(cities.keys())
          other_cities.remove(start_city)
          valid_sequences = []

             if is_path_valid(full_path):
