busy_intervals.append([210, 240])
busy_intervals.append([270, 300])

# Natalie has no busy intervals

busy_intervals.append([0, 150])
busy_intervals.append([270, 480])

# Vincent's busy intervals
busy_intervals.append([0, 30])
busy_intervals.append([90, 180])
busy_intervals.append([210, 300])
busy_intervals.append([330, 480])

# Now sort the busy_intervals by start time
busy_intervals.sort(key=lambda x: x[0])

# Merge intervals
merged_busy = []
current_start, current_end = busy_intervals[0]
for i in range(1, len(busy_intervals)):
    if busy_intervals[i][0] <= current_end:
        # Overlap, so extend current_end if needed
        current_end = max(current_end, busy_intervals[i][1])
    else:
        # No overlap, add current interval and start new
        merged_busy.append([current_start, current_end])
        current_start, current_end = busy_intervals[i]
merged_busy.append([current_start, current_end])

# Now, merged_busy has the merged intervals. Now find free times between them and within work hours.
free_slots = []
# Check before first busy interval
if work_start < merged_busy[0][0]:
    free_slots.append([work_start, merged_busy[0][0]])

# Check between merged busy intervals
for i in range(len(merged_busy)-1):
    gap_start = merged_busy[i][1]
    gap_end = merged_busy[i+1][0]
    if gap_start < gap_end:
        free_slots.append([gap_start, gap_end])

# Check after last busy interval
if merged_busy[-1][1] < work_end:
    free_slots.append([merged_busy[-1][1], work_end])

# Now, find a free slot that has duration at least 30 minutes
meeting_duration = 30
for slot in free_slots:
    start, end = slot
    if end - start >= meeting_duration:
        # Found a slot
        meeting_start = start
        meeting_end = meeting_start + meeting_duration
        # Convert to time string from minutes
        # Since base is 9:00, add minutes to 9:00
        base_hour = 9
        total_minutes_start = meeting_start
        hours_start = total_minutes_start // 60
        minutes_start = total_minutes_start % 60
        time_start = f"{base_hour + hours_start:02d}:{minutes_start:02d}"

        total_minutes_end = meeting_end
        hours_end = total_minutes_end // 60
        minutes_end = total_minutes_end % 60
        time_end = f"{base_hour + hours_end:02d}:{minutes_end:02d}"

        day = "Monday"
        print(f"{time_start}:{time_end}")
        print(day)
        break
