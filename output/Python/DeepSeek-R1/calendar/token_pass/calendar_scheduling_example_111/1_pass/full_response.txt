def main():
    # Define work hours in minutes from 9:00 (0) to 17:00 (480)
    work_start = 0
    work_end = 480
    meeting_duration = 30

    # Busy intervals in minutes from 9:00
    busy_intervals = {
        "Gregory": [[0, 60], [90, 150], [210, 240], [270, 300]],
        "Natalie": [],
        "Christine": [[0, 150], [270, 480]],
        "Vincent": [[0, 30], [90, 180], [210, 300], [330, 480]]
    }

    # Compute free intervals for each person within work hours
    free_intervals = {}
    for person, intervals in busy_intervals.items():
        free = []
        current = work_start
        # Sort intervals by start time
        intervals.sort(key=lambda x: x[0])
        for start, end in intervals:
            if current < start:
                free.append([current, start])
            current = max(current, end)
        if current < work_end:
            free.append([current, work_end])
        free_intervals[person] = free

    # Find common free intervals by intersecting all free intervals
    common_free = free_intervals["Gregory"]
    for person in ["Natalie", "Christine", "Vincent"]:
        new_common = []
        i = j = 0
        list_b = free_intervals[person]
        while i < len(common_free) and j < len(list_b):
            a_start, a_end = common_free[i]
            b_start, b_end = list_b[j]
            start = max(a_start, b_start)
            end = min(a_end, b_end)
            if start < end:
                new_common.append([start, end])
            if a_end < b_end:
                i += 1
            else:
                j += 1
        common_free = new_common

    # Find the first common free interval that can fit the meeting
    for start, end in common_free:
        if end - start >= meeting_duration:
            meeting_start = start
            meeting_end = meeting_start + meeting_duration
            break
    else:
        # If no suitable interval found (though problem states there is one)
        meeting_start = None
        meeting_end = None

    # Convert minutes to time strings
    def minutes_to_time(minutes):
        total_minutes = minutes
        hours = total_minutes // 60
        mins = total_minutes % 60
        return f"{hours + 9:02d}:{mins:02d}"

    start_time_str = minutes_to_time(meeting_start)
    end_time_str = minutes_to_time(meeting_end)

    # Output the result
    print(f"{start_time_str}:{end_time_str}")
    print("Monday")

if __name__ == "__main__":
    main()
