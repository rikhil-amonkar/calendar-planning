def find_common_free(free1, free2):
    common = []
    i = 0
    j = 0
    while i < len(free1) and j < len(free2):
        start1, end1 = free1[i]
        start2, end2 = free2[j]
        start_max = max(start1, start2)
        end_min = min(end1, end2)
        if start_max < end_min:
            common.append((start_max, end_min))
        if end1 < end2:
            i += 1
        else:
            j += 1
    return common
