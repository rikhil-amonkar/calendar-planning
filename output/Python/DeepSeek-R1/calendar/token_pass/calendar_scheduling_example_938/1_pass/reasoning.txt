First, function find_free_intervals(busy_list, work_start, work_end):
    # busy_list is list of (start, end) in minutes
    # sort busy_list by start
    if not busy_list:
        return [(work_start, work_end)]  # entirely free
    sorted_busy = sorted(busy_list, key=lambda x: x[0])
    free_intervals = []
    current = work_start
    for start, end in sorted_busy:
        if current < start:
            free_intervals.append( (current, start) )
        current = max(current, end)
    if current < work_end:
        free_intervals.append( (current, work_end) )
    return free_intervals

Next, function find_overlapping_intervals(intervals1, intervals2):
    # find intervals that are common to both lists
    # both lists are sorted by start time
    overlapping = []
    i = 0
    j = 0
    while i < len(intervals1) and j < len(intervals2):
        start1, end1 = intervals1[i]
        start2, end2 = intervals2[j]
        # find the overlap between these two intervals
        start_max = max(start1, start2)
        end_min = min(end1, end2)
        if start_max < end_min:
            overlapping.append( (start_max, end_min) )
        # move the pointer for the interval that ends first
        if end1 < end2:
            i += 1
        else:
            j += 1
    return overlapping
