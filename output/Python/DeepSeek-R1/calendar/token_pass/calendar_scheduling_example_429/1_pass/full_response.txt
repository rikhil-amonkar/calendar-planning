def time_to_minutes(time_str):
    parts = time_str.split(':')
    hours = int(parts[0])
    minutes = int(parts[1])
    return hours * 60 + minutes

def minutes_to_time(minutes):
    hours = minutes // 60
    mins = minutes % 60
    return f"{hours:02d}:{mins:02d}"

merged_busy = []
current_start, current_end = busy_intervals[0]
for i in range(1, len(busy_intervals)):
    interval = busy_intervals[i]
    if interval[0] <= current_end:
        current_end = max(current_end, interval[1])
    else:
        merged_busy.append((current_start, current_end))
        current_start, current_end = interval
merged_busy.append((current_start, current_end))

free_intervals = []

    free_intervals.append((work_start_min, merged_busy[0][0]))

for i in range(len(merged_busy) - 1):

    free_end = merged_busy[i+1][0]
    if free_start < free_end:

if merged_busy[-1][1] < work_end_min:
    free_intervals.append((merged_busy[-1][1], work_end_min))

for free_start, free_end in free_intervals:
