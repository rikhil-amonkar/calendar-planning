def main():
    # Define work hours for Tuesday (since Nathan cannot meet on Monday)
    work_start = 9 * 60  # 9:00 in minutes
    work_end = 11 * 60   # 11:00 in minutes (Amanda doesn't want to meet after 11:00, so we consider up to 11:00)
    
    # Amanda's busy intervals on Tuesday (in minutes from midnight)
    amanda_busy = [
        (9 * 60, 9 * 60 + 30),   # 9:00-9:30
        (10 * 60, 10 * 60 + 30)   # 10:00-10:30
    ]
    
    # Nathan's busy intervals on Tuesday (in minutes from midnight)
    nathan_busy = [
        (9 * 60, 10 * 60 + 30)    # 9:00-10:30
    ]
    
    meeting_duration = 30  # minutes
    
    # Check every possible 30-minute slot between work_start and work_end
    for start_minute in range(work_start, work_end, 30):
        end_minute = start_minute + meeting_duration
        if end_minute > work_end:
            continue
        
        # Check if Amanda is free
        amanda_free = True
        for busy_start, busy_end in amanda_busy:
            if not (end_minute <= busy_start or start_minute >= busy_end):
                amanda_free = False
                break
        
        # Check if Nathan is free
        nathan_free = True
        for busy_start, busy_end in nathan_busy:
            if not (end_minute <= busy_start or start_minute >= busy_end):
                nathan_free = False
                break
        
        if amanda_free and nathan_free:
            # Format the time as HH:MM
            def format_time(minutes):
                hours = minutes // 60
                mins = minutes % 60
                return f"{hours:02d}:{mins:02d}"
            
            start_time_str = format_time(start_minute)
            end_time_str = format_time(end_minute)
            print(f"Tuesday {start_time_str}:{end_time_str}")
            return
    
    print("No suitable time found")  # According to the problem, a solution exists, so this should not happen

if __name__ == "__main__":
    main()
