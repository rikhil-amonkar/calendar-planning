Function get_free_intervals(busy_list, start=0, end=480):
    sorted_busy = sorted(busy_list, key=lambda x: x[0])
    free_intervals = []
    current = start
    for s,e in sorted_busy:
        if current < s:
            free_intervals.append( (current, s) )
        current = e
    if current < end:
        free_intervals.append( (current, end) )
    return free_intervals

This returns list of free intervals as (start, end) where end is not included, so [start, end)

def find_common_free(free1, free2):
    common = []
    i = 0
    j = 0
    while i < len(free1) and j < len(free2):
        interval1 = free1[i]
        interval2 = free2[j]
        start = max(interval1[0], interval2[0])
        end = min(interval1[1], interval2[1])
        if start < end:
            common.append( (start, end) )
        if interval1[1] < interval2[1]:
            i += 1
        else:
            j += 1
    return common
