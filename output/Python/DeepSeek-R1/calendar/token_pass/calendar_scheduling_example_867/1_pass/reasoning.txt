Define function get_free_intervals(busy_intervals, start, end):
    # busy_intervals is list of (s,e) where s inclusive, e exclusive
    # start and end are the overall range
    # sort busy intervals by s
    sorted_busy = sorted(busy_intervals, key=lambda x: x[0])
    free_intervals = []
    current = start
    for busy in sorted_busy:
