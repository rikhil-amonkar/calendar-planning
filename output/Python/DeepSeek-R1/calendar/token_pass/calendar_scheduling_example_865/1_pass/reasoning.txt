free_intervals = []

free_intervals.append( (current, s) )  # free from current to s

def get_free_intervals(busy_intervals, start_min=0, end_min=480):
    busy_sorted = sorted(busy_intervals, key=lambda x: x[0])
    free_intervals = []
    current = start_min
    for busy in busy_sorted:
        s, e = busy
        if current < s:
            free_intervals.append((current, s))
        current = e
    if current < end_min:
        free_intervals.append((current, end_min))
    return free_intervals

def minutes_to_time(minutes):
    total_minutes = minutes
    hours = 9 + total_minutes // 60
    minutes_part = total_minutes % 60
    return f"{hours:02d}:{minutes_part:02d}"

            if start_common < end_common:

    # Sort common_free by start time to ensure we get the earliest first
