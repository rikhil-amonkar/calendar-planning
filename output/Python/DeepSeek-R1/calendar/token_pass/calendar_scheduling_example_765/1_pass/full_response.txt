def main():
    # Define work hours in minutes from midnight
    work_start = 9 * 60  # 09:00
    work_end = 17 * 60   # 17:00

    # Meeting duration in minutes
    meeting_duration = 30

    # Busy intervals for each participant per day (in minutes from midnight)
    schedules = {
        'Monday': {
            'Joshua': [(15*60, 15*60 + 30)],  # 15:00-15:30
            'Joyce': [
                (9*60, 9*60 + 30),    # 09:00-09:30
                (10*60, 11*60),        # 10:00-11:00
                (11*60 + 30, 12*60 + 30), # 11:30-12:30
                (13*60, 15*60),        # 13:00-15:00
                (15*60 + 30, 17*60)    # 15:30-17:00
            ]
        },
        'Tuesday': {
            'Joshua': [
                (11*60 + 30, 12*60),   # 11:30-12:00
                (13*60, 13*60 + 30),   # 13:00-13:30
                (14*60 + 30, 15*60)    # 14:30-15:00
            ],
            'Joyce': [(9*60, 17*60)]   # 09:00-17:00
        },
        'Wednesday': {
            'Joshua': [],
            'Joyce': [
                (9*60, 9*60 + 30),    # 09:00-09:30
                (10*60, 11*60),        # 10:00-11:00
                (12*60 + 30, 15*60 + 30), # 12:30-15:30
                (16*60, 16*60 + 30)    # 16:00-16:30
            ]
        }
    }

    # Prefer after 12:00 on Monday for Joyce
    monday_cutoff = 12 * 60  # 12:00 in minutes

    days = ['Monday', 'Tuesday', 'Wednesday']

    def get_free_intervals(busy_intervals, start, end):
        if not busy_intervals:
            return [(start, end)]
        sorted_busy = sorted(busy_intervals, key=lambda x: x[0])
        free = []
        current = start
        for bus_start, bus_end in sorted_busy:
            if current < bus_start:
                free.append((current, bus_start))
            current = max(current, bus_end)
        if current < end:
            free.append((current, end))
        return free

    def adjust_for_preference(intervals, cutoff):
        adjusted = []
        for start, end in intervals:
            if end <= cutoff:
                continue
            if start < cutoff:
                start = cutoff
            if start < end:
                adjusted.append((start, end))
        return adjusted

    def find_meeting_slot(day, free1, free2, duration):
        i = j = 0
        while i < len(free1) and j < len(free2):
            start1, end1 = free1[i]
            start2, end2 = free2[j]
            start_max = max(start1, start2)
            end_min = min(end1, end2)
            if end_min - start_max >= duration:
                return start_max, start_max + duration
            if end1 < end2:
                i += 1
            else:
                j += 1
        return None

    def minutes_to_time(minutes):
        hours = minutes // 60
        mins = minutes % 60
        return f"{hours:02d}:{mins:02d}"

    for day in days:
        busy_joshua = schedules[day]['Joshua']
        busy_joyce = schedules[day]['Joyce']
        
        free_joshua = get_free_intervals(busy_joshua, work_start, work_end)
        free_joyce = get_free_intervals(busy_joyce, work_start, work_end)
        
        if day == 'Monday':
            free_joyce = adjust_for_preference(free_joyce, monday_cutoff)
        
        slot = find_meeting_slot(day, free_joshua, free_joyce, meeting_duration)
        if slot:
            start_minutes, end_minutes = slot
            start_time = minutes_to_time(start_minutes)
            end_time = minutes_to_time(end_minutes)
            print(f"{day}:{start_time}:{end_time}")
            return

    print("No suitable slot found")

if __name__ == "__main__":
    main()
