def main():
    # Meeting duration in minutes
    duration = 60
    # Work hours: from 0 to 480 minutes (9:00 to 17:00)
    min_time = 0
    max_time = 480
    # Latest start time: max_time - duration = 420

    # Define blocked intervals for each day for each person
    # Intervals are (start, end) with end exclusive
    gary_monday_blocked = [(30, 60), (120, 240), (300, 330), (450, 480)]
    david_monday_blocked = [(0, 30), (60, 240), (330, 450)]

    gary_tuesday_blocked = [(0, 30), (90, 120), (330, 420)]
    david_tuesday_blocked = [(0, 30), (60, 90), (120, 210), (240, 330), (360, 420), (450, 480)]

    # List of days with their blocked intervals for Gary and David
    days = [
        {"name": "Monday", "gary_blocked": gary_monday_blocked, "david_blocked": david_monday_blocked},
        {"name": "Tuesday", "gary_blocked": gary_tuesday_blocked, "david_blocked": david_tuesday_blocked}
    ]

    # Function to check if for a given start time S and blocked list, the interval [S, S+duration) is free
    def is_free(S, blocked_list):
        for blocked in blocked_list:
            start_block, end_block = blocked
            # Check if [S, S+duration) overlaps with [start_block, end_block)
            if S < end_block and S + duration > start_block:
                return False
        return True

    # Iterate over days
    for day in days:
        day_name = day["name"]
        gary_blocked = day["gary_blocked"]
        david_blocked = day["david_blocked"]
        # Iterate over possible start times from min_time to max_time - duration
        for S in range(min_time, max_time - duration + 1):
            if is_free(S, gary_blocked) and is_free(S, david_blocked):
                # Found a slot
                # Calculate start time
                start_min = S
                hour_start = 9 + start_min // 60
                minute_start = start_min % 60
                end_min = S + duration
                hour_end = 9 + end_min // 60
                minute_end = end_min % 60
                # Format time string
                time_str = f"{hour_start:02d}:{minute_start:02d}:{hour_end:02d}:{minute_end:02d}"
                print(f"{day_name}: {time_str}")
                return

    # If no slot found, but problem says there exists a solution, so we assume found.
    print("No suitable time found.")

if __name__ == "__main__":
    main()
