def main():
    # Define work hours: 9:00 to 17:00 in minutes from 9:00 (0 to 480 minutes)
    work_start = 0
    work_end = 480  # 17:00 is 8 hours after 9:00 -> 8*60=480
    
    # Define busy intervals for each participant in minutes from 9:00
    # Intervals are half-open [start, end)
    busy_intervals = []
    
    # Melissa: meetings at 10:00-11:00, 12:30-14:00, 15:00-15:30
    busy_intervals.append((60, 120))   # 10:00-11:00
    busy_intervals.append((210, 300))  # 12:30-14:00
    busy_intervals.append((360, 390))  # 15:00-15:30
    
    # Gregory: blocked 12:30-13:00, 15:30-16:00
    busy_intervals.append((210, 240))  # 12:30-13:00
    busy_intervals.append((390, 420))  # 15:30-16:00
    
    # Victoria: busy at 9:00-9:30, 10:30-11:30, 13:00-14:00, 14:30-15:00, 15:30-16:30
    busy_intervals.append((0, 30))     # 9:00-9:30
    busy_intervals.append((90, 150))   # 10:30-11:30
    busy_intervals.append((240, 300))  # 13:00-14:00
    busy_intervals.append((330, 360))  # 14:30-15:00
    busy_intervals.append((390, 450))  # 15:30-16:30
    
    # Thomas: meetings at 10:00-12:00, 12:30-13:00, 14:30-16:00
    busy_intervals.append((60, 180))   # 10:00-12:00
    busy_intervals.append((210, 240))  # 12:30-13:00
    busy_intervals.append((330, 420))  # 14:30-16:00
    
    # Jennifer: blocked at 9:00-9:30, 10:00-10:30, 11:00-13:00, 13:30-14:30, 15:00-15:30, 16:00-16:30
    busy_intervals.append((0, 30))     # 9:00-9:30
    busy_intervals.append((60, 90))    # 10:00-10:30
    busy_intervals.append((120, 240))  # 11:00-13:00
    busy_intervals.append((270, 330))  # 13:30-14:30
    busy_intervals.append((360, 390))  # 15:00-15:30
    busy_intervals.append((420, 450))  # 16:00-16:30
    
    # Wayne and Catherine have no busy intervals, so not added
    
    # Sort busy intervals by start time
    busy_intervals.sort(key=lambda x: x[0])
    
    # Merge busy intervals
    merged_busy = []
    if busy_intervals:
        current_start, current_end = busy_intervals[0]
        for interval in busy_intervals[1:]:
            if interval[0] <= current_end:  # Overlapping or touching
                current_end = max(current_end, interval[1])
            else:
                merged_busy.append((current_start, current_end))
                current_start, current_end = interval
        merged_busy.append((current_start, current_end))
    else:
        merged_busy = []
    
    # Find free intervals within work hours
    free_intervals = []
    previous_end = work_start
    for busy in merged_busy:
        if busy[0] > previous_end:
            free_intervals.append((previous_end, busy[0]))
        previous_end = busy[1]
    if previous_end < work_end:
        free_intervals.append((previous_end, work_end))
    
    # Find free intervals that are at least 30 minutes long and after 14:00 (300 minutes)
    meeting_duration = 30
    suitable_slots = []
    for free in free_intervals:
        start, end = free
        if end - start >= meeting_duration:
            # Check if the slot starts at or after 14:00 (300 minutes)
            if start >= 300:
                suitable_slots.append(free)
    
    # Since there is a solution, we take the first suitable slot after 14:00
    if suitable_slots:
        slot_start, slot_end = suitable_slots[0]
        # But we need a 30-minute meeting, so we can take from slot_start to slot_start + 30
        # Since the free interval might be longer, we take the first 30 minutes after 14:00
        meeting_start = slot_start
        meeting_end = meeting_start + meeting_duration
        # Convert minutes to time string
        def min_to_time(minutes):
            hours = 9 + minutes // 60
            mins = minutes % 60
            return f"{hours:02d}:{mins:02d}"
        start_time_str = min_to_time(meeting_start)
        end_time_str = min_to_time(meeting_end)
        print(f"Monday {start_time_str}:{end_time_str}")
    else:
        # Fallback: if no slot after 14:00, take any slot, but according to problem, there is a solution.
        # But in this case, there is a slot before, but Wayne prefers after, so we should have found one.
        print("No suitable slot found after 14:00")

if __name__ == "__main__":
    main()
