def main():
    # Define the start and end of the work day in minutes from 9:00
    work_start = 0  # 9:00
    work_end = 480  # 17:00

    # Define busy intervals for each participant in minutes from 9:00
    # Each interval is (start, end) meaning [start, end) busy
    busy_intervals = []
    
    # Cynthia's busy times
    busy_intervals.append((30, 90))   # 9:30-10:30
    busy_intervals.append((150, 180)) # 11:30-12:00
    busy_intervals.append((240, 270)) # 13:00-13:30
    busy_intervals.append((360, 420)) # 15:00-16:00
    
    # Lauren's busy times
    busy_intervals.append((0, 30))    # 9:00-9:30
    busy_intervals.append((90, 120))  # 10:30-11:00
    busy_intervals.append((150, 180)) # 11:30-12:00
    busy_intervals.append((240, 270)) # 13:00-13:30
    busy_intervals.append((300, 330)) # 14:00-14:30
    busy_intervals.append((360, 390)) # 15:00-15:30
    busy_intervals.append((420, 480)) # 16:00-17:00
    
    # Robert's busy times
    busy_intervals.append((90, 120))  # 10:30-11:00
    busy_intervals.append((150, 180)) # 11:30-12:00
    busy_intervals.append((210, 270)) # 12:30-13:30
    busy_intervals.append((300, 420)) # 14:00-16:00
    
    # Steven and Roy are free all day, so no busy intervals

    # Merge busy intervals
    busy_intervals.sort(key=lambda x: x[0])  # sort by start time
    merged_busy = []
    if busy_intervals:
        current_start, current_end = busy_intervals[0]
        for interval in busy_intervals[1:]:
            if interval[0] <= current_end:
                # Overlapping intervals, update the end if needed
                current_end = max(current_end, interval[1])
            else:
                # No overlap, add the current interval and reset
                merged_busy.append((current_start, current_end))
                current_start, current_end = interval
        merged_busy.append((current_start, current_end))
    else:
        merged_busy = []

    # Find free intervals by looking at gaps between merged busy intervals and boundaries
    free_intervals = []
    # Before first busy interval
    if merged_busy and merged_busy[0][0] > work_start:
        free_intervals.append((work_start, merged_busy[0][0]))
    
    # Between busy intervals
    for i in range(len(merged_busy) - 1):
        gap_start = merged_busy[i][1]
        gap_end = merged_busy[i+1][0]
        if gap_start < gap_end:
            free_intervals.append((gap_start, gap_end))
    
    # After last busy interval
    if merged_busy and merged_busy[-1][1] < work_end:
        free_intervals.append((merged_busy[-1][1], work_end))
    
    # If no busy intervals, the whole day is free
    if not merged_busy:
        free_intervals.append((work_start, work_end))

    # Find the first free interval that has at least 30 minutes
    meeting_duration = 30
    for start, end in free_intervals:
        duration = end - start
        if duration >= meeting_duration:
            # Found a slot, convert to time string
            meeting_start = start
            meeting_end = meeting_start + meeting_duration
            # Convert minutes to HH:MM
            start_hour = 9 + meeting_start // 60
            start_minute = meeting_start % 60
            end_hour = 9 + meeting_end // 60
            end_minute = meeting_end % 60
            # Format as HH:MM
            time_str = f"{start_hour:02d}:{start_minute:02d}:{end_hour:02d}:{end_minute:02d}"
            print("Monday", time_str)
            return
    
    # If no slot found, but problem says there is a solution
    print("No suitable time found")

if __name__ == "__main__":
    main()
