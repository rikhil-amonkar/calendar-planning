def get_free_intervals(busy_list, start_time, end_time):
    # sort the busy list by start time
    sorted_busy = sorted(busy_list, key=lambda x: x[0])
    free_intervals = []
    current = start_time
    for busy in sorted_busy:
        if busy[0] > current:
            free_intervals.append((current, busy[0]))
        current = max(current, busy[1])
    if current < end_time:
        free_intervals.append((current, end_time))
    return free_intervals

# Function to find overlapping intervals between two lists
def find_overlapping_intervals(intervals1, intervals2):
    overlaps = []
    for i1 in intervals1:
        for i2 in intervals2:
            start = max(i1[0], i2[0])
            end = min(i1[1], i2[1])
            if start < end:
                overlaps.append((start, end))
    return overlaps
