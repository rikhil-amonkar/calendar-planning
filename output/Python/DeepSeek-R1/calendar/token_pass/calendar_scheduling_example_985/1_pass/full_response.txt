def main():
    days = ["Monday", "Tuesday", "Wednesday", "Thursday", "Friday"]
    
    diane_busy_per_day = {
        'Monday': [(180, 210), (360, 390)],
        'Tuesday': [(60, 120), (150, 180), (210, 240), (420, 480)],
        'Wednesday': [(0, 30), (330, 360), (450, 480)],
        'Thursday': [(390, 450)],
        'Friday': [(30, 150), (330, 360), (420, 480)]
    }
    
    matthew_busy_per_day = {
        'Monday': [(0, 60), (90, 480)],
        'Tuesday': [(0, 480)],
        'Wednesday': [(0, 120), (180, 330), (420, 480)],
        'Thursday': [(0, 420)],
        'Friday': [(0, 480)]
    }
    
    def merge_intervals(intervals):
        if not intervals:
            return []
        sorted_intervals = sorted(intervals, key=lambda x: x[0])
        merged = [sorted_intervals[0]]
        for current in sorted_intervals[1:]:
            last = merged[-1]
            if current[0] <= last[1]:
                merged[-1] = (last[0], max(last[1], current[1]))
            else:
                merged.append(current)
        return merged

    def get_free_intervals(busy_intervals, total=480):
        merged_busy = merge_intervals(busy_intervals)
        free = []
        start = 0
        for busy in merged_busy:
            if start < busy[0]:
                free.append((start, busy[0]))
            start = busy[1]
        if start < total:
            free.append((start, total))
        return free

    def intersect_intervals(intervals1, intervals2):
        i = j = 0
        common = []
        while i < len(intervals1) and j < len(intervals2):
            low = max(intervals1[i][0], intervals2[j][0])
            high = min(intervals1[i][1], intervals2[j][1])
            if low < high:
                common.append((low, high))
            if intervals1[i][1] < intervals2[j][1]:
                i += 1
            else:
                j += 1
        return common

    def minutes_to_time(m):
        total_minutes = 9 * 60 + m
        hours = total_minutes // 60
        minutes = total_minutes % 60
        return f"{hours:02d}:{minutes:02d}"

    for day in days:
        diane_busy = diane_busy_per_day.get(day, [])
        matthew_busy = matthew_busy_per_day.get(day, [])
        
        free_diane = get_free_intervals(diane_busy)
        free_matthew = get_free_intervals(matthew_busy)
        
        common_free = intersect_intervals(free_diane, free_matthew)
        
        if day == "Wednesday":
            adjusted_common = []
            for interval in common_free:
                start = max(interval[0], 210)
                end = interval[1]
                if start < end:
                    adjusted_common.append((start, end))
            common_free = adjusted_common
            
        for interval in common_free:
            start, end = interval
            if end - start >= 60:
                time_str = f"{minutes_to_time(start)}:{minutes_to_time(start + 60)}"
                print(day)
                print(time_str)
                return
                
    print("No suitable time found.")

if __name__ == "__main__":
    main()
