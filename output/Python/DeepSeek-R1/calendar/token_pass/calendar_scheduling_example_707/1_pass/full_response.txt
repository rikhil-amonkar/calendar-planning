def time_to_minutes(time_str):
    hours, minutes = map(int, time_str.split(':'))
    return hours * 60 + minutes

def minutes_to_time(minutes):
    hours = minutes // 60
    mins = minutes % 60
    return f"{hours:02d}:{mins:02d}"

def get_free_slots(busy_intervals, day_start, day_end):
    free_slots = []
    current_time = day_start
    for start, end in busy_intervals:
        if current_time < start:
            free_slots.append((current_time, start))
        current_time = max(current_time, end)
    if current_time < day_end:
        free_slots.append((current_time, day_end))
    return free_slots

def find_meeting_slot(ryan_busy, adam_busy, day_start, day_end, duration, day_name, adam_constraint_start=None):
    ryan_free = get_free_slots(ryan_busy, day_start, day_end)
    adam_free = get_free_slots(adam_busy, day_start, day_end)
    
    if adam_constraint_start is not None:
        adam_free = [(start, end) for start, end in adam_free if start >= adam_constraint_start]
    
    common_free = []
    for r_start, r_end in ryan_free:
        for a_start, a_end in adam_free:
            start = max(r_start, a_start)
            end = min(r_end, a_end)
            if start < end:
                common_free.append((start, end))
    
    for start, end in common_free:
        if end - start >= duration:
            slot_start = start
            slot_end = slot_start + duration
            if slot_end <= end:
                return minutes_to_time(slot_start), minutes_to_time(slot_end)
    return None, None
