Code for merging intervals:

def merge_intervals(intervals):
    if not intervals:
        return []
    sorted_intervals = sorted(intervals, key=lambda x: x[0])
    merged = []
    start, end = sorted_intervals[0]
    for i in range(1, len(sorted_intervals)):
        if sorted_intervals[i][0] <= end:
            end = max(end, sorted_intervals[i][1])
        else:
            merged.append((start, end))
            start, end = sorted_intervals[i]
    merged.append((start, end))
    return merged

Then, free intervals from merged blocked intervals and range (min_time, max_time):

def get_free_intervals(blocked_list, min_time, max_time):
    merged_blocked = merge_intervals(blocked_list)
    free_intervals = []
    current_start = min_time
    for blocked in merged_blocked:
        if current_start < blocked[0]:
            free_intervals.append((current_start, blocked[0]))
        current_start = blocked[1]
    if current_start < max_time:
        free_intervals.append((current_start, max_time))
    return free_intervals

    for start1, end1 in intervals1:
        for start2, end2 in intervals2:

            if start_overlap < end_overlap:
