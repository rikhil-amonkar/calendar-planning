```python
def main():
    # Define work hours (9:00 to 17:00) in minutes from 0 (9:00) to 480 (17:00)
    day_start = 9 * 60
    day_end = 17 * 60
    duration = 30
    total_minutes = day_end - day_start
    free = [True] * total_minutes

    # Define busy intervals for each participant in minutes relative to 9:00
    participants = {
        "Joe": [(9*60+30 - day_start, 10*60 - day_start), (10*60+30 - day_start, 11*60 - day_start)],
        "Keith": [(11*60+30 - day_start, 12*60 - day_start), (15*60 - day_start, 15*60+30 - day_start)],
        "Patricia": [(9*60 - day_start, 9*60+30 - day_start), (13*60 - day_start, 13*60+30 - day_start)],
        "Nancy": [(9*60 - day_start, 11*60 - day_start), (11*60+30 - day_start, 16*60+30 - day_start)],
        "Pamela": [
            (9*60 - day_start, 10*60 - day_start),
            (10*60+30 - day_start, 11*60 - day_start),
            (11*60+30 - day_start, 12*60+30 - day_start),
            (13*60 - day_start, 14*60 - day_start),
            (14*60+30 - day_start, 15*60 - day_start),
            (15*60+30 - day_start, 16*60 - day_start),
            (16*60+30 - day_start, 17*60 - day_start)
        ]
    }

    # Mark busy times in the free array
    for intervals in participants.values():
        for start, end in intervals:
            for i in range(start, min(end, total_minutes)):
                free[i] = False

    # Find the first available slot of duration minutes
    found = False
    start_index = -1
    for i in range(0, total_minutes - duration + 1):
        if all(free[i + j] for j in range(duration)):
            start_index = i
            found = True
            break

    if not found:
        print("No suitable time found.")
        return

    # Convert start index back to time
    total_minutes_start = day_start + start_index
    hours_start = total_minutes_start // 60
    minutes_start = total_minutes_start % 60

    total_minutes_end = total_minutes_start + duration
    hours_end = total_minutes_end // 60
    minutes_end = total_minutes_end % 60

    # Format the output
    print(f"Monday {hours_start:02d}:{minutes_start:02d}:{hours_end:02d}:{minutes_end:02d}")

if __name__ == "__main__":
    main()
```
