    def compute_free(busy_intervals, day_start, day_end):
        free = []
        busy_sorted = sorted(busy_intervals, key=lambda x: x[0])
        current = day_start
        for start, end in busy_sorted:
            if current < start:
                free.append((current, start))
            current = max(current, end)
        if current < day_end:
            free.append((current, day_end))
        return free
