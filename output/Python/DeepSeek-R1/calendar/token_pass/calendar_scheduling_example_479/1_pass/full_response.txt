def main():
    # Work hours: 9:00 to 17:00 (8 hours = 480 minutes)
    work_start = 9 * 60  # 9:00 in minutes
    work_end = 17 * 60   # 17:00 in minutes
    duration = 60         # Meeting duration in minutes

    # Convert each busy interval to minutes since midnight (but relative to 9:00 as 0)
    # We'll represent times as minutes from 9:00 (0 to 480)
    busy_intervals = []
    
    # Joshua's busy intervals
    busy_intervals.extend([(11*60 - work_start, 12*60 + 30 - work_start),
                          (13*60 + 30 - work_start, 14*60 + 30 - work_start),
                          (16*60 + 30 - work_start, work_end - work_start)])
    
    # Jerry's busy intervals
    busy_intervals.extend([(0, 30),
                          (10*60 + 30 - work_start, 12*60 - work_start),
                          (12*60 + 30 - work_start, 13*60 - work_start),
                          (13*60 + 30 - work_start, 14*60 - work_start),
                          (14*60 + 30 - work_start, 15*60 - work_start),
                          (15*60 + 30 - work_start, 16*60 - work_start)])
    
    # Jesse's busy intervals
    busy_intervals.extend([(0, 30),
                          (10*60 + 30 - work_start, 12*60 - work_start),
                          (12*60 + 30 - work_start, 13*60 - work_start),
                          (14*60 + 30 - work_start, 15*60 - work_start),
                          (15*60 + 30 - work_start, 16*60 + 30 - work_start)])
    
    # Kenneth's busy intervals
    busy_intervals.extend([(10*60 + 30 - work_start, 12*60 + 30 - work_start),
                          (13*60 + 30 - work_start, 14*60 - work_start),
                          (14*60 + 30 - work_start, 15*60 - work_start),
                          (15*60 + 30 - work_start, 16*60 - work_start),
                          (16*60 + 30 - work_start, work_end - work_start)])

    # Create a timeline for the work day (480 minutes)
    timeline = [True] * (work_end - work_start)  # True means free
    
    # Mark busy intervals on the timeline
    for start, end in busy_intervals:
        for i in range(int(start), int(end)):
            if i < len(timeline):
                timeline[i] = False

    # Find the first available slot of duration minutes
    start_minute = None
    count = 0
    for i in range(len(timeline)):
        if timeline[i]:
            count += 1
            if count == duration:
                start_minute = i - duration + 1
                break
        else:
            count = 0

    # Convert start_minute back to time
    total_minutes_start = work_start + start_minute
    hours_start = total_minutes_start // 60
    minutes_start = total_minutes_start % 60

    total_minutes_end = total_minutes_start + duration
    hours_end = total_minutes_end // 60
    minutes_end = total_minutes_end % 60

    # Format the output
    print(f"{hours_start:02d}:{minutes_start:02d}:{hours_end:02d}:{minutes_end:02d}")
    print("Monday")

if __name__ == "__main__":
    main()
