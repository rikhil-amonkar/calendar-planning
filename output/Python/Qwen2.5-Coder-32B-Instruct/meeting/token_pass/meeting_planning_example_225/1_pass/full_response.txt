```python
import json
from datetime import datetime, timedelta

# Define travel times as a dictionary of dictionaries
travel_times = {
    "Sunset District": {"North Beach": 29, "Union Square": 30, "Alamo Square": 17},
    "North Beach": {"Sunset District": 27, "Union Square": 7, "Alamo Square": 16},
    "Union Square": {"Sunset District": 26, "North Beach": 10, "Alamo Square": 15},
    "Alamo Square": {"Sunset District": 16, "North Beach": 15, "Union Square": 14}
}

# Define meeting constraints
meeting_constraints = {
    "Sarah": {"location": "North Beach", "start": "16:00", "end": "18:15", "min_duration": 60},
    "Jeffrey": {"location": "Union Square", "start": "15:00", "end": "22:00", "min_duration": 75},
    "Brian": {"location": "Alamo Square", "start": "16:00", "end": "17:30", "min_duration": 75}
}

def parse_time(time_str):
    return datetime.strptime(time_str, "%H:%M")

def format_time(dt):
    return dt.strftime("%H:%M")

def calculate_schedule():
    current_location = "Sunset District"
    current_time = parse_time("9:00")
    itinerary = []

    # Sort friends by their earliest available time to try and meet them as soon as possible
    sorted_friends = sorted(meeting_constraints.items(), key=lambda x: parse_time(x[1]['start']))

    for friend_name, constraints in sorted_friends:
        location = constraints['location']
        start_time = parse_time(constraints['start'])
        end_time = parse_time(constraints['end'])
        min_duration = timedelta(minutes=constraints['min_duration'])

        # Calculate the latest time we can leave the current location to still meet the friend
        latest_leave_time = end_time - min_duration

        # Calculate travel time to the friend's location
        travel_time = timedelta(minutes=travel_times[current_location][location])

        # Check if we can reach the friend's location in time
        if current_time + travel_time <= latest_leave_time:
            # We can meet the friend
            meeting_start_time = current_time + travel_time
            meeting_end_time = meeting_start_time + min_duration

            # Add the meeting to the itinerary
            itinerary.append({
                "action": "meet",
                "location": location,
                "person": friend_name,
                "start_time": format_time(meeting_start_time),
                "end_time": format_time(meeting_end_time)
            })

            # Update the current time and location
            current_time = meeting_end_time
            current_location = location

    return itinerary

# Calculate the optimal schedule
optimal_itinerary = calculate_schedule()

# Output the result as a JSON-formatted dictionary
result = {"itinerary": optimal_itinerary}
print(json.dumps(result, indent=2))
```

This code calculates the optimal meeting schedule based on the given constraints and outputs it as a JSON-formatted dictionary. It takes into account travel times and ensures that all meetings fit within the specified constraints.
