```python
import json
from datetime import datetime, timedelta

# Define travel times between locations
travel_times = {
    ('The Castro', 'Alamo Square'): 8,
    ('The Castro', 'Union Square'): 19,
    ('The Castro', 'Chinatown'): 20,
    ('Alamo Square', 'The Castro'): 8,
    ('Alamo Square', 'Union Square'): 14,
    ('Alamo Square', 'Chinatown'): 16,
    ('Union Square', 'The Castro'): 19,
    ('Union Square', 'Alamo Square'): 15,
    ('Union Square', 'Chinatown'): 7,
    ('Chinatown', 'The Castro'): 22,
    ('Chinatown', 'Alamo Square'): 17,
    ('Chinatown', 'Union Square'): 7,
}

# Define friends' availability and meeting durations
friends = {
    'Emily': {'location': 'Alamo Square', 'start': '11:45', 'end': '15:15', 'min_duration': 105},
    'Barbara': {'location': 'Union Square', 'start': '16:45', 'end': '18:15', 'min_duration': 60},
    'William': {'location': 'Chinatown', 'start': '17:15', 'end': '19:00', 'min_duration': 105},
}

# Convert times to datetime objects for easier manipulation
def convert_to_datetime(time_str):
    return datetime.strptime(time_str, '%H:%M')

# Check if a meeting can fit within a person's availability
def can_meet(start, end, person):
    person_start = convert_to_datetime(friends[person]['start'])
    person_end = convert_to_datetime(friends[person]['end'])
    return person_start <= start <= end <= person_end

# Calculate the total duration of a meeting
def meeting_duration(start, end):
    return int((end - start).total_seconds() / 60)

# Backtracking function to find the optimal schedule
def backtrack(current_location, current_time, visited, itinerary):
    if len(visited) == len(friends):
        return itinerary

    best_itinerary = None
    for friend, details in friends.items():
        if friend not in visited:
            location = details['location']
            min_duration = details['min_duration']
            
            # Calculate travel time to the next location
            travel_time = travel_times[(current_location, location)]
            next_time = current_time + timedelta(minutes=travel_time)
            
            # Find the latest possible start time for the meeting
            meeting_start = max(next_time, convert_to_datetime(details['start']))
            meeting_end = meeting_start + timedelta(minutes=min_duration)
            
            # Check if the meeting fits within the person's availability
            if can_meet(meeting_start, meeting_end, friend):
                new_itinerary = itinerary + [{
                    "action": "meet",
                    "location": location,
                    "person": friend,
                    "start_time": meeting_start.strftime('%H:%M'),
                    "end_time": meeting_end.strftime('%H:%M')
                }]
                
                result = backtrack(location, meeting_end, visited | {friend}, new_itinerary)
                if result:
                    if not best_itinerary or meeting_duration(convert_to_datetime(result[-1]['start_time']), convert_to_datetime(result[-1]['end_time'])) > meeting_duration(convert_to_datetime(best_itinerary[-1]['start_time']), convert_to_datetime(best_itinerary[-1]['end_time'])):
                        best_itinerary = result
    
    return best_itinerary

# Start the backtracking process
start_time = convert_to_datetime('9:00')
initial_location = 'The Castro'
optimal_itinerary = backtrack(initial_location, start_time, set(), [])

# Output the result as a JSON-formatted dictionary
output = {
    "itinerary": optimal_itinerary
}

print(json.dumps(output, indent=2))
```

This code models the problem using a graph representation of travel times, defines the constraints for each friend, and uses a backtracking algorithm to explore all possible schedules. It evaluates each schedule based on the constraints and selects the optimal one that maximizes the time spent meeting friends. The result is formatted as a JSON object and printed.
