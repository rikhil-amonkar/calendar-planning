# Define travel times as a dictionary of dictionaries

# Define friends' availability and required meeting durations
friends = {
    "Laura": {"location": "The Castro", "start": "19:45", "end": "21:30", "duration": 105},
    "Daniel": {"location": "Golden Gate Park", "start": "21:15", "end": "21:45", "duration": 15},
    "William": {"location": "Embarcadero", "start": "7:00", "end": "9:00", "duration": 90},
    "Karen": {"location": "Russian Hill", "start": "14:30", "end": "19:45", "duration": 30},

    "Kimberly": {"location": "North Beach", "start": "15:45", "end": "19:15", "duration": 30}
}

def parse_time(time_str):
    return datetime.strptime(time_str, "%H:%M")

def add_minutes_to_time(time, minutes):
    return (time + timedelta(minutes=minutes)).strftime("%H:%M")

def can_meet(start, end, required_duration):
    start_time = parse_time(start)
    end_time = parse_time(end)
    return (end_time - start_time).total_seconds() / 60 >= required_duration

def find_best_schedule(current_location, current_time, visited_friends, itinerary):
    global best_itinerary
    if len(visited_friends) == len(friends):
        if len(itinerary) > len(best_itinerary):
            best_itinerary = itinerary.copy()
        return

    for friend, details in friends.items():
        if friend in visited_friends:
            continue

        travel_time = travel_times[current_location][details["location"]]
        meeting_start = add_minutes_to_time(current_time, travel_time)

        if can_meet(meeting_start, details["end"], details["duration"]):
            meeting_end = add_minutes_to_time(meeting_start, details["duration"])
            if parse_time(meeting_end) <= parse_time(details["end"]):
                new_itinerary = itinerary + [{
                    "action": "meet",
                    "location": details["location"],
                    "person": friend,
                    "start_time": meeting_start,
                    "end_time": meeting_end
                }]
                find_best_schedule(details["location"], meeting_end, visited_friends | {friend}, new_itinerary)

best_itinerary = []
find_best_schedule("Fisherman's Wharf", "9:00", set(), [])

print(json.dumps({"itinerary": best_itinerary}, indent=2))
