import json
from datetime import datetime, timedelta

# Constants
arrival_time = 9 * 60  # 9:00 AM in minutes since midnight
travel_times = {

    ('Union Square', 'Sunset District'): 26,
    ('Mission District', 'Union Square'): 15,
    ('Mission District', 'Bayview'): 15,

    ('Bayview', 'Sunset District'): 23,
    ('Sunset District', 'Union Square'): 30,
    ('Sunset District', 'Mission District'): 24,
    ('Sunset District', 'Bayview'): 22,
}

# Constraints
constraints = {
    'Rebecca': {'location': 'Mission District', 'start': 11 * 60 + 30, 'end': 20 * 60 + 15, 'duration': 120},
    'Karen': {'location': 'Bayview', 'start': 12 * 60 + 45, 'end': 15 * 60, 'duration': 120},
    'Carol': {'location': 'Sunset District', 'start': 10 * 60 + 15, 'end': 11 * 60 + 45, 'duration': 30},
}

def parse_time(time_str):
    return int(time_str.split(':')[0]) * 60 + int(time_str.split(':')[1])

def format_time(minutes):
    hours = minutes // 60
    minutes = minutes % 60
    return f"{hours}:{minutes}" if minutes >= 10 else f"{hours}:0{minutes}"

def get_possible_meetings(constraint):
    possible_meetings = []
    start = constraint['start']
    end = constraint['end']
    duration = constraint['duration']
    for start_time in range(start, end - duration + 1):
        end_time = start_time + duration
        possible_meetings.append((start_time, end_time))
    return possible_meetings

def find_optimal_schedule(arrival_time, constraints, travel_times):
    possible_schedules = []
    locations = ['Union Square'] + [constraint['location'] for constraint in constraints.values()]
    
    # Generate possible meeting times for each person
    possible_meetings = {name: get_possible_meetings(constraint) for name, constraint in constraints.items()}
    
    def backtrack(current_location, current_time, visited, itinerary):
        if len(visited) == len(constraints):
            possible_schedules.append(itinerary.copy())
            return
        
        for name, constraint in constraints.items():
            if name not in visited:
                location = constraint['location']
                for start_time, end_time in possible_meetings[name]:
                    travel_time = travel_times[(current_location, location)]
                    if current_time + travel_time <= start_time:
                        backtrack(location, end_time, visited | {name}, itinerary + [(name, location, start_time, end_time)])
    
    backtrack('Union Square', arrival_time, set(), [])
    
    # Select the schedule that visits the most people
    optimal_schedule = max(possible_schedules, key=len, default=[])
    
    # Format the optimal schedule as required
    formatted_schedule = []
    current_location = 'Union Square'
    current_time = arrival_time
    for name, location, start_time, end_time in optimal_schedule:
        travel_time = travel_times[(current_location, location)]
        formatted_schedule.append({
            "action": "travel",
            "location": location,
            "person": None,
            "start_time": format_time(current_time),
            "end_time": format_time(current_time + travel_time)
        })
        formatted_schedule.append({
            "action": "meet",
            "location": location,
            "person": name,
            "start_time": format_time(start_time),
            "end_time": format_time(end_time)
        })
        current_location = location
        current_time = end_time
    
    return {"itinerary": formatted_schedule}

optimal_schedule = find_optimal_schedule(arrival_time, constraints, travel_times)
print(json.dumps(optimal_schedule, indent=2))
