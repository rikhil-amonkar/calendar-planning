def convert_to_minutes(time_str):
    h, m = map(int, time_str.split(':'))
    return h * 60 + m

def convert_to_time(minutes):
    h = minutes // 60
    m = minutes % 60
    return f"{h}:{m}"

def find_best_schedule(start_location, start_time, constraints, travel_times):
    def backtrack(current_location, current_time, visited, schedule):
        nonlocal best_schedule, best_meetings

        # Check if we can improve the best schedule
        if len(visited) > best_meetings:
            best_schedule = schedule[:]
            best_meetings = len(visited)

        # Try to visit each friend
        for name, constraint in constraints.items():
            if name in visited:
                continue

            location = constraint['location']
            required_start = constraint['start']
            required_end = constraint['end']
            required_duration = constraint['duration']

            # Calculate travel time
            travel_time = travel_times.get((current_location, location), float('inf'))

            # Calculate potential meeting start and end times
            potential_start = current_time + travel_time
            potential_end = potential_start + required_duration

            # Check if the meeting can happen within the friend's availability
            if required_start <= potential_start and potential_end <= required_end:
                # Add the meeting to the schedule
                schedule.append({
                    'action': 'meet',

                    'person': name,
                    'start_time': convert_to_time(potential_start),

                })

                # Recur for the next meeting
                backtrack(location, potential_end, visited, schedule)

                # Backtrack

                visited.remove(name)

    best_schedule = []
    best_meetings = 0

    # Start backtracking from the initial location and time
    backtrack(start_location, start_time, set(), [])

    return best_schedule
