def time_to_minutes(time_str):
    """Convert time in 'H:MM' format to minutes since midnight."""
    h, m = map(int, time_str.split(':'))
    return h * 60 + m

def minutes_to_time(minutes):
    """Convert minutes since midnight to 'H:MM' format."""
    h = minutes // 60
    m = minutes % 60
    return f"{h}:{m:02}"

def is_valid_schedule(schedule, meeting):
    """Check if adding a meeting to the schedule is valid."""
    last_meeting = schedule[-1] if schedule else None
    meeting_start = time_to_minutes(meeting['start'])
    meeting_end = time_to_minutes(meeting['end'])
    meeting_duration = meeting['duration']
    
    # Check if the meeting can fit within its time window
    if meeting_end - meeting_start < meeting_duration:
        return False
    
    # Check if the meeting can start after traveling from the last location
    if last_meeting:
        last_end_time = last_meeting['end_time']
        travel_time = travel_times[(last_meeting['location'], meeting['location'])]
        if last_end_time + travel_time > meeting_start:
            return False
    
    return True

def backtrack(schedule, available_meetings, best_schedule):
    """Backtrack to find the optimal schedule."""
    if not available_meetings:
        if len(schedule) > len(best_schedule[0]):
            best_schedule[0] = schedule[:]
        return
    
    for i, meeting in enumerate(available_meetings):
        if is_valid_schedule(schedule, meeting):
            last_meeting = schedule[-1] if schedule else None
            travel_time = travel_times[(last_meeting['location'], meeting['location'])] if last_meeting else 0
            start_time = max(last_meeting['end_time'] + travel_time if last_meeting else time_to_minutes("09:00"), time_to_minutes(meeting['start']))
            end_time = start_time + meeting['duration']
            
            if end_time <= time_to_minutes(meeting['end']):
                schedule.append({
                    "action": "meet",
                    "location": meeting['location'],
                    "person": meeting['name'],
                    "start_time": start_time,
                    "end_time": end_time
                })
                
                backtrack(schedule, available_meetings[:i] + available_meetings[i+1:], best_schedule)
                
                schedule.pop()

def find_optimal_schedule():
    """Find the optimal meeting schedule."""
    best_schedule = [[]]
    backtrack([], meetings, best_schedule)
    
    # Convert times to 'H:MM' format
    for meeting in best_schedule[0]:
        meeting['start_time'] = minutes_to_time(meeting['start_time'])
        meeting['end_time'] = minutes_to_time(meeting['end_time'])
    
    return {"itinerary": best_schedule[0]}
