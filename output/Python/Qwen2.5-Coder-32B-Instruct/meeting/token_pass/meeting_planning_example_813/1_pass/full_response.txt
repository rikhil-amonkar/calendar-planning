# Function to convert minutes since midnight to HH:MM format
def minutes_to_hhmm(minutes):
    hours = minutes // 60
    minutes = minutes % 60
    return f"{hours}:{minutes}" if minutes >= 10 else f"{hours}:{minutes}"

# Dynamic programming to find the optimal schedule
def find_optimal_schedule():
    start_time = 9 * 60  # 9:00 AM
    locations = list(travel_times.keys())
    n = len(locations)
    
    # dp[time][location] -> (max_meetings, path)

    for t in range(start_time, 24*60):
        for loc in locations:
            max_meetings, path = dp[t][loc]
            for next_loc in locations:
                travel_time = travel_times[loc][next_loc]
                if t + travel_time >= 24*60:
                    continue

                new_meetings = max_meetings
                for friend, info in friends.items():
                    if info['location'] == next_loc and info['start'] <= t + travel_time and info['end'] >= t + travel_time + info['min_duration']:

                        })

                    dp[t + travel_time][next_loc] = (new_meetings, new_path)

    best_path = []
    for t in range(start_time, 24*60):
        for loc in locations:
            if dp[t][loc][0] > best_meetings:
                best_meetings = dp[t][loc][0]
                best_path = dp[t][loc][1]
    
    return best_path

# Generate the optimal schedule
