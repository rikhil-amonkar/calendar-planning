# Meeting constraints
constraints = [
    {"name": "Emily", "location": "Richmond District", "start_time": "19:00", "end_time": "21:00", "min_duration": 15},
    {"name": "Margaret", "location": "Financial District", "start_time": "16:30", "end_time": "20:15", "min_duration": 75},

    {"name": "Jeffrey", "location": "Golden Gate Park", "start_time": "11:15", "end_time": "14:30", "min_duration": 120}

def time_to_minutes(time_str):
    return int(time_str[:2]) * 60 + int(time_str[3:])

def minutes_to_time(minutes):
    hours = minutes // 60
    minutes = minutes % 60
    return f"{hours}:{minutes:02}"

def find_optimal_schedule(start_location, start_time, constraints, travel_times):
    current_location = start_location
    current_time = time_to_minutes(start_time)
    itinerary = []

    # Sort constraints by start time
    constraints.sort(key=lambda x: time_to_minutes(x['start_time']))

    for constraint in constraints:
        person_name = constraint['name']
        location = constraint['location']
        start_time_minutes = time_to_minutes(constraint['start_time'])
        end_time_minutes = time_to_minutes(constraint['end_time'])
        min_duration = constraint['min_duration']

        # Calculate travel time to the next location
        if current_location != location:
            travel_time = travel_times[(current_location, location)]
            current_time += travel_time

        # Check if we can meet the person for the required duration
        if current_time + min_duration <= end_time_minutes:
            meeting_start_time = max(current_time, start_time_minutes)
            meeting_end_time = meeting_start_time + min_duration
            itinerary.append({
                "action": "meet",
                "location": location,
                "person": person_name,
                "start_time": minutes_to_time(meeting_start_time),
                "end_time": minutes_to_time(meeting_end_time)
            })
            current_time = meeting_end_time
            current_location = location

    return itinerary

# Generate the optimal schedule
optimal_schedule = find_optimal_schedule("Nob Hill", "9:00", constraints, travel_times)

# Output the result as JSON
result = {"itinerary": optimal_schedule}
print(json.dumps(result, indent=2))
