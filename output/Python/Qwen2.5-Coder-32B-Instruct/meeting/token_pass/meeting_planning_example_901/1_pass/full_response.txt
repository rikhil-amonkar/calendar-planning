def parse_time(time_str):
    return datetime.strptime(time_str, '%H:%M')

def format_time(time_obj):
    return time_obj.strftime('%H:%M')

def find_schedule(start_location, start_time):
    current_location = start_location
    current_time = parse_time(start_time)
    itinerary = []
    
    # Sort meetings by start time
    sorted_meetings = sorted(meetings.items(), key=lambda x: parse_time(x[1]['start']))
    
    for name, details in sorted_meetings:
        meeting_start = parse_time(details['start'])
        meeting_end = parse_time(details['end'])
        min_duration = details['min_duration']
        location = details['location']
        
        # Calculate travel time to the meeting location
        travel_time = travel_times.get((current_location, location), float('inf'))
        meeting_time_required = timedelta(minutes=min_duration)
        travel_time_delta = timedelta(minutes=travel_time)
        
        # Check if we can reach the meeting on time
        if current_time + travel_time_delta <= meeting_start:
            meeting_start = max(current_time + travel_time_delta, meeting_start)
            meeting_end = meeting_start + meeting_time_required
            
            # Check if the meeting fits within the person's availability
            if meeting_end <= meeting_end:
                itinerary.append({
                    "action": "meet",
                    "location": location,
                    "person": name,
                    "start_time": format_time(meeting_start),
                    "end_time": format_time(meeting_end)
                })
                current_time = meeting_end
                current_location = location
    
    return itinerary

schedule = find_schedule('Russian Hill', '9:00')
