def find_meeting_time(raymond_schedule, billy_schedule, donald_schedule, meeting_duration=0.5):
    # Convert work hours to minutes for easier calculations
    work_start = 9 * 60  # 9:00 AM
    work_end = 17 * 60   # 5:00 PM

    # Convert meeting duration to minutes
    meeting_minutes = int(meeting_duration * 60)

    # Convert schedules to lists of tuples (start, end) in minutes
    raymond_blocked = [(9*60, 9*60+30), (11*60+30, 12*60), (13*60, 13*60+30), (15*60, 15*60+30)]
    billy_blocked = [(10*60, 10*60+30), (12*60, 13*60), (16*60+30, 17*60)]
    donald_blocked = [(9*60, 9*60+30), (10*60, 11*60), (12*60, 13*60), (14*60, 14*60+30), (16*60, 17*60)]

    # Function to check if a time slot is free for all participants
    def is_slot_free(slot_start, slot_end):
        for start, end in raymond_blocked + billy_blocked + donald_blocked:
            if not (slot_end <= start or slot_start >= end):
                return False
        return True

    # Iterate over possible half-hour slots
    for start in range(work_start, work_end - meeting_minutes + 1, 30):
        end = start + meeting_minutes
        if is_slot_free(start, end):
            # Convert start and end back to HH:MM format
            start_time = f"{start // 60:02}:{start % 60:02}"
            end_time = f"{end // 60:02}:{end % 60:02}"
            return f"{start_time}:{end_time} Monday"
