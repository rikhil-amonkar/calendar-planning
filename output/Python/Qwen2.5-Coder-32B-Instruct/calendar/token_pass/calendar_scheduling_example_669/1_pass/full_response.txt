from datetime import datetime, timedelta

def parse_schedule(schedule_str):
    """Parses a string of busy times into a list of tuples (start, end)"""
    busy_times = []
    for period in schedule_str.split(';'):
        start, end = period.strip().split(' to ')
        busy_times.append((datetime.strptime(start, '%H:%M'), datetime.strptime(end, '%H:%M')))
    return busy_times

def get_free_slots(busy_times, work_start, work_end):
    """Generates free slots from busy times within work hours"""
    current_time = work_start
    free_slots = []
    for start, end in busy_times:
        if current_time < start:
            free_slots.append((current_time, start))
        current_time = max(current_time, end)
    if current_time < work_end:
        free_slots.append((current_time, work_end))
    return free_slots

def find_meeting_slot(jean_busy, doris_busy, work_start, work_end, meeting_duration, doris_avoid_after):
    """Finds a suitable meeting slot considering all constraints"""
    jean_free = get_free_slots(jean_busy, work_start, work_end)
    doris_free = get_free_slots(doris_busy, work_start, work_end)
    
    for jean_start, jean_end in jean_free:
        for doris_start, doris_end in doris_free:
            # Find overlap
            meeting_start = max(jean_start, doris_start)
            meeting_end = min(jean_end, doris_end)
            if meeting_start + meeting_duration <= meeting_end:
                # Check Doris's preference
                if meeting_start < doris_avoid_after:
                    return meeting_start.strftime('%H:%M'), meeting_end.strftime('%H:%M')
    return None
