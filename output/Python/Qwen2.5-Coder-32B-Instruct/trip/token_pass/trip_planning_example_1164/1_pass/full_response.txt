import json
from collections import defaultdict

# Define the constraints
constraints = {
    "Reykjavik": {"days": 2, "meet_friend": (3, 4)},
    "Stockholm": {"days": 2, "meet_friend": (4, 5)},
    "Porto": {"days": 5, "wedding": (13, 17)},
    "Nice": {"days": 3},
    "Venice": {"days": 4},
    "Vienna": {"days": 3, "workshop": (11, 13)},
    "Split": {"days": 3},
    "Copenhagen": {"days": 2}
}

# Define the flight network as a graph
flight_network = {
    "Copenhagen": ["Vienna", "Stockholm", "Split", "Nice", "Venice"],
    "Nice": ["Stockholm", "Reykjavik", "Porto", "Vienna", "Copenhagen"],
    "Split": ["Copenhagen", "Vienna"],
    "Reykjavik": ["Nice", "Vienna", "Copenhagen", "Stockholm"],
    "Stockholm": ["Copenhagen", "Nice", "Reykjavik", "Vienna", "Split"],
    "Venice": ["Nice", "Vienna", "Copenhagen"],
    "Vienna": ["Copenhagen", "Nice", "Venice", "Reykjavik", "Stockholm", "Split", "Porto"],
    "Porto": ["Nice", "Vienna", "Copenhagen"]
}

def find_path(current_city, current_day, itinerary):
    if current_day > 17:
        return False
    
    # Add current city to the itinerary if it's not already added
    if current_city not in [entry['place'] for entry in itinerary]:
        itinerary.append({"day_range": f"Day {current_day}-{current_day + constraints[current_city]['days'] - 1}", "place": current_city})
    
    # Check if all constraints are met
    if all([constraint in itinerary for constraint in constraints]):
        return True
    
    # Try to move to the next city
    for next_city in flight_network[current_city]:
        if next_city not in [entry['place'] for entry in itinerary]:
            if find_path(next_city, current_day + constraints[current_city]['days'], itinerary):
                return True
    
    return False

def plan_itinerary():
    itinerary = []
    
    # Start from Nice as it has no specific constraints other than being 3 days
    if find_path("Nice", 1, itinerary):
        return itinerary
    else:
        return None

# Generate the itinerary
itinerary = plan_itinerary()

# Validate and format the output
if itinerary:
    # Sort the itinerary by day range
    itinerary.sort(key=lambda x: int(x['day_range'].split('-')[0].split(' ')[1]))
    
    # Output the itinerary as JSON
    print(json.dumps({"itinerary": itinerary}, indent=4))
else:
    print("No valid itinerary found.")
