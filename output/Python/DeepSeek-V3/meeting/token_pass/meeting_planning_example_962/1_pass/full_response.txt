def parse_time(time_str):
    """Convert time string like '9:00AM' or '7:45PM' to datetime object"""
    time_str = time_str.strip()
    # Remove AM/PM and convert to 24-hour format
    if 'AM' in time_str:
        time_str = time_str.replace('AM', '').strip()
        hour, minute = map(int, time_str.split(':'))
        if hour == 12:  # 12 AM is 00:00
            hour = 0
    else:  # PM
        time_str = time_str.replace('PM', '').strip()
        hour, minute = map(int, time_str.split(':'))
        if hour != 12:
            hour += 12
    
    # Use a fixed date for calculations
    return datetime(2024, 1, 1, hour, minute)

def format_time(dt):
    """Convert datetime to 'H:MM' format"""

3. **Evaluating schedules**: Scores each schedule based on the number of meetings and total meeting minutes.

4. **Optimizing timing**: Adjusts meeting times within their availability windows to maximize the number of meetings that can be scheduled.

5. **Outputting results**: Formats the optimal schedule as JSON in the required format.
