def parse_time(time_str):
    """Convert time string to minutes since midnight"""
    if 'AM' in time_str or 'PM' in time_str:
        dt = datetime.strptime(time_str, '%I:%M%p')
    else:
        dt = datetime.strptime(time_str, '%H:%M')
    return dt.hour * 60 + dt.minute

def format_time(minutes):
    """Convert minutes since midnight to time string"""
    hours = minutes // 60
    mins = minutes % 60
    return f"{hours}:{mins:02d}"

    def find_best_schedule(self):
        """Find the optimal meeting schedule using backtracking"""
        best_schedule = []
        max_meetings = 0
        max_duration = 0
        
        # Try different meeting orders (limit to permutations of 5 meetings to avoid combinatorial explosion)
        meeting_indices = list(range(len(self.meetings)))
        
        # We'll use a greedy approach with backtracking
        def backtrack(current_time, current_loc, visited, schedule, remaining_meetings):
            nonlocal best_schedule, max_meetings, max_duration
            
            # If we've tried all meetings or it's getting late (past 10 PM)
            if not remaining_meetings or current_time >= parse_time("10:00PM"):
                # Evaluate this schedule
                num_meetings = len(schedule)
                total_duration = sum(m["duration"] for m in schedule)
                
                # Prefer more meetings, then longer total meeting time
                if (num_meetings > max_meetings or 
                    (num_meetings == max_meetings and total_duration > max_duration)):
                    max_meetings = num_meetings
                    max_duration = total_duration
                    best_schedule = schedule.copy()
                return
            
            # Try each remaining meeting
            for i, meeting_idx in enumerate(remaining_meetings):
                meeting = self.meetings[meeting_idx]
                
                # Calculate travel time to this meeting
                travel_time = self.get_travel_time(current_loc, meeting["location"])
                arrival_time = current_time + travel_time
                
                # Check if we can arrive during availability window
                if arrival_time < meeting["available_start"]:
                    # We can wait until they're available
                    start_time = meeting["available_start"]
                elif arrival_time <= meeting["available_end"] - meeting["min_duration"]:
                    # We can start immediately or shortly after arrival
                    start_time = arrival_time
                else:
                    # Cannot meet this person
                    continue
                
                # Check if we have enough time for minimum duration
                end_time = start_time + meeting["min_duration"]
                if end_time > meeting["available_end"]:
                    continue
                
                # Add this meeting to schedule
                new_schedule = schedule + [{
                    "person": meeting["person"],
                    "location": meeting["location"],
                    "start_time": start_time,
                    "end_time": end_time,
                    "duration": meeting["min_duration"]
                }]
                
                # Continue with remaining meetings
                new_remaining = remaining_meetings[:i] + remaining_meetings[i+1:]
                backtrack(end_time, meeting["location"], visited | {meeting_idx}, 
                         new_schedule, new_remaining)
        
        # Start backtracking
        backtrack(self.start_time, self.current_location, set(), [], meeting_indices)
        
        # Try to add more meetings by adjusting durations
        improved = True
        while improved:
            improved = False
            for i in range(len(best_schedule)):
                # Try to extend each meeting if possible
                meeting_info = None
                for m in self.meetings:
                    if m["person"] == best_schedule[i]["person"]:
                        meeting_info = m
                        break
                
                if not meeting_info:
                    continue
                
                # Calculate maximum possible extension
                current_end = best_schedule[i]["end_time"]
                max_end = meeting_info["available_end"]
                
                # Check next meeting constraint
                if i < len(best_schedule) - 1:
                    next_start = best_schedule[i + 1]["start_time"]
                    travel_to_next = self.get_travel_time(
                        best_schedule[i]["location"], 
                        best_schedule[i + 1]["location"]
                    )
                    max_end = min(max_end, next_start - travel_to_next)
                
                # Extend if possible
                if current_end < max_end:
                    extension = max_end - current_end
                    if extension > 0:
                        best_schedule[i]["end_time"] = max_end
                        best_schedule[i]["duration"] += extension
                        improved = True
        
        return best_schedule
    
    def create_itinerary(self, schedule):
        """Convert schedule to itinerary format"""
        itinerary = []
        for meeting in schedule:
            itinerary.append({
                "action": "meet",
                "location": meeting["location"],
                "person": meeting["person"],
                "start_time": format_time(meeting["start_time"]),
                "end_time": format_time(meeting["end_time"])
            })

    def solve(self):
        """Main solving function"""
        schedule = self.find_best_schedule()
        
        # Sort schedule by start time
        schedule.sort(key=lambda x: x["start_time"])
        
        itinerary = self.create_itinerary(schedule)
        
        return {
            "itinerary": itinerary,
            "total_meetings": len(schedule),
            "total_meeting_minutes": sum(m["duration"] for m in schedule)
        }

This program uses a backtracking approach to find the optimal meeting schedule:

1. **Travel Time Matrix**: All travel times are stored in a matrix for quick lookup.

2. **Meeting Constraints**: Each friend's availability and minimum meeting duration are defined.

3. **Backtracking Algorithm**: The program explores different meeting orders, ensuring:
   - Travel times are accounted for between meetings
   - Meetings occur within each friend's availability window
   - Minimum durations are respected

4. **Optimization**: After finding a schedule with maximum meetings, it tries to extend meeting durations where possible.

5. **Output**: The result is formatted as required JSON with 24-hour time format.

The algorithm prioritizes maximizing the number of meetings first, then maximizes total meeting time. It accounts for all travel constraints and ensures meetings don't overlap or violate time windows.

When run, this program will compute and output the optimal meeting schedule based on the given constraints.
