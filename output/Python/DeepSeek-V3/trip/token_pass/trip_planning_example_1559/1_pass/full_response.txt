import json
from typing import List, Dict, Tuple, Optional

class TripPlanner:
    def __init__(self):
        # City visit requirements (city: days needed)
        self.city_days = {
            'Valencia': 2,
            'Oslo': 3,
            'Lyon': 4,
            'Prague': 3,
            'Paris': 4,
            'Nice': 4,
            'Seville': 5,

            'Mykonos': 5,
            'Lisbon': 2
        }
        
        # Special constraints
        self.special_constraints = {
            'Valencia': {'days': [3, 4], 'type': 'meet_friends'},
            'Seville': {'days': [5, 6, 7, 8, 9], 'type': 'annual_show'},
            'Oslo': {'days': [13, 14, 15], 'type': 'meet_friend'},
            'Mykonos': {'days': [21, 22, 23, 24, 25], 'type': 'wedding'}
        }
        
        # Direct flight connections (undirected graph)
        self.flights = {
            'Lisbon': ['Paris', 'Seville', 'Prague', 'Valencia', 'Nice', 'Oslo', 'Lyon'],
            'Paris': ['Lisbon', 'Oslo', 'Valencia', 'Nice', 'Lyon', 'Tallinn', 'Prague', 'Seville'],
            'Lyon': ['Nice', 'Prague', 'Paris', 'Valencia', 'Oslo'],
            'Nice': ['Lyon', 'Paris', 'Oslo', 'Lisbon', 'Mykonos'],
            'Oslo': ['Tallinn', 'Paris', 'Prague', 'Nice', 'Lyon', 'Lisbon'],
            'Prague': ['Lyon', 'Lisbon', 'Oslo', 'Paris', 'Valencia', 'Tallinn'],
            'Seville': ['Lisbon', 'Paris', 'Valencia'],
            'Tallinn': ['Oslo', 'Paris', 'Prague'],
            'Valencia': ['Paris', 'Lisbon', 'Lyon', 'Prague', 'Seville'],
            'Mykonos': ['Nice']
        }
        
        # Total days
        self.total_days = 25
        
    def find_path(self, current_city: str, remaining_cities: set, 
                  current_day: int, itinerary: List[Dict], 
                  city_allocated_days: Dict[str, int]) -> Optional[List[Dict]]:
        """
        Recursive backtracking to find valid itinerary
        """
        # Base case: all days used
        if current_day > self.total_days:
            # Check if all cities have required days
            if all(city_allocated_days.get(city, 0) == self.city_days[city] 
                   for city in self.city_days):
                return itinerary
            return None
        
        # Base case: all cities visited with required days
        if all(city_allocated_days.get(city, 0) == self.city_days[city] 
               for city in self.city_days):
            # Fill remaining days with current city
            if current_day <= self.total_days:
                remaining = self.total_days - current_day + 1
                if remaining > 0:
                    itinerary.append({
                        'day_range': f'Day {current_day}-{self.total_days}',
                        'place': current_city
                    })
                    city_allocated_days[current_city] = city_allocated_days.get(current_city, 0) + remaining
                return itinerary
            return None
        
        # Try to stay in current city if we need more days here
        current_city_needed = self.city_days[current_city] - city_allocated_days.get(current_city, 0)
        
        # Check special constraints for current day
        must_move = False
        target_city = None
        
        for city, constraint in self.special_constraints.items():
            if current_day in constraint['days']:
                if city != current_city:
                    must_move = True
                    target_city = city
                    break
        
        if must_move and target_city:
            # Must move to target city
            if target_city in self.flights[current_city]:
                # Calculate days to stay in current city before moving
                days_here = 0
                if current_city_needed > 0:
                    # Stay as many days as possible before the constraint day
                    days_here = min(current_city_needed, target_city_days[0] - current_day)
                
                if days_here > 0:
                    itinerary.append({
                        'day_range': f'Day {current_day}-{current_day + days_here - 1}',
                        'place': current_city
                    })
                    city_allocated_days[current_city] = city_allocated_days.get(current_city, 0) + days_here
                    current_day += days_here
                
                # Move to target city
                itinerary.append({
                    'day_range': f'Day {current_day}',
                    'place': f'Travel from {current_city} to {target_city}'
                })
                
                # Start stay in target city
                target_needed = self.city_days[target_city] - city_allocated_days.get(target_city, 0)
                days_there = 1  # At least the travel day counts
                
                # Calculate how many consecutive days we can stay
                constraint_days = [d for d in self.special_constraints[target_city]['days'] if d >= current_day]
                if constraint_days:
                    days_there = max(days_there, len(constraint_days))
                
                days_there = min(days_there, target_needed)
                
                itinerary.append({
                    'day_range': f'Day {current_day}-{current_day + days_there - 1}',
                    'place': target_city
                })
                city_allocated_days[target_city] = city_allocated_days.get(target_city, 0) + days_there
                
                result = self.find_path(target_city, remaining_cities, 
                                       current_day + days_there, itinerary.copy(), 
                                       city_allocated_days.copy())
                if result:
                    return result
            return None
        
        # If we need to stay in current city
        if current_city_needed > 0:
            # Calculate maximum consecutive days we can stay
            max_stay = current_city_needed
            
            # Check for upcoming constraints
            for day in range(current_day, current_day + current_city_needed + 1):
                if day > self.total_days:
                    max_stay = min(max_stay, self.total_days - current_day + 1)
                    break
                    
                for city, constraint in self.special_constraints.items():
                    if day in constraint['days'] and city != current_city:
                        max_stay = min(max_stay, day - current_day)
                        break
            
            if max_stay > 0:
                new_itinerary = itinerary.copy()
                new_allocated = city_allocated_days.copy()
                
                new_itinerary.append({
                    'day_range': f'Day {current_day}-{current_day + max_stay - 1}',
                    'place': current_city
                })
                new_allocated[current_city] = new_allocated.get(current_city, 0) + max_stay
                
                # Try continuing from here
                result = self.find_path(current_city, remaining_cities, 
                                       current_day + max_stay, new_itinerary, new_allocated)
                if result:
                    return result
        
        # Try moving to another city
        for next_city in self.flights[current_city]:
            if next_city == current_city:
                continue
                
            next_city_needed = self.city_days[next_city] - city_allocated_days.get(next_city, 0)
            
            if next_city_needed > 0:
                # Check if we can move to this city
                # Add travel day
                new_itinerary = itinerary.copy()
                new_allocated = city_allocated_days.copy()
                
                new_itinerary.append({
                    'day_range': f'Day {current_day}',
                    'place': f'Travel from {current_city} to {next_city}'
                })
                
                # Calculate how many days to stay in next city
                days_there = next_city_needed
                
                # Check constraints
                for day in range(current_day + 1, current_day + next_city_needed + 1):
                    if day > self.total_days:
                        days_there = min(days_there, self.total_days - current_day)
                        break
                        
                    for city, constraint in self.special_constraints.items():
                        if day in constraint['days'] and city != next_city:
                            days_there = min(days_there, day - current_day - 1)
                            break
                
                if days_there > 0:
                    new_itinerary.append({
                        'day_range': f'Day {current_day + 1}-{current_day + days_there}',
                        'place': next_city
                    })
                    new_allocated[next_city] = new_allocated.get(next_city, 0) + days_there + 1  # +1 for travel day
                    
                    result = self.find_path(next_city, remaining_cities, 
                                           current_day + days_there + 1, new_itinerary, new_allocated)
                    if result:
                        return result
        
        return None
    
    def plan_itinerary(self) -> Dict:
        """
        Main function to plan the itinerary
        """
        # Start from any city that has flights to others
        start_city = 'Lisbon'  # Good starting point with many connections
        
        # Initialize tracking
        itinerary = []
        city_allocated_days = {}
        
        # Find path using backtracking
        result = self.find_path(start_city, set(self.city_days.keys()), 
                               1, itinerary, city_allocated_days)
        
        if not result:
            # If backtracking fails, use heuristic approach
            return self.heuristic_plan()
        
        # Process result to merge consecutive stays in same city
        processed_itinerary = []
        current_entry = None
        
        for entry in result:
            if 'Travel' in entry['place']:
                if current_entry:
                    processed_itinerary.append(current_entry)
                    current_entry = None
                processed_itinerary.append(entry)
            else:
                if current_entry and current_entry['place'] == entry['place']:
                    # Merge consecutive stays
                    end_day = int(entry['day_range'].split('-')[1].replace('Day ', ''))
                    current_entry['day_range'] = f"{current_entry['day_range'].split('-')[0]}-{end_day}"
                else:
                    if current_entry:
                        processed_itinerary.append(current_entry)
                    current_entry = entry
        
        if current_entry:
            processed_itinerary.append(current_entry)
        
        return {'itinerary': processed_itinerary}
    
    def heuristic_plan(self) -> Dict:
        """
        Heuristic approach when backtracking fails
        """
        # Build itinerary based on constraints
        itinerary = []
        
        # Day 1-2: Start in Lisbon (2 days)
        itinerary.append({'day_range': 'Day 1-2', 'place': 'Lisbon'})

    def validate_itinerary(self, itinerary: List[Dict]) -> bool:
        """
        Validate that the itinerary meets all constraints
        """
        city_days_count = {city: 0 for city in self.city_days}
        current_day = 1
        
        for entry in itinerary:
            if 'Travel' in entry['place']:
                # Travel day counts for both cities
                parts = entry['place'].replace('Travel from ', '').replace(' to ', ' ').split()
                if len(parts) >= 3:
                    from_city = parts[0]
                    to_city = parts[2]
                    city_days_count[from_city] = city_days_count.get(from_city, 0) + 1
                    city_days_count[to_city] = city_days_count.get(to_city, 0) + 1
                current_day += 1
            else:
                day_range = entry['day_range']
                city = entry['place']
                
                if '-' in day_range:
                    start, end = map(int, day_range.replace('Day ', '').split('-'))
                    days = end - start + 1
                else:
                    days = 1
                
                city_days_count[city] = city_days_count.get(city, 0) + days
                current_day += days
        
        # Check if all cities have required days
        for city, required in self.city_days.items():
            if city_days_count.get(city, 0) < required:
                print(f"City {city} has {city_days_count.get(city, 0)} days but needs {required}")
                return False
        
        return True
