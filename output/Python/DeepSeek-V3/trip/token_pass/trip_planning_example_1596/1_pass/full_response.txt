def validate_itinerary(itinerary: List[Dict], required_days: Dict[str, int], 
                      fixed_constraints: List[Tuple[int, int, str]], graph: Dict[str, List[str]]) -> bool:
    """Validate that the itinerary meets all constraints."""

    city_days = {city: 0 for city in required_days.keys()}

            start_day = int(day_range.split(" ")[1].split("-")[0])

            days = 1

        if city_days.get(city, 0) != required:
            return False
    
    # Check fixed constraints
    for start_day, end_day, city in fixed_constraints:
        found = False
        for segment in itinerary:
            seg_city = segment["place"]
            day_range = segment["day_range"]
            
            if "-" in day_range:
                seg_start = int(day_range.split(" ")[1].split("-")[0])
                seg_end = int(day_range.split("-")[1])
                
                # Check if this segment covers the required days
                if seg_city == city:
                    # The segment must cover all required days
                    if seg_start <= start_day and seg_end >= end_day:
                        found = True
                        break
            else:
                seg_day = int(day_range.split(" ")[1])
                if seg_city == city and seg_day == start_day == end_day:
                    found = True
                    break
        
        if not found:
            return False
    
    # Check connectivity between consecutive segments
    for i in range(len(itinerary) - 1):
        city1 = itinerary[i]["place"]
        city2 = itinerary[i + 1]["place"]
        
        if city1 != city2 and not has_direct_flight(graph, city1, city2):
            return False
    
    return True
