import json
from itertools import permutations

def solve_puzzle():
    # Define the possible values for each category
    names = ["Eric", "Peter", "Arnold"]
    mothers = ["Holly", "Aniya", "Janelle"]
    foods = ["pizza", "grilled cheese", "spaghetti"]
    
    houses = [1, 2, 3]
    
    # Generate all possible permutations for each category
    for name_perm in permutations(names, 3):
        for mother_perm in permutations(mothers, 3):
            for food_perm in permutations(foods, 3):
                # Create assignment dictionaries
                assignment = {}
                for i in range(3):
                    house = houses[i]
                    assignment[house] = {
                        'Name': name_perm[i],
                        'Mother': mother_perm[i],
                        'Food': food_perm[i]
                    }
                
                # Check all clues
                # Clue 1: The person who loves the spaghetti eater and Peter are next to each other.
                spaghetti_house = None
                peter_house = None
                for house in houses:
                    if assignment[house]['Food'] == 'spaghetti':
                        spaghetti_house = house
                    if assignment[house]['Name'] == 'Peter':
                        peter_house = house
                
                if spaghetti_house is None or peter_house is None:
                    continue
                
                if abs(spaghetti_house - peter_house) != 1:
                    continue
                
                # Clue 2: The person who loves eating grilled cheese is directly left of 
                # The person whose mother's name is Aniya.
                grilled_cheese_house = None
                aniya_house = None
                for house in houses:
                    if assignment[house]['Food'] == 'grilled cheese':
                        grilled_cheese_house = house
                    if assignment[house]['Mother'] == 'Aniya':
                        aniya_house = house
                
                if grilled_cheese_house is None or aniya_house is None:
                    continue
                
                if grilled_cheese_house + 1 != aniya_house:
                    continue
                
                # Clue 3: The person who loves eating grilled cheese is Eric.
                if assignment[grilled_cheese_house]['Name'] != 'Eric':
                    continue
                
                # Clue 4: Peter is The person whose mother's name is Holly.
                if assignment[peter_house]['Mother'] != 'Holly':
                    continue
                
                # All clues satisfied - we found the solution
                solution = {
                    "solution": {
                        "header": ["House", "Name", "Mother", "Food"],
                        "rows": []
                    }
                }
                
                for house in houses:
                    row = [
                        str(house),
                        assignment[house]['Name'],
                        assignment[house]['Mother'],
                        assignment[house]['Food']
                    ]
                    solution["solution"]["rows"].append(row)
                
                return solution
    
    # If no solution found (shouldn't happen with valid puzzle)
    return {"solution": {"header": ["House", "Name", "Mother", "Food"], "rows": []}}

if __name__ == "__main__":
    result = solve_puzzle()
    print(json.dumps(result, indent=2))
