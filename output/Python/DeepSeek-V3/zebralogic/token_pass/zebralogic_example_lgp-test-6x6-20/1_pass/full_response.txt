def solve():

                                    eric_house = h

                                valid = False

                            for h in houses:
                                if assignment[h]['Music'] == 'pop' and h == 3:
                                    valid = False

                            if not (assignment[eric_house]['Music'] == 'country'):
                                valid = False

                            bob_house = None
                            for h in houses:
                                if assignment[h]['Name'] == 'Bob':
                                    bob_house = h
                            if not (assignment[bob_house]['Drink'] == 'coffee'):
                                valid = False

                            if not (root_beer_house is not None and janelle_house is not None and janelle_house == root_beer_house + 1):
                                valid = False

                            sarah_house = None

                                valid = False
